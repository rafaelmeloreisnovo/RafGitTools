# S02_C: COERÊNCIA ESTÉREO entre canais L/R → direção de fonte
# Usa diferença de fase para triangular ângulo de chegada
import numpy as np
def gcc_phat(L, R, sr=16000):
    """Generalized Cross-Correlation Phase Transform"""
    n = len(L) + len(R)
    L_fft, R_fft = np.fft.rfft(L, n), np.fft.rfft(R, n)

seed_S03_A_kalman_6dof() { cat << 'SEED'
# S03_A: KALMAN FILTER → fusão 6-DOF (acc+gyr) → pose 3D estável
import numpy as np
class Kalman6DOF:
    def __init__(self, dt=0.02):
        self.x = np.zeros(6)  # [roll, pitch, yaw, vRoll, vPitch, vYaw]
        self.P = np.eye(6) * 0.1
        self.Q = np.eye(6) * 0.001  # process noise
        self.R = np.eye(3) * 0.05   # measurement noise
        self.F = np.eye(6); self.F[0,3]=dt; self.F[1,4]=dt; self.F[2,5]=dt
        self.H = np.zeros((3,6)); self.H[0,0]=1; self.H[1,1]=1; self.H[2,2]=1
    def step(self, acc, gyr):
        # Predict
        self.x = self.F @ self.x
        self.P = self.F @ self.P @ self.F.T + self.Q
        # Measure roll/pitch from accelerometer
        roll = np.arctan2(acc[1], acc[2])
        pitch = np.arctan2(-acc[0], np.sqrt(acc[1]**2+acc[2]**2))
        z = np.array([roll, pitch, self.x[2] + gyr[2]*0.02])
        # Update
        y = z - self.H @ self.x
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        self.x = self.x + K @ y
        self.P = (np.eye(6) - K @ self.H) @ self.P
        return self.x[:3]  # roll, pitch, yaw em rad

seed_S03_B_madgwick_ahrs() { cat << 'SEED'
# S03_B: MADGWICK AHRS — quaternion de orientação absoluta
# Mais leve que Kalman, sem matriz; ótimo para ARM
import numpy as np
class Madgwick:
    def __init__(self, beta=0.041, dt=0.02):
        self.q = np.array([1.0, 0.0, 0.0, 0.0])
        self.beta = beta; self.dt = dt
    def update_imu(self, gx, gy, gz, ax, ay, az):
        q = self.q
        norm = np.linalg.norm([ax, ay, az])
        if norm < 1e-6: return q
        ax, ay, az = ax/norm, ay/norm, az/norm
        F = np.array([
            2*(q[1]*q[3] - q[0]*q[2]) - ax,
            2*(q[0]*q[1] + q[2]*q[3]) - ay,
            2*(0.5 - q[1]**2 - q[2]**2) - az
        ])
        J = np.array([
            [-2*q[2], 2*q[3], -2*q[0], 2*q[1]],
            [2*q[1], 2*q[0], 2*q[3], 2*q[2]],
            [0, -4*q[1], -4*q[2], 0]
        ])
        step = J.T @ F
        step = step / (np.linalg.norm(step) + 1e-9)
        qdot = 0.5 * np.array([
            -q[1]*gx - q[2]*gy - q[3]*gz,
             q[0]*gx + q[2]*gz - q[3]*gy,
             q[0]*gy - q[1]*gz + q[3]*gx,
             q[0]*gz + q[1]*gy - q[2]*gx
        ]) - self.beta * step
        self.q = self.q + qdot * self.dt
        self.q = self.q / np.linalg.norm(self.q)
        return self.q
SEED
}

# Roda telefone em 8 e ajusta offsets para que esfera de leituras seja centrada
import numpy as np
def fit_sphere(points):
    # Ajusta esfera (x-a)² + (y-b)² + (z-c)² = r²
    A = np.column_stack([2*points, np.ones(len(points))])
    b = (points**2).sum(axis=1)

import numpy as np
class GPSKalman:
    def __init__(self):
        self.x = np.zeros(4)  # [lat, lon, vlat, vlon]
        self.P = np.eye(4) * 1e-4
        self.Q = np.eye(4) * 1e-8
    def update(self, lat, lon, vlat=0, vlon=0, acc_m=10):
        R = np.eye(2) * (acc_m / 111000)**2  # m → graus

import numpy as np
class HMM42:
    def __init__(self, n_states=42):

class ActiveInference:

import numpy as np
class QuantumOmega:
    def __init__(self, n_attractors=42):
        # Amplitudes complexas; |ψ|² = probabilidade

# S14_D: AUTH POR VOZ — speaker verification por embedding
import numpy as np
def voice_auth(probe_wav, enrolled_emb, threshold=0.78):
    import librosa
    y, sr = librosa.load(probe_wav, sr=16000)
    probe = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=20).mean(axis=1)
    enrolled = np.load(enrolled_emb)
    sim = np.dot(probe,enrolled)/(np.linalg.norm(probe)*np.linalg.norm(enrolled)+1e-9)
    return {"authorized": sim > threshold, "similarity": float(sim)}

# S15_B: SERVICE MANAGER tipo systemd com unit files + dependencias
import subprocess, time, os, json
class ServiceMgr:
    def __init__(self, units_dir='~/.rafaelia/units'):
        self.units_dir = os.path.expanduser(units_dir)
        self.units = {}; self.procs = {}
    def load(self):
        for f in os.listdir(self.units_dir):
            if f.endswith('.unit'):
                self.units[f[:-5]] = json.load(open(f'{self.units_dir}/{f}'))
    def start(self, name):
        u = self.units[name]
        for dep in u.get('after',[]):
            if dep not in self.procs: self.start(dep)
        self.procs[name] = subprocess.Popen(u['exec'], shell=True)
    def watchdog(self):
        while True:
            for name, p in list(self.procs.items()):
                if p.poll() is not None: self.start(name)
            time.sleep(5)
SEED
}

seed_S15_D_crash_safe_backoff() { cat << 'SEED'
# S15_D: SUPERVISOR com backoff EXPONENCIAL após N crashes
import subprocess, time
class Supervisor:
    def __init__(self, cmd, max_fails=5, window=300):
        self.cmd=cmd; self.max=max_fails; self.window=window; self.fails=[]
    def run(self):
        while True:
            subprocess.Popen(self.cmd, shell=True).wait()
            self.fails.append(time.time())
            self.fails = [t for t in self.fails if time.time()-t < self.window]
            if len(self.fails) >= self.max:
                backoff = 2 ** (len(self.fails) - self.max + 1) * 60
                print(f"Backoff {backoff}s"); time.sleep(backoff)
SEED
}

# S17_A: MANIFEST SHA256 + Merkle root assinado pelo keystore HW
import hashlib, os, json, subprocess
def manifest_dir(d='~/.rafaelia'):
    base = os.path.expanduser(d)
    files = {}
    for root, _, fs in os.walk(base):
        for f in fs:
            p = os.path.join(root, f)
            with open(p,'rb') as h: files[os.path.relpath(p, base)] = \
                hashlib.sha256(h.read()).hexdigest()
    root_hash = hashlib.sha256(''.join(sorted(files.values())).encode()).hexdigest()
    json.dump({"files": files, "root": root_hash},
              open(f'{base}/MANIFEST.json','w'), indent=2)
    sig = subprocess.run(['termux-keystore','sign','raf_root','-'],
                        input=root_hash, capture_output=True, text=True).stdout
    return {"root": root_hash, "signature": sig}

# S17_D: UPDATE DIFERENCIAL — baixa só arquivos mudados
import hashlib, json, os, urllib.request
def diff_update(remote_manifest_url, local_manifest):
    remote = json.loads(urllib.request.urlopen(remote_manifest_url).read())
    local = json.load(open(local_manifest))
    changed = [p for p,rh in remote['files'].items() if local['files'].get(p) != rh]
    base_url = remote_manifest_url.rsplit('/',1)[0]
    for path in changed:
        data = urllib.request.urlopen(f'{base_url}/{path}').read()
        if hashlib.sha256(data).hexdigest() == remote['files'][path]:
            open(os.path.expanduser(f'~/.rafaelia/{path}'),'wb').write(data)
    return changed

        if self.health():
