/* ═══════════════════════════════════════════════════════════════════════════
 * §4 — CRC32C BRANCHLESS (compartilhado por log/audit/config)
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 crc32c(const u8 *p, u32 n){
  u32 c = 0xFFFFFFFFu;
  for(u32 i=0;i<n;i++){
    c ^= (u32)p[i];
    for(u32 j=0;j<8u;j++) c = (c>>1) ^ (0x82F63B78u & -(c&1u));
  }
  return ~c;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §5 — ARENA com CANARY + MEMORY POISON (ESQ3 + U03)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define ARENA_SZ (16u*1024u*1024u)
#define CANARY_MAGIC 0xDEADC0DEu
#define POISON_BYTE  0xDEu
#define MAX_ALLOCS 1024u

typedef struct { u32 offset, size, canary_pre, canary_post; } AllocRec;
static u8 G_ARENA[ARENA_SZ] __attribute__((aligned(64)));
static u32 G_ABUMP = 0u;
static AllocRec G_ALLOC_RECS[MAX_ALLOCS];
static u32 G_N_ALLOCS = 0u;

static void *raf_alloc(u32 n){
  if(n == 0) return (void*)0;
  u32 s = (G_ABUMP + 63u) & ~63u;
  u32 e = s + 4u + n + 4u;
  if(e > ARENA_SZ || G_N_ALLOCS >= MAX_ALLOCS) return (void*)0;
  u32 *pre  = (u32*)(G_ARENA + s);
  u32 *post = (u32*)(G_ARENA + s + 4u + n);
  *pre = CANARY_MAGIC; *post = CANARY_MAGIC;
  G_ALLOC_RECS[G_N_ALLOCS].offset = s;
  G_ALLOC_RECS[G_N_ALLOCS].size = n;
  G_ALLOC_RECS[G_N_ALLOCS].canary_pre = s;
  G_ALLOC_RECS[G_N_ALLOCS].canary_post = s + 4u + n;
  G_N_ALLOCS++; G_ABUMP = e;
  /* U03: poison antes de entregar (detecta uso de dados não-inicializados) */
  u8 *data = G_ARENA + s + 4u;
  for(u32 i=0;i<n;i++) data[i] = POISON_BYTE;
  return data;
}

static RafErr raf_check_canaries(void){
  for(u32 i=0; i<G_N_ALLOCS; i++){
    u32 *pre  = (u32*)(G_ARENA + G_ALLOC_RECS[i].canary_pre);
    u32 *post = (u32*)(G_ARENA + G_ALLOC_RECS[i].canary_post);
    if(*pre != CANARY_MAGIC || *post != CANARY_MAGIC) return RAF_ERR_CANARY;
  }
  return RAF_OK;
}

/* U03: reset com poison — detecta uso após reset */
static void raf_arena_reset_poison(void){
  for(u32 i=0;i<G_ABUMP;i++) G_ARENA[i] = POISON_BYTE;
  G_ABUMP = 0u; G_N_ALLOCS = 0u;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §6 — ATOMIC + BARRIERS (ESQ1)
 * ═══════════════════════════════════════════════════════════════════════════ */
static void raf_barrier(void){
  __asm__ volatile("" ::: "memory");
#if defined(__aarch64__)
  __asm__ volatile("dmb ish" ::: "memory");
#elif defined(__arm__)
  __asm__ volatile("dmb" ::: "memory");
#elif defined(__x86_64__) || defined(__i386__)
  __asm__ volatile("mfence" ::: "memory");
#endif
}
static u32 raf_cas32(volatile u32 *p, u32 e, u32 d){
  return __sync_bool_compare_and_swap(p,e,d) ? 1u : 0u;
}
static u32 raf_atomic_add(volatile u32 *p, u32 v){ return __sync_fetch_and_add(p,v); }
static u64 raf_atomic_add64(volatile u64 *p, u64 v){ return __sync_fetch_and_add(p,v); }
static u32 raf_load_acquire(volatile u32 *p){ u32 v=*p; raf_barrier(); return v; }
static void raf_store_release(volatile u32 *p, u32 v){ raf_barrier(); *p = v; }

/* ═══════════════════════════════════════════════════════════════════════════
 * §7 — SAFE MATH (ESQ4)
 * ═══════════════════════════════════════════════════════════════════════════ */
static RafErr raf_add_u32(u32 a, u32 b, u32 *out){
  u32 r = a + b;
  if(r < a) return RAF_ERR_OVERFLOW;
  *out = r; return RAF_OK;
}
static RafErr raf_mul_u32(u32 a, u32 b, u32 *out){
  if(!a||!b){ *out = 0; return RAF_OK; }
  u64 r = (u64)a*(u64)b;
  if(r > 0xFFFFFFFFull) return RAF_ERR_OVERFLOW;
  *out = (u32)r; return RAF_OK;
}
static RafErr raf_sub_u32(u32 a, u32 b, u32 *out){
  if(b > a) return RAF_ERR_OVERFLOW;
  *out = a - b; return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §8 — TIMING-SAFE COMPARE (ESQ5)
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 raf_ct_eq(const void *a, const void *b, u32 n){
  const u8 *x=a, *y=b; u32 d=0;
  for(u32 i=0;i<n;i++) d |= (u32)(x[i]^y[i]);
  return (d - 1u) >> 31;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §9 — U02: STRING SAFE (strlcpy/strlcat) — vetor #1 de bugs em C
 *   Tamanho de destino respeitado, sempre null-terminado, retorna len de src.
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 raf_strlcpy(char *dst, const char *src, u32 dst_size){
  u32 src_len = _slen(src);
  if(dst_size == 0) return src_len;
  u32 copy = (src_len < dst_size - 1u) ? src_len : dst_size - 1u;
  _mcpy(dst, src, copy);
  dst[copy] = 0;
  return src_len;  /* retorna len original — caller detecta truncate */
}

static u32 raf_strlcat(char *dst, const char *src, u32 dst_size){
  u32 dst_len = 0;
  while(dst_len < dst_size && dst[dst_len]) dst_len++;
  if(dst_len == dst_size) return dst_size + _slen(src);
  u32 src_len = _slen(src);
  u32 avail = dst_size - dst_len - 1u;
  u32 copy = (src_len < avail) ? src_len : avail;
  _mcpy(dst + dst_len, src, copy);
  dst[dst_len + copy] = 0;
  return dst_len + src_len;
}

static u32 raf_strneq(const char *a, const char *b, u32 n){
  for(u32 i=0;i<n;i++){
    if(a[i] != b[i]) return 0u;
    if(!a[i]) return 1u;
  }
  return 1u;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §10 — U01: WATCHDOG TIMER
 *   pet() = renova prazo; check() = vê se expirou; kick() = força reset.
 *   Sem isto: travamentos não detectados podem rodar para sempre.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  volatile u64 last_pet_ns;
  u64 timeout_ns;
  volatile u32 kicked;       /* sinaliza ação de reset solicitada */
  volatile u32 pet_count;
} Watchdog;

static Watchdog G_WDT;

static void raf_wdt_init(Watchdog *w, u32 timeout_ms){
  w->last_pet_ns = _now_ns();
  w->timeout_ns = (u64)timeout_ms * 1000000ULL;
  w->kicked = 0u; w->pet_count = 0u;
}
static void raf_wdt_pet(Watchdog *w){
  w->last_pet_ns = _now_ns();
  raf_atomic_add(&w->pet_count, 1u);
}
static RafErr raf_wdt_check(Watchdog *w){
  u64 elapsed = _now_ns() - w->last_pet_ns;
  if(elapsed > w->timeout_ns){
    raf_store_release(&w->kicked, 1u);
    return RAF_ERR_WATCHDOG;
  }
  return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §11 — U04: DEADLINE PROPAGATION
 *   Toda operação recebe Deadline (timestamp absoluto em ns).
 *   raf_deadline_remaining() retorna ns restantes ou negativo se expirou.
 *   Sem isto: chamadas em cadeia não compartilham budget de tempo.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { u64 deadline_ns; } Deadline;

static Deadline raf_deadline_from_ms(u32 ms){
  Deadline d; d.deadline_ns = _now_ns() + (u64)ms * 1000000ULL; return d;
}
static i64 raf_deadline_remaining_ns(const Deadline *d){
  u64 now = _now_ns();
  if(d->deadline_ns <= now) return -1;
  return (i64)(d->deadline_ns - now);
}
static RafErr raf_deadline_check(const Deadline *d){
  return raf_deadline_remaining_ns(d) < 0 ? RAF_ERR_DEADLINE : RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §12 — U05: CIRCUIT BREAKER
 *   Após N falhas em janela, abre circuito por T segundos.
 *   Estados: CLOSED (normal), OPEN (rejeita), HALF_OPEN (testa 1 chamada).
 *   Sem isto: dependência falhando vira cascata de timeouts.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef enum { CB_CLOSED=0, CB_OPEN=1, CB_HALF_OPEN=2 } CBState;
typedef struct {
  volatile u32 state;
  volatile u32 fail_count;
  volatile u64 opened_at_ns;
  u32 fail_threshold;
  u64 cooldown_ns;
} CircuitBreaker;

static void raf_cb_init(CircuitBreaker *cb, u32 threshold, u32 cooldown_ms){
  cb->state = CB_CLOSED;
  cb->fail_count = 0;
  cb->opened_at_ns = 0;
  cb->fail_threshold = threshold;
  cb->cooldown_ns = (u64)cooldown_ms * 1000000ULL;
}
static RafErr raf_cb_allow(CircuitBreaker *cb){
  u32 s = raf_load_acquire(&cb->state);
  if(s == CB_OPEN){
    /* checa se cooldown passou */
    if(_now_ns() - cb->opened_at_ns >= cb->cooldown_ns){
      raf_cas32(&cb->state, CB_OPEN, CB_HALF_OPEN);
      return RAF_OK;  /* permite 1 tentativa */
    }
    return RAF_ERR_BREAKER;
  }
  return RAF_OK;  /* CLOSED ou HALF_OPEN */
}
static void raf_cb_success(CircuitBreaker *cb){
  cb->fail_count = 0;
  raf_store_release(&cb->state, CB_CLOSED);
}
static void raf_cb_failure(CircuitBreaker *cb){
  u32 fc = raf_atomic_add(&cb->fail_count, 1u) + 1u;
  if(fc >= cb->fail_threshold){
    cb->opened_at_ns = _now_ns();
    raf_store_release(&cb->state, CB_OPEN);
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §13 — ENTROPY POOL (ESQ6) — base para CSPRNG, nonce, UUID
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { u64 pool[8]; u32 mix_count; } EntropyPool;

static void raf_entropy_mix(EntropyPool *p, u64 sample){
  u32 idx = p->mix_count & 7u;
  u64 v = p->pool[idx] ^ sample;
  v = (v<<13)|(v>>51);
  v *= 0x9E3779B97F4A7C15ULL;
  p->pool[idx] = v;
  p->pool[(idx+1)&7u] ^= (v >> 17);
  p->mix_count++;
}
static void raf_entropy_seed(EntropyPool *p){
  _mset(p, 0, sizeof(*p));
  raf_entropy_mix(p, _now_ns());
  u64 sa; raf_entropy_mix(p, (u64)(unsigned long)&sa);
  raf_entropy_mix(p, (u64)(unsigned long)&raf_entropy_seed);
  raf_entropy_mix(p, _now_ns());
  raf_entropy_mix(p, (u64)(unsigned long)G_ARENA);
}
static u64 raf_entropy_extract(EntropyPool *p){
  u64 r = 0;
  for(u32 i=0;i<8u;i++){
    r ^= p->pool[i];
    r = (r<<7)|(r>>57);
    r *= 0xBF58476D1CE4E5B9ULL;
  }
  raf_entropy_mix(p, r);
  return r;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §14 — U07: REPLAY PROTECTION (nonce monotônico)
 *   Cada operação consome counter monotônico global.
 *   Verificador rejeita nonces ≤ último visto.
 *   Sem isto: ataques de replay em qualquer protocolo.
 * ═══════════════════════════════════════════════════════════════════════════ */
static volatile u64 G_NONCE_COUNTER = 1;
static volatile u64 G_LAST_SEEN_NONCE = 0;

static u64 raf_nonce_get(void){
  return raf_atomic_add64(&G_NONCE_COUNTER, 1ULL);
}
static RafErr raf_nonce_verify(u64 nonce){
  /* CAS loop: aceita se maior que último, atualiza */
  for(;;){
    u64 last = G_LAST_SEEN_NONCE;
    if(nonce <= last) return RAF_ERR_REPLAY;
    if(__sync_bool_compare_and_swap(&G_LAST_SEEN_NONCE, last, nonce)){
      return RAF_OK;
    }
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §15 — U08: CORRELATION ID (UUID v4-like de entropia)
 *   16 bytes para tracing distribuído. Cada request tem o seu.
 *   Sem isto: impossível correlacionar logs em sistemas multi-componente.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { u8 bytes[16]; } CorrId;

static EntropyPool G_CORR_POOL;
static u32 G_CORR_INITED = 0;

static void raf_corr_init(void){
  if(!G_CORR_INITED){
    raf_entropy_seed(&G_CORR_POOL);
    G_CORR_INITED = 1;
  }
}
static CorrId raf_corr_new(void){
  raf_corr_init();
  CorrId id;
  u64 a = raf_entropy_extract(&G_CORR_POOL);
  u64 b = raf_entropy_extract(&G_CORR_POOL);
  for(u32 i=0;i<8u;i++){
    id.bytes[i]   = (u8)(a >> (i*8));
    id.bytes[8+i] = (u8)(b >> (i*8));
  }
  /* marca como v4 (RFC 4122): bits específicos */
  id.bytes[6] = (id.bytes[6] & 0x0Fu) | 0x40u;  /* version 4 */
  id.bytes[8] = (id.bytes[8] & 0x3Fu) | 0x80u;  /* variant 1 */
  return id;
}
static void raf_corr_to_hex(const CorrId *id, char out[33]){
  static const char H[]="0123456789abcdef";
  for(u32 i=0;i<16u;i++){
    out[i*2]   = H[id->bytes[i] >> 4];
    out[i*2+1] = H[id->bytes[i] & 0xFu];
  }
  out[32] = 0;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §16 — U09: METRICS HISTOGRAM (p50/p95/p99 via buckets exponenciais)
 *   Buckets em escala log2: bucket[i] = valores em [2^i, 2^(i+1)).
 *   Cobre 1 ns até 2^31 ns ≈ 2 segundos.
 *   Sem isto: relatórios só com média escondem outliers críticos.
 * ═══════════════════════════════════════════════════════════════════════════ */
#define HIST_BUCKETS 32u

typedef struct {
  volatile u32 buckets[HIST_BUCKETS];
  volatile u64 sum_ns;
  volatile u32 count;
} Histogram;

static u32 raf_log2_u32(u32 v){
  if(v == 0) return 0;
  u32 r = 0;
  while(v >>= 1) r++;
  return r;
}

static void raf_hist_record(Histogram *h, u64 value_ns){
  u32 idx = raf_log2_u32((u32)(value_ns > 0xFFFFFFFFull ? 0xFFFFFFFFu : value_ns));
  if(idx >= HIST_BUCKETS) idx = HIST_BUCKETS - 1u;
  raf_atomic_add(&h->buckets[idx], 1u);
  raf_atomic_add64(&h->sum_ns, value_ns);
  raf_atomic_add(&h->count, 1u);
}

/* Percentil: encontra bucket que contém o p-ésimo valor */
static u64 raf_hist_percentile(const Histogram *h, u32 percent){
  u32 total = h->count;
  if(total == 0) return 0;
  u64 target = ((u64)total * (u64)percent) / 100ULL;
  u64 cum = 0;
  for(u32 i=0;i<HIST_BUCKETS;i++){
    cum += h->buckets[i];
    if(cum >= target){
      /* retorna meio do bucket: ~1.5 * 2^i */
      return (1ULL << i) + (1ULL << (i > 0 ? i-1 : 0));
    }
  }
  return 1ULL << (HIST_BUCKETS - 1u);
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §17 — U10: POOL ALLOCATOR (objetos fixos, mais previsível que arena)
 *   N slots de tamanho fixo, free list em índices.
 *   Alloc/free O(1) sem fragmentação.
 *   Sem isto: arena cresce monotonicamente; sem reciclagem.
 * ═══════════════════════════════════════════════════════════════════════════ */
#define POOL_OBJ_SIZE 256u
#define POOL_N_OBJS   64u

typedef struct {
  u8 slots[POOL_N_OBJS][POOL_OBJ_SIZE];
  u32 free_list[POOL_N_OBJS];
  volatile u32 free_head;
  u32 init_done;
} ObjPool;

static ObjPool G_POOL;

static void raf_pool_init(ObjPool *p){
  /* free list inicial: 0 → 1 → 2 → ... → N-1 → 0xFFFFFFFF */
  for(u32 i=0; i<POOL_N_OBJS-1; i++) p->free_list[i] = i + 1;
  p->free_list[POOL_N_OBJS-1] = 0xFFFFFFFFu;
  p->free_head = 0;
  p->init_done = 1;
}

static void *raf_pool_get(ObjPool *p){
  if(!p->init_done) raf_pool_init(p);
  for(;;){
    u32 head = raf_load_acquire(&p->free_head);
    if(head == 0xFFFFFFFFu) return (void*)0;  /* pool cheio */
    u32 next = p->free_list[head];
    if(raf_cas32(&p->free_head, head, next)){
      /* poison para detectar uso de dado velho */
      _mset(p->slots[head], POISON_BYTE, POOL_OBJ_SIZE);
      return p->slots[head];
    }
  }
}

static void raf_pool_put(ObjPool *p, void *obj){
  /* calcula índice a partir do ponteiro */
  i64 diff = (u8*)obj - (u8*)p->slots;
  if(diff < 0 || diff >= (i64)(POOL_N_OBJS * POOL_OBJ_SIZE)) return;
  u32 idx = (u32)(diff / POOL_OBJ_SIZE);
  /* poison antes de devolver */
  _mset(p->slots[idx], POISON_BYTE, POOL_OBJ_SIZE);
  /* push atômico na free list */
  for(;;){
    u32 head = raf_load_acquire(&p->free_head);
    p->free_list[idx] = head;
    if(raf_cas32(&p->free_head, head, idx)) return;
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §18 — U11: BLOOM FILTER (dedup probabilístico)
 *   Conjunto sem armazenar elementos. ~1% falso-positivo, 0% falso-negativo.
 *   Sem isto: dedup exige hash set completo (memória O(n) em vez de O(1)).
 * ═══════════════════════════════════════════════════════════════════════════ */
#define BLOOM_BITS 8192u    /* 1KB de bits */
#define BLOOM_HASHES 3u

typedef struct { u8 bits[BLOOM_BITS/8u]; volatile u32 count; } Bloom;

/* 3 hashes baratos: variações de FNV-1a com seeds diferentes */
static u32 raf_bloom_hash(const u8 *data, u32 len, u32 seed){
  u32 h = 0x811C9DC5u ^ seed;
  for(u32 i=0;i<len;i++){
    h ^= (u32)data[i];
    h *= 0x01000193u;
  }
  return h % BLOOM_BITS;
}

static void raf_bloom_add(Bloom *b, const u8 *data, u32 len){
  for(u32 s=0; s<BLOOM_HASHES; s++){
    u32 bit = raf_bloom_hash(data, len, s * 0xDEADBEEFu);
    b->bits[bit/8u] |= (u8)(1u << (bit & 7u));
  }
  raf_atomic_add(&b->count, 1u);
}
static u32 raf_bloom_contains(const Bloom *b, const u8 *data, u32 len){
  for(u32 s=0; s<BLOOM_HASHES; s++){
    u32 bit = raf_bloom_hash(data, len, s * 0xDEADBEEFu);
    if(!(b->bits[bit/8u] & (u8)(1u << (bit & 7u)))) return 0u;
  }
  return 1u;  /* probavelmente presente (1% falso positivo) */
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §19 — U12 + ESQ9: STRUCTURED LOG append-only com CRC + campos k=v
 *   Cada entrada tem: ts, level, msg, fields[k][v], crc.
 *   Sem isto: parsing de logs por regex; tracing impossível.
 * ═══════════════════════════════════════════════════════════════════════════ */
#define LOG_RING 256u
#define LOG_MSG_MAX 64u
#define LOG_FIELDS_MAX 4u
#define LOG_FIELD_KEY_MAX 16u
#define LOG_FIELD_VAL_MAX 32u
#define LOG_MAGIC 0x4C4F4732u

typedef enum { LVL_TRACE=0, LVL_DEBUG, LVL_INFO, LVL_WARN, LVL_ERROR, LVL_FATAL } LogLevel;

typedef struct {
  char key[LOG_FIELD_KEY_MAX];
  char val[LOG_FIELD_VAL_MAX];
} LogField;

typedef struct {
  u32 magic;
  u64 ts_ns;
  u32 level;
  CorrId corr_id;
  char msg[LOG_MSG_MAX];
  u32 n_fields;
  LogField fields[LOG_FIELDS_MAX];
  u32 crc;
} LogEntry;

static LogEntry G_LOG[LOG_RING];
static volatile u32 G_LOG_HEAD = 0;
static volatile u32 G_LOG_COUNT = 0;

/* ═══════════════════════════════════════════════════════════════════════════
 * §20 — U06: AUDIT TRAIL HMAC (log tamper-proof)
 *   Cada entry de log audit assinada com HMAC-CRC (não cripto-forte, mas
 *   detecta tampering casual; para forensics real, usar HMAC-SHA256).
 *   Chain: hmac_i = HMAC(key, entry_i || hmac_{i-1})
 *   Sem isto: log pode ser editado retroativamente sem detecção.
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 G_AUDIT_KEY[8] = {
  0x52414641u, 0xDEADBEEFu, 0xCAFEBABEu, 0x12345678u,
  0xFEEDFACEu, 0xBADC0DEDu, 0xABCDEF01u, 0x23456789u
};
static volatile u32 G_AUDIT_CHAIN = 0xFFFFFFFFu;

static u32 raf_audit_hmac(const u8 *data, u32 len, u32 prev_chain){
  /* HMAC-CRC simplificado: CRC(key XOR data) XOR prev_chain */
  u32 h = prev_chain;
  for(u32 i=0;i<len;i++){
    u8 b = data[i] ^ ((u8)G_AUDIT_KEY[i & 7u]);
    h ^= (u32)b;
    for(u32 j=0;j<8u;j++) h = (h>>1) ^ (0x82F63B78u & -(h&1u));
  }
  return ~h;
}

static u32 raf_audit_chain_current(void){ return G_AUDIT_CHAIN; }

/* Verifica integridade do log via reconstrução de chain */
static u32 raf_audit_verify(void){
  u32 chain = 0xFFFFFFFFu;
  u32 n = G_LOG_COUNT < LOG_RING ? G_LOG_COUNT : LOG_RING;
  u32 ok = 0;
  for(u32 i=0; i<n; i++){
    if(G_LOG[i].magic != LOG_MAGIC) continue;
    u32 expected = raf_audit_hmac((const u8*)&G_LOG[i],
                                    sizeof(LogEntry) - 4u, chain);
    if(expected == G_LOG[i].crc) ok++;
    chain = expected;
  }
  return ok;
}

/* Log estruturado com correlation id e audit chain */
static void raf_log_with(LogLevel lvl, const CorrId *cid, const char *msg,
                         const char *k1, const char *v1,
                         const char *k2, const char *v2){
  u32 idx;
  for(;;){
    u32 old = raf_load_acquire(&G_LOG_HEAD);
    u32 new_h = (old + 1u) % LOG_RING;
    if(raf_cas32(&G_LOG_HEAD, old, new_h)){ idx = old; break; }
  }
  LogEntry *e = &G_LOG[idx];
  _mset(e, 0, sizeof(*e));
  e->magic = LOG_MAGIC;
  e->ts_ns = _now_ns();
  e->level = (u32)lvl;
  if(cid) e->corr_id = *cid;
  raf_strlcpy(e->msg, msg, LOG_MSG_MAX);
  if(k1 && v1){
    raf_strlcpy(e->fields[0].key, k1, LOG_FIELD_KEY_MAX);
    raf_strlcpy(e->fields[0].val, v1, LOG_FIELD_VAL_MAX);
    e->n_fields = 1;
  }
  if(k2 && v2){
    raf_strlcpy(e->fields[1].key, k2, LOG_FIELD_KEY_MAX);
    raf_strlcpy(e->fields[1].val, v2, LOG_FIELD_VAL_MAX);
    e->n_fields = 2;
  }
  /* assina com audit chain */
  u32 prev_chain = raf_load_acquire(&G_AUDIT_CHAIN);
  e->crc = raf_audit_hmac((const u8*)e, sizeof(LogEntry) - 4u, prev_chain);
  raf_store_release(&G_AUDIT_CHAIN, e->crc);
  raf_atomic_add(&G_LOG_COUNT, 1u);
}

/* Wrapper simples */
static void raf_log(LogLevel lvl, const char *msg){
  raf_log_with(lvl, (void*)0, msg, (void*)0, (void*)0, (void*)0, (void*)0);
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §21 — U14: ERROR INJECTION (chaos engineering)
 *   Probabilidade configurável de falha em pontos críticos.
 *   Útil para testar caminhos de erro em produção.
 *   Sem isto: caminhos de erro nunca exercitados.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  volatile u32 enabled;
  u32 probability_ppm;   /* 0..1000000 (ppm = parts per million) */
  EntropyPool pool;
  volatile u32 injection_count;
} ChaosCfg;

static ChaosCfg G_CHAOS = {0, 0, {{0}, 0}, 0};

static void raf_chaos_enable(u32 prob_ppm){
  raf_entropy_seed(&G_CHAOS.pool);
  G_CHAOS.probability_ppm = prob_ppm;
  raf_store_release(&G_CHAOS.enabled, 1u);
}
static void raf_chaos_disable(void){
  raf_store_release(&G_CHAOS.enabled, 0u);
}
/* Chamado em pontos críticos: retorna RAF_ERR_CHAOS se sorteado */
static RafErr raf_chaos_maybe_fail(void){
  if(!raf_load_acquire(&G_CHAOS.enabled)) return RAF_OK;
  u64 r = raf_entropy_extract(&G_CHAOS.pool);
  u32 sample_ppm = (u32)(r % 1000000ULL);
  if(sample_ppm < G_CHAOS.probability_ppm){
    raf_atomic_add(&G_CHAOS.injection_count, 1u);
    return RAF_ERR_CHAOS;
  }
  return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════

#define MAX_GATES 8

/* ═══════════════════════════════════════════════════════════════════════════
 * §23 — RATE LIMITER (ESQ13)
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u32 capacity, refill_per_sec;
  volatile u32 tokens;
  volatile u64 last_refill_ns;
} RateLimiter;

static void raf_rl_init(RateLimiter *rl, u32 cap, u32 refill){
  rl->capacity = cap; rl->refill_per_sec = refill;
  rl->tokens = cap; rl->last_refill_ns = _now_ns();
}
static RafErr raf_rl_try(RateLimiter *rl){
  u64 now = _now_ns();
  u64 elapsed = now - rl->last_refill_ns;
  u32 add = (u32)((elapsed * (u64)rl->refill_per_sec) / 1000000000ULL);
  if(add > 0){
    u32 nt = rl->tokens + add;
    if(nt > rl->capacity) nt = rl->capacity;
    rl->tokens = nt; rl->last_refill_ns = now;
  }
  for(;;){
    u32 cur = raf_load_acquire(&rl->tokens);
    if(cur == 0) return RAF_ERR_RATELIM;
    if(raf_cas32(&rl->tokens, cur, cur - 1)) return RAF_OK;
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §24 — CONFIG (ESQ11) + U15: CHECKSUM ALL THE THINGS
 *   Toda config validada + CRC + version stamp.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u32 schema_version;
  u32 max_log;
  u32 rate_limit_rps;
  u32 watchdog_ms;
  u32 chaos_enabled;
  u32 chaos_ppm;
  u32 cb_threshold;
  u32 cb_cooldown_ms;
  u32 crc;
} RafConfig;

static RafConfig G_CFG = {
  .schema_version = 1,
  .max_log = 256,
  .rate_limit_rps = 1000,
  .watchdog_ms = 5000,
  .chaos_enabled = 0,
  .chaos_ppm = 0,
  .cb_threshold = 5,
  .cb_cooldown_ms = 10000,
  .crc = 0
};

static RafErr raf_config_validate(const RafConfig *c){
  if(c->schema_version != 1) return RAF_ERR_ABI;
  if(c->max_log == 0 || c->max_log > 8192) return RAF_ERR_RANGE;
  if(c->rate_limit_rps == 0 || c->rate_limit_rps > 1000000) return RAF_ERR_RANGE;
  if(c->watchdog_ms < 10 || c->watchdog_ms > 600000) return RAF_ERR_RANGE;
  if(c->chaos_ppm > 1000000) return RAF_ERR_RANGE;
  if(c->cb_threshold == 0 || c->cb_threshold > 1000) return RAF_ERR_RANGE;
  if(c->cb_cooldown_ms > 600000) return RAF_ERR_RANGE;
  if(c->crc != 0){
    u32 expected = crc32c((const u8*)c, sizeof(RafConfig) - 4u);
    if(c->crc != expected) return RAF_ERR_CRC;
  }
  return RAF_OK;
}

static RafErr raf_config_seal(RafConfig *c){
  RafErr e = raf_config_validate(c);
  if(e != RAF_OK) return e;
  c->crc = crc32c((const u8*)c, sizeof(RafConfig) - 4u);
  return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §25 — SIGNALS (ESQ2) + PANIC (ESQ10) + SHUTDOWN (ESQ12)
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef void (*ShutdownFn)(void);
#define MAX_HOOKS 16
static ShutdownFn G_HOOKS[MAX_HOOKS];
static u32 G_N_HOOKS = 0;
static volatile u32 G_SHUTDOWN_REQ = 0;

static RafErr raf_atexit(ShutdownFn fn){
  if(G_N_HOOKS >= MAX_HOOKS) return RAF_ERR_OOM;
  G_HOOKS[G_N_HOOKS++] = fn;
  return RAF_OK;
}

static void raf_shutdown(void){
  raf_log(LVL_INFO, "shutdown");
  for(i32 i=(i32)G_N_HOOKS - 1; i >= 0; i--){
    if(G_HOOKS[i]) G_HOOKS[i]();
  }
}

static __attribute__((noreturn)) void raf_panic(const char *reason){
  _w("\n[PANIC] ");
  _w(reason);
  _w("\n  log_count: "); _pd(G_LOG_COUNT);
  _w("  audit_chain: "); _ph(raf_audit_chain_current());
  raf_log(LVL_FATAL, reason);
  _ex_fn(RAF_ERR_PANIC);
  for(;;){}
}

#ifdef RAF_LIBC
static void raf_signal_handler(int sig){
  const char *name = "unknown";
  switch(sig){
    case SIGSEGV: name = "SIGSEGV"; break;
    case SIGINT:  name = "SIGINT"; break;
    case SIGBUS:  name = "SIGBUS"; break;
    case SIGTERM: name = "SIGTERM"; break;
  }
  raf_log(LVL_FATAL, name);
  raf_store_release(&G_SHUTDOWN_REQ, 1);
  if(sig == SIGSEGV || sig == SIGBUS) _Exit(sig);
}
static void raf_signals_install(void){
  signal(SIGSEGV, raf_signal_handler);
  signal(SIGINT,  raf_signal_handler);
  signal(SIGBUS,  raf_signal_handler);
  signal(SIGTERM, raf_signal_handler);
  signal(SIGPIPE, SIG_IGN);
}

static void raf_signals_install(void){}
#endif

 * §26 — TESTES B0-B69 (70 testes — todos os 30 mecanismos)
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 G_PASS=0, G_FAIL=0;
static void _T(u32 ok, const char *name){
  if(ok){ G_PASS++; _w("  [PASS] "); }

  _T(raf_nonce_verify(n2 + 100) == RAF_OK, "B33:nonce_verify_higher");
  _T(raf_nonce_verify(n2) == RAF_ERR_REPLAY, "B34:nonce_verify_replay");

  }

  /* bloom add/check */
  Bloom bl; _mset(&bl, 0, sizeof(bl));
  t0 = _now_ns();
  for(u32 i=0;i<10000;i++){
    raf_bloom_add(&bl, (const u8*)&i, 4);
