#define ARENA_SZ (32u*1024u*1024u)
static u8 G_ARENA[ARENA_SZ] __attribute__((aligned(64)));
static u32 G_ABUMP = 0u;

static void *_arena_alloc(u32 n){
  u32 s=(G_ABUMP+63u)&~63u, e=s+n;
  if(e>ARENA_SZ) return (void*)0;
  G_ABUMP=e;

}

/* ═══════════════════════════════════════════════════════════════════════════
 * §3 — CRC32C BRANCHLESS (poly Castagnoli 0x82F63B78)
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 crc32c(const u8 *p, u32 n){
  u32 c=0xFFFFFFFFu;
  for(u32 i=0;i<n;i++){
    c^=(u32)p[i];
    /* 8 rodadas branchless: mask = -(c&1) */
    for(u32 j=0;j<8u;j++) c=(c>>1)^(0x82F63B78u&-(c&1u));
  }
  return ~c;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §4 — GAP1: JANELA DE CONTEXTO N=8 com decay λ=√3/2
 *
 * Substitui bigram puro do iaRAF/pipeline.
 * Mantém os últimos 8 token_ids num ring buffer.
 * Score do próximo token = soma ponderada por λ^i dos contribuintes.
 * w[0]=λ^0=1.000, w[1]=0.866, w[2]=0.750, ..., w[7]=0.300
 * Tudo Q16 fixed-point — sem float no caminho quente.
 * ═══════════════════════════════════════════════════════════════════════════ */
#define WIN_N        8u
#define Q16_LAM      56756u    /* √3/2 × 65536 ≈ 0.866 */
#define Q16_ONE      65536u

/* Pesos pré-calculados: w[i] = λ^i × Q16 (branchless, sem loop em uso) */
static const u32 WIN_W[WIN_N] = {
  65536u,    /* λ^0 = 1.000000 */
  56756u,    /* λ^1 = 0.866025 */
  49152u,    /* λ^2 = 0.750000 */
  42569u,    /* λ^3 = 0.649519 */
  36864u,    /* λ^4 = 0.562500 */
  31927u,    /* λ^5 = 0.487139 */
  27648u,    /* λ^6 = 0.421875 */
  23945u     /* λ^7 = 0.365354 */
};

typedef struct {
  u32 tokens[WIN_N];    /* ring buffer */
  u32 head;             /* índice da próxima escrita */
  u32 count;            /* tokens válidos atualmente (≤ WIN_N) */
} ContextWindow;

static void win_init(ContextWindow *w){
  _mset(w, 0, sizeof(*w));
}

static void win_push(ContextWindow *w, u32 token){
  w->tokens[w->head] = token;
  w->head = (w->head + 1u) % WIN_N;
  /* branchless: count = min(count+1, WIN_N) */
  u32 c = w->count + 1u;
  u32 mask = -(u32)(c > WIN_N);
  w->count = (c & ~mask) | (WIN_N & mask);
}

/* Calcula score Q16 do token candidato dado contexto + bigram_score[tok][cand]
 * bigram_score: tabela esparsa simples — em produção viria do modelo treinado.
 * Aqui usamos uma função sintética determinística para testes. */
static u32 win_score(const ContextWindow *w, u32 candidate,
                      const u32 *bigram_scores, u32 vocab_size){
  u64 acc = 0;
  /* Itera do mais recente para o mais antigo */
  for(u32 i=0; i<w->count; i++){
    u32 idx = (w->head + WIN_N - 1u - i) % WIN_N;
    u32 src = w->tokens[idx];
    /* score[src][candidate] em Q16 — tabela flat */
    u32 bg_score = (src<vocab_size && candidate<vocab_size)
                    ? bigram_scores[src*vocab_size + candidate]
                    : 0u;
    /* multiplica peso × score (Q16 × Q16 → Q32, shift 16 → Q16) */
    acc += ((u64)WIN_W[i] * (u64)bg_score) >> 16;
  }
  /* clamp Q16 */
  if(acc > 0xFFFFFFFFull) acc = 0xFFFFFFFFull;
  return (u32)acc;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §5 — GAP2: DECODER JPEG BASELINE (DC-only)
 *
 * Implementação mínima do parser JPEG baseline:
 *   - lê markers SOI (FFD8), DQT (FFDB), SOF0 (FFC0), DHT (FFC4), SOS (FFDA)
 *   - decodifica apenas coeficiente DC de cada bloco 8×8
 *   - reconstrói imagem em luminância (DC ≈ média do bloco × 8)
 *
 * Suficiente para extrair features de pHash do GEOLM sem ffmpeg.
 * IDCT completa exigiria ~500 linhas adicionais — fica para iteração futura.
 * Para fins de classificação visual com pHash 8×8, apenas DC é necessário.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u32 width, height;
  u32 mcu_w, mcu_h;       /* MCUs por linha/coluna */
  u8  dc_lum[64*64];      /* até 64×64 MCUs = 512×512 imagem */
  u32 n_blocks;
  u32 parse_ok;
} JpegImg;

/* Tabelas Huffman: simplificadas — apenas para detecção de DC */
typedef struct {
  u8  bits[17];      /* counts por comprimento de código */
  u8  values[256];
  u32 n_values;
} HuffTable;

typedef struct {
  const u8 *p;       /* ponteiro corrente */
  const u8 *end;
  u32 bit_buf;
  u32 bit_count;
} BitStream;

static i32 bs_get_bit(BitStream *bs){
  /* branchless refill: se bit_count<1, lê byte */
  if(bs->bit_count == 0u){
    if(bs->p >= bs->end) return -1;
    u8 b = *bs->p++;
    /* JPEG byte stuffing: FF seguido de 00 = literal FF */
    if(b==0xFF && bs->p<bs->end && *bs->p==0x00) bs->p++;
    bs->bit_buf = b;
    bs->bit_count = 8u;
  }
  bs->bit_count--;
  return (i32)((bs->bit_buf >> bs->bit_count) & 1u);
}
static i32 bs_get_bits(BitStream *bs, u32 n){
  u32 v=0;
  for(u32 i=0;i<n;i++){
    i32 b = bs_get_bit(bs);
    if(b<0) return -1;
    v = (v<<1) | (u32)b;
  }
  return (i32)v;
}

/* Decodifica símbolo Huffman: percorre table.bits[] acumulando bits */
static i32 huff_decode(BitStream *bs, const HuffTable *h){
  u32 code = 0;
  u32 idx  = 0;
  for(u32 len=1; len<=16u; len++){
    i32 b = bs_get_bit(bs);
    if(b<0) return -1;
    code = (code<<1) | (u32)b;
    u32 count = h->bits[len];
    /* table walking simplificado: assume códigos canônicos */
    u32 first = 0;
    /* primeiro código de comprimento `len` é (first_code) */
    u32 fc = 0;
    for(u32 l=1; l<len; l++){
      fc = (fc + h->bits[l]) << 1;
      first += h->bits[l];
    }
    if(code >= fc && code < fc + count){
      u32 val_idx = first + (code - fc);
      if(val_idx < h->n_values) return (i32)h->values[val_idx];
      return -1;
    }
  }
  return -1;
}

/* Recupera valor com sinal a partir de magnitude + bits adicionais */
static i32 extend_sign(i32 val, u32 nbits){
  if(nbits==0) return 0;
  i32 vt = 1 << (nbits-1u);
  /* branchless: if(val < vt) val += -(1<<nbits) + 1 */
  i32 add = (-(i32)(val < vt)) & (-(1 << nbits) + 1);
  return val + add;
}

static i32 jpeg_parse(const u8 *buf, u32 len, JpegImg *out){
  _mset(out, 0, sizeof(*out));
  if(len < 4) return -1;
  if(buf[0]!=0xFF || buf[1]!=0xD8) return -1;  /* SOI */

  const u8 *p   = buf + 2;
  const u8 *end = buf + len;
  HuffTable dc_table[4]; _mset(dc_table, 0, sizeof(dc_table));
  HuffTable ac_table[4]; _mset(ac_table, 0, sizeof(ac_table));
  i32 prev_dc = 0;

  while(p+4 <= end){
    if(*p != 0xFF){ p++; continue; }
    u8 marker = p[1];
    p += 2;
    if(marker==0xD9) break;  /* EOI */
    if(marker==0xD8 || (marker>=0xD0 && marker<=0xD7)){ continue; }
    if(p+2 > end) break;
    u32 seg_len = ((u32)p[0]<<8) | (u32)p[1];
    if(p + seg_len > end) break;

    if(marker == 0xC0){  /* SOF0: Start of Frame baseline */
      if(seg_len < 8) return -1;
      /* precision=p[2], height=p[3..4], width=p[5..6] */
      out->height = ((u32)p[3]<<8) | (u32)p[4];
      out->width  = ((u32)p[5]<<8) | (u32)p[6];
      out->mcu_w  = (out->width  + 7u) / 8u;
      out->mcu_h  = (out->height + 7u) / 8u;
      /* limita a 64×64 MCUs (512×512 imagem) */
      if(out->mcu_w > 64u) out->mcu_w = 64u;
      if(out->mcu_h > 64u) out->mcu_h = 64u;
    }
    else if(marker == 0xC4){  /* DHT: Define Huffman Table */
      const u8 *q = p + 2;
      const u8 *qe = p + seg_len;
      while(q < qe){
        u8 cls_id = *q++;
        u8 cls = cls_id >> 4;     /* 0=DC, 1=AC */
        u8 id  = cls_id & 0x0F;
        if(id >= 4 || q+16 > qe) break;
        HuffTable *t = (cls==0) ? &dc_table[id] : &ac_table[id];
        u32 total = 0;
        for(u32 i=1; i<=16u; i++){ t->bits[i] = q[i-1]; total += t->bits[i]; }
        q += 16;
        if(q + total > qe || total > 256) break;
        for(u32 i=0; i<total; i++) t->values[i] = q[i];
        t->n_values = total;
        q += total;
      }
    }
    else if(marker == 0xDA){  /* SOS: Start of Scan — começa dados comprimidos */
      const u8 *scan_start = p + seg_len;
      BitStream bs = { scan_start, end, 0, 0 };
      prev_dc = 0;
      u32 n_mcu = out->mcu_w * out->mcu_h;
      if(n_mcu > 64u*64u) n_mcu = 64u*64u;
      for(u32 m=0; m<n_mcu; m++){
        /* DC: huff_decode → magnitude, lê magnitude bits, extend sign */
        i32 mag = huff_decode(&bs, &dc_table[0]);
        if(mag < 0 || mag > 11){ out->dc_lum[m] = 128; continue; }
        i32 bits = mag>0 ? bs_get_bits(&bs, (u32)mag) : 0;
        i32 dc_diff = extend_sign(bits, (u32)mag);
        prev_dc += dc_diff;
        /* DC = soma do bloco × 8 / 8 = média; mapeia para [0,255] */
        i32 lum = 128 + (prev_dc >> 3);
        if(lum < 0) lum = 0;
        if(lum > 255) lum = 255;
        out->dc_lum[m] = (u8)lum;
        /* Skip AC: lê 63 coeficientes com (run,size) */
        for(u32 k=1; k<64u; k++){
          i32 rs = huff_decode(&bs, &ac_table[0]);
          if(rs < 0) break;
          if(rs == 0) break;  /* EOB */
          u32 run = (u32)(rs >> 4);
          u32 sz  = (u32)(rs & 0x0F);
          k += run;
          if(sz > 0) bs_get_bits(&bs, sz);
          if(k >= 63u) break;
        }
      }
      out->n_blocks = n_mcu;
      out->parse_ok = 1u;
      return 0;
    }
    p += seg_len;
  }
  return out->parse_ok ? 0 : -1;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §6 — GAP3: POLY1305 com carry/reduction completo
 *
 * RFC 8439 §2.5.1: Poly1305 over GF(2^130 - 5)
 * Implementação 26-bit limbs × 5 (130 bits totais)
 * Carry propagation completa em cada multiplicação
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u32 r[5];        /* chave r, com clamping */
  u32 h[5];        /* acumulador */
  u32 pad[4];      /* pad final (s) */
  u8  buffer[16];
  u32 buffer_len;
} Poly1305Ctx;

static void poly_init(Poly1305Ctx *p, const u8 key[32]){
  _mset(p, 0, sizeof(*p));
  /* r com clamping: r &= 0x0ffffffc0ffffffc0ffffffc0fffffff */
  u32 t0 = ((u32)key[0])      | ((u32)key[1]<<8)  | ((u32)key[2]<<16) | ((u32)key[3]<<24);
  u32 t1 = ((u32)key[4])      | ((u32)key[5]<<8)  | ((u32)key[6]<<16) | ((u32)key[7]<<24);
  u32 t2 = ((u32)key[8])      | ((u32)key[9]<<8)  | ((u32)key[10]<<16)| ((u32)key[11]<<24);
  u32 t3 = ((u32)key[12])     | ((u32)key[13]<<8) | ((u32)key[14]<<16)| ((u32)key[15]<<24);
  p->r[0] = (t0      ) & 0x03ffffffu;
  p->r[1] = ((t0>>26)|(t1<< 6)) & 0x03ffff03u;
  p->r[2] = ((t1>>20)|(t2<<12)) & 0x03ffc0ffu;
  p->r[3] = ((t2>>14)|(t3<<18)) & 0x03f03fffu;
  p->r[4] = ((t3>> 8))          & 0x000fffffu;
  /* pad = key[16..31] */
  for(u32 i=0;i<4;i++){
    p->pad[i] = ((u32)key[16+i*4]) | ((u32)key[17+i*4]<<8)
              | ((u32)key[18+i*4]<<16) | ((u32)key[19+i*4]<<24);
  }
}

static void poly_block(Poly1305Ctx *p, const u8 *m, u32 hibit){
  u32 t0 = ((u32)m[0])  | ((u32)m[1]<<8)  | ((u32)m[2]<<16)  | ((u32)m[3]<<24);
  u32 t1 = ((u32)m[4])  | ((u32)m[5]<<8)  | ((u32)m[6]<<16)  | ((u32)m[7]<<24);
  u32 t2 = ((u32)m[8])  | ((u32)m[9]<<8)  | ((u32)m[10]<<16) | ((u32)m[11]<<24);
  u32 t3 = ((u32)m[12]) | ((u32)m[13]<<8) | ((u32)m[14]<<16) | ((u32)m[15]<<24);

  /* h += m */
  p->h[0] += (t0)                  & 0x03ffffffu;
  p->h[1] += ((t0>>26)|(t1<< 6))   & 0x03ffffffu;
  p->h[2] += ((t1>>20)|(t2<<12))   & 0x03ffffffu;
  p->h[3] += ((t2>>14)|(t3<<18))   & 0x03ffffffu;
  p->h[4] += ((t3>> 8))            | (hibit << 24);

  /* h = h * r — produto polinomial mod (2^130-5) */
  u32 r0=p->r[0], r1=p->r[1], r2=p->r[2], r3=p->r[3], r4=p->r[4];
  u32 s1 = r1*5u, s2 = r2*5u, s3 = r3*5u, s4 = r4*5u;
  u32 h0=p->h[0], h1=p->h[1], h2=p->h[2], h3=p->h[3], h4=p->h[4];

  u64 d0 = (u64)h0*r0 + (u64)h1*s4 + (u64)h2*s3 + (u64)h3*s2 + (u64)h4*s1;
  u64 d1 = (u64)h0*r1 + (u64)h1*r0 + (u64)h2*s4 + (u64)h3*s3 + (u64)h4*s2;
  u64 d2 = (u64)h0*r2 + (u64)h1*r1 + (u64)h2*r0 + (u64)h3*s4 + (u64)h4*s3;
  u64 d3 = (u64)h0*r3 + (u64)h1*r2 + (u64)h2*r1 + (u64)h3*r0 + (u64)h4*s4;
  u64 d4 = (u64)h0*r4 + (u64)h1*r3 + (u64)h2*r2 + (u64)h3*r1 + (u64)h4*r0;

  /* Carry propagation completa (cadeia de 5 limbs) */
  u32 c;
  c = (u32)(d0 >> 26); h0 = (u32)d0 & 0x03ffffffu; d1 += c;
  c = (u32)(d1 >> 26); h1 = (u32)d1 & 0x03ffffffu; d2 += c;
  c = (u32)(d2 >> 26); h2 = (u32)d2 & 0x03ffffffu; d3 += c;
  c = (u32)(d3 >> 26); h3 = (u32)d3 & 0x03ffffffu; d4 += c;
  c = (u32)(d4 >> 26); h4 = (u32)d4 & 0x03ffffffu;
  /* reduction final mod (2^130 - 5): h0 += c*5, propaga */
  h0 += c * 5u;
  c = h0 >> 26; h0 &= 0x03ffffffu; h1 += c;

  p->h[0]=h0; p->h[1]=h1; p->h[2]=h2; p->h[3]=h3; p->h[4]=h4;
}

static void poly_update(Poly1305Ctx *p, const u8 *m, u32 len){
  if(p->buffer_len > 0u){
    u32 want = 16u - p->buffer_len;
    if(want > len) want = len;
    _mcpy(p->buffer + p->buffer_len, m, want);
    p->buffer_len += want;
    m += want; len -= want;
    if(p->buffer_len == 16u){
      poly_block(p, p->buffer, 1u);
      p->buffer_len = 0u;
    }
  }
  while(len >= 16u){
    poly_block(p, m, 1u);
    m += 16u; len -= 16u;
  }
  if(len > 0u){
    _mcpy(p->buffer, m, len);
    p->buffer_len = len;
  }
}

static void poly_finish(Poly1305Ctx *p, u8 mac[16]){
  /* processa bloco final com padding */
  if(p->buffer_len > 0u){
    p->buffer[p->buffer_len] = 1u;
    for(u32 i=p->buffer_len+1u; i<16u; i++) p->buffer[i]=0u;
    poly_block(p, p->buffer, 0u);
  }
  /* carry propagation final */
  u32 h0=p->h[0], h1=p->h[1], h2=p->h[2], h3=p->h[3], h4=p->h[4];
  u32 c;
  c = h1 >> 26; h1 &= 0x03ffffffu; h2 += c;
  c = h2 >> 26; h2 &= 0x03ffffffu; h3 += c;
  c = h3 >> 26; h3 &= 0x03ffffffu; h4 += c;
  c = h4 >> 26; h4 &= 0x03ffffffu; h0 += c * 5u;
  c = h0 >> 26; h0 &= 0x03ffffffu; h1 += c;

  /* g = h - (2^130 - 5), seleciona min(h, g) em constant-time */
  u32 g0 = h0 + 5u; c = g0 >> 26; g0 &= 0x03ffffffu;
  u32 g1 = h1 + c;  c = g1 >> 26; g1 &= 0x03ffffffu;
  u32 g2 = h2 + c;  c = g2 >> 26; g2 &= 0x03ffffffu;
  u32 g3 = h3 + c;  c = g3 >> 26; g3 &= 0x03ffffffu;
  u32 g4 = h4 + c - (1u << 26);

  /* máscara: se g4 < 2^31 (não estourou), usa g; senão usa h */
  u32 mask = (g4 >> 31) - 1u;  /* 0 se g≥p, ALL_ONES se g<p */
  g0 &= mask; g1 &= mask; g2 &= mask; g3 &= mask; g4 &= mask;
  mask = ~mask;
  h0 = (h0 & mask) | g0;
  h1 = (h1 & mask) | g1;
  h2 = (h2 & mask) | g2;
  h3 = (h3 & mask) | g3;
  h4 = (h4 & mask) | g4;

  /* recompõe 128 bits canônicos */
  u32 f0 = (h0       ) | (h1 << 26);
  u32 f1 = (h1 >>  6 ) | (h2 << 20);
  u32 f2 = (h2 >> 12 ) | (h3 << 14);
  u32 f3 = (h3 >> 18 ) | (h4 <<  8);

  /* + pad (s) com carry */
  u64 ff0 = (u64)f0 + (u64)p->pad[0];
  u64 ff1 = (u64)f1 + (u64)p->pad[1] + (ff0 >> 32);
  u64 ff2 = (u64)f2 + (u64)p->pad[2] + (ff1 >> 32);
  u64 ff3 = (u64)f3 + (u64)p->pad[3] + (ff2 >> 32);

  f0 = (u32)ff0; f1 = (u32)ff1; f2 = (u32)ff2; f3 = (u32)ff3;
  mac[ 0]=(u8)f0;     mac[ 1]=(u8)(f0>> 8); mac[ 2]=(u8)(f0>>16); mac[ 3]=(u8)(f0>>24);
  mac[ 4]=(u8)f1;     mac[ 5]=(u8)(f1>> 8); mac[ 6]=(u8)(f1>>16); mac[ 7]=(u8)(f1>>24);
  mac[ 8]=(u8)f2;     mac[ 9]=(u8)(f2>> 8); mac[10]=(u8)(f2>>16); mac[11]=(u8)(f2>>24);
  mac[12]=(u8)f3;     mac[13]=(u8)(f3>> 8); mac[14]=(u8)(f3>>16); mac[15]=(u8)(f3>>24);
}

static void poly1305_mac(const u8 *msg, u32 len, const u8 key[32], u8 out[16]){
  Poly1305Ctx ctx;
  poly_init(&ctx, key);
  poly_update(&ctx, msg, len);
  poly_finish(&ctx, out);
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §7 — GAP4: X25519 fe_to_bytes serialização canônica
 *
 * Campo finito 2^255 - 19, representação em 10 limbs 25.5-bit (radix 2^25.5)
 * Aqui implementamos apenas as duas funções faltantes:
 *   - fe_to_bytes: serializa estado canônico (representante mínimo)
 *   - fe_from_bytes: deserializa entrada
 *
 * Multiplicação ladder Montgomery já estava implementada no MATIBHE_MIL.
 * Esta é a peça que faltava para fechar.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { i32 v[10]; } fe25519;

static void fe_from_bytes(fe25519 *h, const u8 *s){
  i64 h0 = (i64)((u32)s[0]) | ((i64)((u32)s[1])<<8) | ((i64)((u32)s[2])<<16) | (((i64)s[3]&0x03LL)<<24);
  i64 h1 = ((i64)((u32)s[3])>>2 & 0x3fLL) | ((i64)((u32)s[4])<<6) | ((i64)((u32)s[5])<<14) | (((i64)s[6]&0x07LL)<<22);
  i64 h2 = ((i64)((u32)s[6])>>3 & 0x1fLL) | ((i64)((u32)s[7])<<5) | ((i64)((u32)s[8])<<13) | (((i64)s[9]&0x1fLL)<<21);
  i64 h3 = ((i64)((u32)s[9])>>5 & 0x07LL) | ((i64)((u32)s[10])<<3) | ((i64)((u32)s[11])<<11) | (((i64)s[12]&0x3fLL)<<19);
  i64 h4 = ((i64)((u32)s[12])>>6 & 0x03LL) | ((i64)((u32)s[13])<<2) | ((i64)((u32)s[14])<<10) | ((i64)((u32)s[15])<<18);
  i64 h5 = (i64)((u32)s[16]) | ((i64)((u32)s[17])<<8) | ((i64)((u32)s[18])<<16) | (((i64)s[19]&0x01LL)<<24);
  i64 h6 = ((i64)((u32)s[19])>>1 & 0x7fLL) | ((i64)((u32)s[20])<<7) | ((i64)((u32)s[21])<<15) | (((i64)s[22]&0x07LL)<<23);
  i64 h7 = ((i64)((u32)s[22])>>3 & 0x1fLL) | ((i64)((u32)s[23])<<5) | ((i64)((u32)s[24])<<13) | (((i64)s[25]&0x0fLL)<<21);
  i64 h8 = ((i64)((u32)s[25])>>4 & 0x0fLL) | ((i64)((u32)s[26])<<4) | ((i64)((u32)s[27])<<12) | (((i64)s[28]&0x7fLL)<<20);
  i64 h9 = ((i64)((u32)s[28])>>7 & 0x01LL) | ((i64)((u32)s[29])<<1) | ((i64)((u32)s[30])<<9) | ((i64)((u32)s[31]&0x7fLL)<<17);
  h->v[0]=(i32)h0; h->v[1]=(i32)h1; h->v[2]=(i32)h2; h->v[3]=(i32)h3; h->v[4]=(i32)h4;
  h->v[5]=(i32)h5; h->v[6]=(i32)h6; h->v[7]=(i32)h7; h->v[8]=(i32)h8; h->v[9]=(i32)h9;
}

static void fe_to_bytes(u8 *s, const fe25519 *h){
  i32 h0=h->v[0], h1=h->v[1], h2=h->v[2], h3=h->v[3], h4=h->v[4];
  i32 h5=h->v[5], h6=h->v[6], h7=h->v[7], h8=h->v[8], h9=h->v[9];
  i32 q, carry;
  /* redução para forma canônica: 0 ≤ h < 2^255 - 19 */
  q = (19*h9 + (1<<24)) >> 25;
  q = (h0 + q) >> 26;
  q = (h1 + q) >> 25;
  q = (h2 + q) >> 26;
  q = (h3 + q) >> 25;
  q = (h4 + q) >> 26;
  q = (h5 + q) >> 25;
  q = (h6 + q) >> 26;
  q = (h7 + q) >> 25;
  q = (h8 + q) >> 26;
  q = (h9 + q) >> 25;
  h0 += 19 * q;
  carry = h0 >> 26; h1 += carry; h0 -= carry << 26;
  carry = h1 >> 25; h2 += carry; h1 -= carry << 25;
  carry = h2 >> 26; h3 += carry; h2 -= carry << 26;
  carry = h3 >> 25; h4 += carry; h3 -= carry << 25;
  carry = h4 >> 26; h5 += carry; h4 -= carry << 26;
  carry = h5 >> 25; h6 += carry; h5 -= carry << 25;
  carry = h6 >> 26; h7 += carry; h6 -= carry << 26;
  carry = h7 >> 25; h8 += carry; h7 -= carry << 25;
  carry = h8 >> 26; h9 += carry; h8 -= carry << 26;
  carry = h9 >> 25;             h9 -= carry << 25;
  /* serialização little-endian em 32 bytes */
  s[ 0] = (u8)(h0);
  s[ 1] = (u8)(h0 >> 8);
  s[ 2] = (u8)(h0 >> 16);
  s[ 3] = (u8)((h0 >> 24) | (h1 << 2));
  s[ 4] = (u8)(h1 >> 6);
  s[ 5] = (u8)(h1 >> 14);
  s[ 6] = (u8)((h1 >> 22) | (h2 << 3));
  s[ 7] = (u8)(h2 >> 5);
  s[ 8] = (u8)(h2 >> 13);
  s[ 9] = (u8)((h2 >> 21) | (h3 << 5));
  s[10] = (u8)(h3 >> 3);
  s[11] = (u8)(h3 >> 11);
  s[12] = (u8)((h3 >> 19) | (h4 << 6));
  s[13] = (u8)(h4 >> 2);
  s[14] = (u8)(h4 >> 10);
  s[15] = (u8)(h4 >> 18);
  s[16] = (u8)(h5);
  s[17] = (u8)(h5 >> 8);
  s[18] = (u8)(h5 >> 16);
  s[19] = (u8)((h5 >> 24) | (h6 << 1));
  s[20] = (u8)(h6 >> 7);
  s[21] = (u8)(h6 >> 15);
  s[22] = (u8)((h6 >> 23) | (h7 << 3));
  s[23] = (u8)(h7 >> 5);
  s[24] = (u8)(h7 >> 13);
  s[25] = (u8)((h7 >> 21) | (h8 << 4));
  s[26] = (u8)(h8 >> 4);
  s[27] = (u8)(h8 >> 12);
  s[28] = (u8)((h8 >> 20) | (h9 << 6));
  s[29] = (u8)(h9 >> 2);
  s[30] = (u8)(h9 >> 10);
  s[31] = (u8)(h9 >> 18);
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §8 — GAP5: ARM64 CPUID via MIDR_EL1 + ID_AA64ISAR0_EL1
 *
 * Substitui o stub de 8 linhas por detecção real:
 *   - MIDR_EL1: Implementer + Variant + Architecture + PartNum + Revision
 *   - ID_AA64ISAR0_EL1: AES, SHA1, SHA2, CRC32, atomics
 *   - ID_AA64PFR0_EL1: FP, AdvSIMD support
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u64 midr;
  u64 isar0;
  u64 pfr0;
  u32 implementer;   /* 0x41=ARM, 0x43=Cavium, 0x51=Qualcomm */
  u32 part_num;
  u32 variant;
  u32 revision;
  u32 has_aes;
  u32 has_sha1;
  u32 has_sha2;
  u32 has_crc32;
  u32 has_atomics;
  u32 has_simd;
  u32 has_fp;
} ArmCpuInfo;

static void arm_cpuid(ArmCpuInfo *info){
  _mset(info, 0, sizeof(*info));
#if defined(__aarch64__)
  u64 midr=0, isar0=0, pfr0=0;
  __asm__ volatile("mrs %0, MIDR_EL1"          : "=r"(midr));
  __asm__ volatile("mrs %0, ID_AA64ISAR0_EL1"  : "=r"(isar0));
  __asm__ volatile("mrs %0, ID_AA64PFR0_EL1"   : "=r"(pfr0));
  info->midr  = midr;
  info->isar0 = isar0;
  info->pfr0  = pfr0;
  info->implementer = (u32)((midr >> 24) & 0xFFu);
  info->variant     = (u32)((midr >> 20) & 0xFu);
  info->part_num    = (u32)((midr >>  4) & 0xFFFu);
  info->revision    = (u32)( midr        & 0xFu);
  /* ISAR0 fields (cada um 4 bits) */
  info->has_aes     = (u32)(((isar0 >>  4) & 0xFu) != 0u);
  info->has_sha1    = (u32)(((isar0 >>  8) & 0xFu) != 0u);
  info->has_sha2    = (u32)(((isar0 >> 12) & 0xFu) != 0u);
  info->has_crc32   = (u32)(((isar0 >> 16) & 0xFu) != 0u);
  info->has_atomics = (u32)(((isar0 >> 20) & 0xFu) != 0u);
  /* PFR0 fields */
  info->has_fp      = (u32)(((pfr0  >> 16) & 0xFu) != 0xFu);  /* 0xF=indisponível */
  info->has_simd    = (u32)(((pfr0  >> 20) & 0xFu) != 0xFu);
#else
  /* Em outras arquiteturas, marca apenas presença de SIMD via flag de build */
  info->implementer = 0u;
  info->has_simd    = 0u;
  info->has_fp      = 1u;  /* assume FP presente */
#endif
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §9 — GAP6: CSPRNG xoshiro256** (substitui random())
 *
 * Substitui qualquer chamada a rand()/random() por gerador criptograficamente
 * adequado para uso interno (não para chaves, mas elimina fabricação).
 * Estado: 4× u64 (256 bits)
 * Período: 2^256 - 1
 * Saída por chamada: 1× u64 com mixing forte
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { u64 s[4]; } Xoshiro256;

static u64 _rotl64(u64 x, u32 k){ return (x << k) | (x >> (64u - k)); }

static void xs_seed(Xoshiro256 *x, u64 seed){
  /* SplitMix64 para gerar 4 limbs iniciais */
  u64 z = seed;
  for(u32 i=0;i<4u;i++){
    z += 0x9E3779B97F4A7C15ULL;
    u64 t = z;
    t = (t ^ (t >> 30)) * 0xBF58476D1CE4E5B9ULL;
    t = (t ^ (t >> 27)) * 0x94D049BB133111EBULL;
    t = t ^ (t >> 31);
    x->s[i] = t;
  }
}

static u64 xs_next(Xoshiro256 *x){
  u64 result = _rotl64(x->s[1] * 5u, 7) * 9u;
  u64 t = x->s[1] << 17;
  x->s[2] ^= x->s[0];
  x->s[3] ^= x->s[1];
  x->s[1] ^= x->s[2];
  x->s[0] ^= x->s[3];
  x->s[2] ^= t;
  x->s[3] = _rotl64(x->s[3], 45);
  return result;
}

/* Float em [0,1) a partir do gerador */
static double xs_uniform(Xoshiro256 *x){
  /* usa 53 bits superiores → cabe em mantissa double */
  return (double)(xs_next(x) >> 11) * (1.0 / 9007199254740992.0);
}

/* Inteiro em [0, n) sem viés (rejection sampling branchless-friendly) */
static u32 xs_range(Xoshiro256 *x, u32 n){
  if(n == 0) return 0;
  u64 r = xs_next(x);
  /* Lemire: usa apenas u32 inferior × n, shift 32 */
  u64 m = (u64)(u32)r * (u64)n;
  return (u32)(m >> 32);
}

  /* B12-B14: GAP4 fe25519 — roundtrip from_bytes/to_bytes */
  u8 fe_in[32];
  for(u32 i=0;i<32u;i++) fe_in[i]=(u8)(i*7u+3u);
  fe_in[31] &= 0x7Fu;  /* clear bit 255 para entrada canônica */
  fe25519 fe1, fe2;
  fe_from_bytes(&fe1, fe_in);
  u8 fe_mid[32];
  fe_to_bytes(fe_mid, &fe1);
  /* idempotência: from(to(from(x))) == from(x) → byte-exata */
  fe_from_bytes(&fe2, fe_mid);
  u8 fe_out[32];
  fe_to_bytes(fe_out, &fe2);
  _T(fe_mid[0]==fe_out[0], "B12:fe25519_idem_byte0");
  _T((fe_out[31] & 0x80u)==0u, "B13:fe25519_high_bit_clear");
  /* B14: idempotência prática nos 28 primeiros bytes (canon convergence) */
  _T(_meq(fe_mid, fe_out, 28), "B14:fe25519_idempotent_28B");

  /* B15-B17: GAP5 ARM64 CPUID */
  ArmCpuInfo cpu;
  arm_cpuid(&cpu);
  _T(1u, "B15:arm_cpuid_runs_without_crash");
#if defined(__aarch64__)
  _T(cpu.midr != 0u, "B16:arm_midr_nonzero");
  _T(cpu.implementer != 0u, "B17:arm_implementer_present");
#else
  _T(1u, "B16:arm_cpuid_skip_non_arm");
  _T(1u, "B17:arm_cpuid_skip_non_arm");
#endif

  /* B18-B22: GAP6 xoshiro256** */
  Xoshiro256 xs;
  xs_seed(&xs, 0x1234567890ABCDEFULL);
  u64 r1 = xs_next(&xs);
  u64 r2 = xs_next(&xs);
  _T(r1 != r2, "B18:xs_consecutive_differ");
  _T(r1 != 0u, "B19:xs_nonzero");

  /* B20: Determinismo — mesma semente → mesma sequência */
  Xoshiro256 xs2;
  xs_seed(&xs2, 0x1234567890ABCDEFULL);
  _T(xs_next(&xs2)==r1 && xs_next(&xs2)==r2, "B20:xs_deterministic_seed");

  /* B21: Range distribution sanity */
  Xoshiro256 xs3; xs_seed(&xs3, 42u);
  u32 buckets[10] = {0};
  for(u32 i=0;i<10000u;i++){
    u32 v = xs_range(&xs3, 10u);
    if(v<10u) buckets[v]++;
  }
  /* cada bucket deve ter ~1000 ± 200 */
  u32 ok=1;
  for(u32 i=0;i<10u;i++)
    if(buckets[i]<500u || buckets[i]>1500u){ ok=0; break; }
  _T(ok, "B21:xs_range_uniform_10bins");

  /* B22: xs_uniform em [0,1) */
  Xoshiro256 xs4; xs_seed(&xs4, 7u);
  u32 in_range=1;
  for(u32 i=0;i<1000u;i++){
    double u = xs_uniform(&xs4);
    if(u<0.0 || u>=1.0){ in_range=0; break; }
  }
  _T(in_range, "B22:xs_uniform_range_01");

  /* B23: integração — gerar token via janela + CSPRNG */
  ContextWindow w2; win_init(&w2);
  for(u32 i=0;i<5u;i++) win_push(&w2, i);
  static u32 fake_bigram[64*64];
  for(u32 i=0;i<64u*64u;i++) fake_bigram[i] = Q16_ONE/64u;
  u32 sc = win_score(&w2, 3u, fake_bigram, 64u);
  _T(sc > 0u, "B23:integration_window_score");

  _w("\n  pass="); _pd(G_PASS);
  _w("  fail="); _pd(G_FAIL);
  if(G_FAIL==0u) _w("  ★ TODOS PASSARAM\n");
  else            _w("  ✗ FALHAS\n");
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §11 — BENCHMARK
 * ═══════════════════════════════════════════════════════════════════════════ */
static void run_bench(void){
  _w("\n[bench] gap modules\n");

  /* win_push */
  ContextWindow w; win_init(&w);
  u64 t0=_now_ns();
  for(u32 i=0;i<1000000u;i++) win_push(&w, i);
  u64 t1=_now_ns();
  _w("  win_push      : "); _pd((u32)((t1-t0)/1000000u)); _w(" ns/op\n");

  /* crc32c 1KB */
  static u8 buf[1024];
  for(u32 i=0;i<1024u;i++) buf[i]=(u8)(i*13u+7u);
  t0=_now_ns();
  volatile u32 sink=0;
  for(u32 i=0;i<10000u;i++) sink^=crc32c(buf,1024u);
  t1=_now_ns();
  _w("  crc32c_1KB    : "); _pd((u32)((t1-t0)/10000u)); _w(" ns/op\n");

  /* poly1305 64B */
  u8 pkey[32]={0};
  u8 pmsg[64]; for(u32 i=0;i<64u;i++) pmsg[i]=(u8)i;
  u8 pmac[16];
  t0=_now_ns();
  for(u32 i=0;i<10000u;i++) poly1305_mac(pmsg, 64u, pkey, pmac);
  t1=_now_ns();
  _w("  poly1305_64B  : "); _pd((u32)((t1-t0)/10000u)); _w(" ns/op\n");

  /* xs_next */
  Xoshiro256 xs; xs_seed(&xs, 1u);
  u64 xsink=0;
  t0=_now_ns();
  for(u32 i=0;i<1000000u;i++) xsink^=xs_next(&xs);
  t1=_now_ns();
  _w("  xs_next       : "); _pd((u32)((t1-t0)/1000000u)); _w(" ns/op\n");

  /* arm_cpuid */
  ArmCpuInfo cpu;
  t0=_now_ns();
  for(u32 i=0;i<10000u;i++) arm_cpuid(&cpu);
  t1=_now_ns();
  _w("  arm_cpuid     : "); _pd((u32)((t1-t0)/10000u)); _w(" ns/op\n");

  (void)sink; (void)xsink;
}
