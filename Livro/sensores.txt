# ─────────────────────────────────────────────────────────────────────────────
# SEMENTE 01 — MICROFONE → ENTROPIA SONORA
# Canal χ do vetor toroidal
# ─────────────────────────────────────────────────────────────────────────────
seed_01_mic_entropy() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 01: Microfone → Entropia Espectral → canal χ (toroidal)
Captura 2 s de áudio a 16 kHz, calcula entropia de Shannon do
espectro FFT e mapeia para Q16 [0, 65535].
Saída: JSON {chi_q16, entropy_float, freq_peak_hz, crc32c}
"""
import subprocess, struct, math, hashlib, json, os, tempfile, time

SAMPLE_RATE = 16000
DURATION    = 2        # segundos
POLY_CRC32C = 0x82F63B78

def crc32c(data: bytes) -> int:
    crc = 0xFFFFFFFF
    for b in data:
        crc ^= b
        for _ in range(8):
            crc = (crc >> 1) ^ (POLY_CRC32C & -(crc & 1))
    return (~crc) & 0xFFFFFFFF

def record_pcm(secs: int, rate: int) -> bytes:
    """Grava AAC via termux-microphone-record, converte para PCM16 com ffmpeg."""
    aac = tempfile.mktemp(suffix='.aac')
    subprocess.run(
        ['termux-microphone-record', '-e', 'aac', '-r', str(rate),
         '-c', '1', '-l', str(secs), '-f', aac],
        check=True, capture_output=True
    )
    pcm = subprocess.run(
        ['ffmpeg', '-y', '-i', aac, '-ar', str(rate), '-ac', '1',
         '-f', 's16le', '-'],
        capture_output=True, check=True
    ).stdout
    os.unlink(aac)
    return pcm

def shannon_entropy(spectrum: list) -> float:
    total = sum(spectrum) + 1e-12
    return -sum((v/total) * math.log2(v/total + 1e-12) for v in spectrum)

def pcm_to_spectrum(raw: bytes, rate: int, n_bins: int = 512) -> list:
    """FFT manual via Goertzel-like: bins de magnitude"""
    samples = [struct.unpack_from('<h', raw, i*2)[0] / 32768.0
               for i in range(len(raw) // 2)]
    # DFT simplificado para N_bins primeiros
    N = min(n_bins * 2, len(samples))
    mag = []
    for k in range(n_bins):
        re = im = 0.0
        for n, s in enumerate(samples[:N]):
            angle = -2 * math.pi * k * n / N
            re += s * math.cos(angle)
            im += s * math.sin(angle)
        mag.append(math.sqrt(re*re + im*im))
    return mag

def run():
    raw = record_pcm(DURATION, SAMPLE_RATE)
    spec = pcm_to_spectrum(raw, SAMPLE_RATE)
    h = shannon_entropy(spec)
    # Normaliza entropia para Q16: max teórico = log2(512) ≈ 9
    chi_q16 = int(min(h / 9.0, 1.0) * 65535)
    peak_bin = spec.index(max(spec))
    peak_hz  = int(peak_bin * SAMPLE_RATE / (len(spec) * 2))
    checksum = crc32c(raw)
    out = {
        "channel":     "chi",
        "timestamp_ns": time.time_ns(),
        "chi_q16":      chi_q16,
        "entropy":      round(h, 4),
        "peak_hz":      peak_hz,
        "crc32c":       hex(checksum),
        "samples":      len(raw) // 2,
    }
    print(json.dumps(out, indent=2))

if __name__ == "__main__":
    run()
SEED
}

# ─────────────────────────────────────────────────────────────────────────────
# SEMENTE 02 — DETECTOR fΩ ∈ [963, 999] Hz
# Detecta presença da banda de coerência RAFAELIA no ambiente sonoro
# ─────────────────────────────────────────────────────────────────────────────
seed_02_fomega_detector() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 02: Detector fΩ ∈ [963, 999] Hz
Calcula potência de Goertzel em 963 Hz e 999 Hz relativamente ao
espectro total. Emite evento JSON quando coerência > limiar.
Sinal de coerência RAFAELIA: fΩ = 963 ↔ 999 Hz.
"""
import subprocess, struct, math, json, time, os, tempfile

RATE    = 16000
SECS    = 2
F_LOW   = 963.0   # Hz
F_HIGH  = 999.0   # Hz
THRESH  = 0.05    # 5% de potência espectral total = evento de coerência

def record_pcm_samples(secs, rate):
    aac = tempfile.mktemp(suffix='.aac')
    subprocess.run(['termux-microphone-record','-e','aac','-r',str(rate),
                    '-c','1','-l',str(secs),'-f',aac],
                   check=True, capture_output=True)
    raw = subprocess.run(['ffmpeg','-y','-i',aac,'-ar',str(rate),'-ac','1',
                         '-f','s16le','-'], capture_output=True, check=True).stdout
    os.unlink(aac)
    return [struct.unpack_from('<h', raw, i*2)[0] / 32768.0
            for i in range(len(raw) // 2)]

def goertzel(samples, target_hz, rate):
    """Algoritmo de Goertzel — energia na frequência alvo."""
    N = len(samples)
    k = int(0.5 + N * target_hz / rate)
    w = 2 * math.pi * k / N
    cos_w = math.cos(w)
    coeff = 2 * cos_w
    s0 = s1 = s2 = 0.0
    for x in samples:
        s0 = x + coeff * s1 - s2
        s2 = s1; s1 = s0
    power = s1*s1 + s2*s2 - s1*s2*coeff
    return power

def total_power(samples):
    return sum(x*x for x in samples) + 1e-12

def run():
    while True:
        samples = record_pcm_samples(SECS, RATE)
        tp  = total_power(samples)
        p_low  = goertzel(samples, F_LOW,  RATE)
        p_high = goertzel(samples, F_HIGH, RATE)
        ratio_low  = p_low  / tp
        ratio_high = p_high / tp
        coherence  = (ratio_low + ratio_high) / 2.0
        event = coherence >= THRESH
        out = {
            "channel":       "f_omega",
            "timestamp_ns":  time.time_ns(),
            "f963_ratio":    round(ratio_low,  6),
            "f999_ratio":    round(ratio_high, 6),
            "coherence":     round(coherence,  6),
            "event_OMEGA":   event,
            "Phi_ethica_approx": round(1.0 - coherence, 4),
        }
        print(json.dumps(out))
        if event:
            subprocess.run(['termux-notification','--title','fΩ DETECTED',
                '--content',f'Coerência {coherence:.4f} em [{F_LOW},{F_HIGH}] Hz',
                '--priority','high'], capture_output=True)
        time.sleep(SECS)

if __name__ == "__main__":
    run()
SEED
}

# ─────────────────────────────────────────────────────────────────────────────
# SEMENTE 03 — ACELERÔMETRO + GIROSCÓPIO → VETOR u,v TOROIDAL
# Canal u (aceleração) + canal v (rotação)
# ─────────────────────────────────────────────────────────────────────────────
seed_03_imu_toroidal() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 03: IMU → Estado Toroidal u,v
Lê acelerômetro + giroscópio via termux-sensor,
calcula magnitude 3D, normaliza para Q16, mapeia para T^7.
Detecta quedas (|a| > 2g por Δt), vibração de máquinas (FFT-Z).
"""
import subprocess, json, math, time

Q16 = 65536
G   = 9.81

def read_sensors(sensors: list, n_samples: int = 10, delay_ms: int = 50):
    """Lê N amostras dos sensores listados."""
    cmd = ['termux-sensor', '-s', ','.join(sensors),
           '-n', str(n_samples), '-d', str(delay_ms)]
    out = subprocess.run(cmd, capture_output=True, text=True, timeout=30).stdout
    data = {}
    for line in out.strip().split('\n'):
        try:
            obj = json.loads(line)
            for name, payload in obj.items():
                if name not in data:
                    data[name] = []
                data[name].append(payload.get('values', []))
        except Exception:
            pass
    return data

def mag3(v): return math.sqrt(sum(x*x for x in v)) if len(v) >= 3 else 0.0

def to_q16(value, vmax): return int(min(abs(value) / vmax, 1.0) * (Q16 - 1))

def run():
    sensors = ['accelerometer', 'gyroscope']
    while True:
        data = read_sensors(sensors, n_samples=20, delay_ms=25)
        acc_samples = data.get('accelerometer', [])
        gyr_samples = data.get('gyroscope', [])

        if not acc_samples:
            time.sleep(0.5); continue

        # Magnitude média de aceleração
        acc_mags = [mag3(s) for s in acc_samples if len(s) >= 3]
        acc_mean = sum(acc_mags) / len(acc_mags) if acc_mags else 0.0
        acc_peak = max(acc_mags) if acc_mags else 0.0

        # Magnitude média de rotação angular (rad/s)
        gyr_mags = [mag3(s) for s in gyr_samples if len(s) >= 3]
        gyr_mean = sum(gyr_mags) / len(gyr_mags) if gyr_mags else 0.0

        # Mapeamento para Q16 toroidal
        # u = aceleração / (4G) → [0,1] → Q16
        u_q16 = to_q16(acc_mean, 4 * G)
        # v = rotação / (10 rad/s) → [0,1] → Q16
        v_q16 = to_q16(gyr_mean, 10.0)

        # Detecção de queda: pico > 3G seguido de queda abaixo de 0.3G
        fall_event = acc_peak > 3 * G and acc_mean < 0.3 * G

        # Vibração de máquina: desvio padrão alto em Z
        z_vals = [s[2] for s in acc_samples if len(s) >= 3]
        z_mean = sum(z_vals) / len(z_vals) if z_vals else 0.0
        z_std  = math.sqrt(sum((x - z_mean)**2 for x in z_vals) / len(z_vals)) if z_vals else 0.0
        machine_vib = z_std > 1.5  # m/s² stddev > 1.5 = vibração

        out = {
            "channel":       "u_v",
            "timestamp_ns":  time.time_ns(),
            "u_q16":         u_q16,
            "v_q16":         v_q16,
            "acc_mean_ms2":  round(acc_mean, 3),
            "acc_peak_ms2":  round(acc_peak, 3),
            "gyr_mean_rads": round(gyr_mean, 3),
            "z_std":         round(z_std, 4),
            "fall_event":    fall_event,
            "machine_vib":   machine_vib,
        }
        print(json.dumps(out))
        if fall_event:
            subprocess.run(['termux-vibrate','-d','1000','-f'])
            subprocess.run(['termux-notification','--title','QUEDA DETECTADA',
                            '--content',f'|a|={acc_peak:.1f} m/s²','--priority','high'])
        time.sleep(0.5)

if __name__ == "__main__":
    run()
SEED
}

# ─────────────────────────────────────────────────────────────────────────────
# SEMENTE 04 — MAGNETÔMETRO → HEADING + CANAL ψ
# ─────────────────────────────────────────────────────────────────────────────
seed_04_magnetometer() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 04: Magnetômetro → Heading 0-359° → canal ψ toroidal
Detecta anomalias de campo magnético (objetos ferrosos próximos,
EMI industrial, mapeamento de campo local).
"""
import subprocess, json, math, time

def read_mag(n=10, delay_ms=50):
    out = subprocess.run(['termux-sensor','-s','magnetic_field',
                         '-n',str(n),'-d',str(delay_ms)],
                        capture_output=True, text=True, timeout=20).stdout
    samples = []
    for line in out.strip().split('\n'):
        try:
            obj = json.loads(line)
            for v in obj.values():
                vals = v.get('values', [])
                if len(vals) >= 3:
                    samples.append(vals)
        except Exception:
            pass
    return samples

def heading_deg(x, y):
    h = math.degrees(math.atan2(y, x))
    return h + 360 if h < 0 else h

def run():
    baseline = None
    while True:
        samples = read_mag(n=20)
        if not samples:
            time.sleep(0.5); continue
        xs = [s[0] for s in samples]
        ys = [s[1] for s in samples]
        zs = [s[2] for s in samples]
        mx = sum(xs) / len(xs)
        my = sum(ys) / len(ys)
        mz = sum(zs) / len(zs)
        mag_total = math.sqrt(mx*mx + my*my + mz*mz)
        hdg = heading_deg(mx, my)

        # Calibração automática de linha de base (primeiros 5 s)
        if baseline is None:
            baseline = mag_total
        anomaly = abs(mag_total - baseline) > baseline * 0.15  # 15% desvio

        # canal ψ: heading / 360 → Q16
        psi_q16 = int((hdg / 360.0) * 65535)

        out = {
            "channel":   "psi",
            "ts_ns":     time.time_ns(),
            "psi_q16":   psi_q16,
            "heading":   round(hdg, 1),
            "mag_uT":    round(mag_total, 2),
            "baseline_uT": round(baseline, 2),
            "anomaly":   anomaly,
            "vec_uT":    [round(mx,2), round(my,2), round(mz,2)],
        }
        print(json.dumps(out))
        time.sleep(0.25)

if __name__ == "__main__":
    run()
SEED
}

# SEMENTE 05 — LUZ AMBIENTE + PROXIMIDADE → CANAL ρ
# ─────────────────────────────────────────────────────────────────────────────
seed_05_light_prox() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 05: Luz + Proximidade → canal ρ (ambiente)
Calcula ciclo dia/noite, detecta cobertura do sensor (mão/bolso),
ajusta brilho de notificações, alimenta índice de contexto ambiental.
"""
import subprocess, json, math, time

def read_env(n=5, delay_ms=100):
    out = subprocess.run(['termux-sensor','-s','light,proximity',
                         '-n',str(n),'-d',str(delay_ms)],
                        capture_output=True, text=True, timeout=15).stdout
    light = []; prox = []
    for line in out.strip().split('\n'):
        try:
            obj = json.loads(line)
            for k, v in obj.items():
                vals = v.get('values', [0])
                if 'light' in k.lower():    light.append(vals[0])
                if 'proxim' in k.lower():   prox.append(vals[0])
        except Exception:
            pass
    return light, prox

def lux_to_q16(lux, max_lux=10000.0):
    # Log escala: log2(lux+1) / log2(max+1) → Q16
    return int(min(math.log2(lux + 1) / math.log2(max_lux + 1), 1.0) * 65535)

def classify_light(lux):
    if lux < 1:   return "escuro"
    if lux < 100: return "interior_baixo"
    if lux < 500: return "interior"
    if lux < 2000: return "nublado"
    return "sol_direto"

def run():
    while True:
        lights, proxs = read_env()
        lux    = sum(lights) / len(lights) if lights else 0.0
        prox_v = sum(proxs)  / len(proxs)  if proxs  else 9999.0
        covered = prox_v < 3.0  # tipicamente 0=coberto, 5=longe
        rho_q16 = lux_to_q16(lux)
        out = {
            "channel":   "rho",
            "ts_ns":     time.time_ns(),
            "rho_q16":   rho_q16,
            "lux":       round(lux, 2),
            "prox_cm":   round(prox_v, 1),
            "covered":   covered,
            "context":   classify_light(lux),
        }
        print(json.dumps(out))
        time.sleep(1.0)

if __name__ == "__main__":
    run()
SEED
}

# ─────────────────────────────────────────────────────────────────────────────
# SEMENTE 06 — GPS → CANAL δ TOROIDAL + TRAIL MERKLE
# ─────────────────────────────────────────────────────────────────────────────
seed_06_gps_merkle() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 06: GPS → canal δ + trilha de auditoria Merkle imutável
Calcula δ de deslocamento entre fixes, detecta geofence por coerência
geométrica, gera hash de posição CRC32C encadeado (Merkle simples).
"""
import subprocess, json, math, time, hashlib

POLY = 0x82F63B78

def crc32c(data: bytes) -> int:
    crc = 0xFFFFFFFF
    for b in data:
        crc ^= b
        for _ in range(8):
            crc = (crc >> 1) ^ (POLY & -(crc & 1))
    return (~crc) & 0xFFFFFFFF

def get_gps():
    out = subprocess.run(['termux-location','-p','gps','-r','once'],
                        capture_output=True, text=True, timeout=30).stdout
    return json.loads(out)

def haversine_m(lat1, lon1, lat2, lon2):
    R = 6371000
    f1, f2 = math.radians(lat1), math.radians(lat2)
    df = math.radians(lat2 - lat1)
    dl = math.radians(lon2 - lon1)
    a = math.sin(df/2)**2 + math.cos(f1)*math.cos(f2)*math.sin(dl/2)**2
    return 2 * R * math.asin(math.sqrt(a))

def pos_to_q16(lat, lon):
    # lat [-90,90] → [0,1] → Q16
    lat_q16 = int((lat + 90) / 180 * 65535)
    # lon [-180,180] → [0,1] → Q16
    lon_q16 = int((lon + 180) / 360 * 65535)
    return lat_q16, lon_q16

def run():
    chain_hash = 0xFFFFFFFF
    prev_lat = prev_lon = None
    trail = []

    while True:
        try:
            fix = get_gps()
        except Exception as e:
            print(json.dumps({"error": str(e)}))
            time.sleep(10); continue

        lat  = fix.get('latitude',  0.0)
        lon  = fix.get('longitude', 0.0)
        alt  = fix.get('altitude',  0.0)
        acc  = fix.get('accuracy',  999.0)
        spd  = fix.get('speed',     0.0)

        delta_m = 0.0
        if prev_lat is not None:
            delta_m = haversine_m(prev_lat, prev_lon, lat, lon)

        lat_q16, lon_q16 = pos_to_q16(lat, lon)
        delta_q16 = int(min(delta_m / 1000, 1.0) * 65535)  # / 1km

        # Merkle: hash desta fix = CRC32C(prev_hash || lat_bytes || lon_bytes)
        payload = (chain_hash.to_bytes(4, 'little') +
                   int(lat * 1e6).to_bytes(4, 'little', signed=True) +
                   int(lon * 1e6).to_bytes(4, 'little', signed=True))
        chain_hash = crc32c(payload)

        # Toroidal δ: movimento / (πφ) rad representado em Q16
        PHI_PI = 3.14159265 * 1.61803398
        delta_torus = int(min(delta_m / (PHI_PI * 100), 1.0) * 65535)

        entry = {
            "channel":    "delta",
            "ts_ns":      time.time_ns(),
            "lat":        round(lat, 7),
            "lon":        round(lon, 7),
            "alt_m":      round(alt, 1),
            "acc_m":      round(acc, 1),
            "speed_ms":   round(spd, 2),
            "delta_m":    round(delta_m, 2),
            "delta_q16":  delta_q16,
            "lat_q16":    lat_q16,
            "lon_q16":    lon_q16,
            "torus_delta":delta_torus,
            "merkle_crc": hex(chain_hash),
            "trail_len":  len(trail),
        }
        trail.append({"hash": hex(chain_hash), "lat": lat, "lon": lon, "ts": entry["ts_ns"]})
        print(json.dumps(entry))
        prev_lat, prev_lon = lat, lon
        time.sleep(5)

if __name__ == "__main__":
    run()
SEED
}

# ─────────────────────────────────────────────────────────────────────────────
# SEMENTE 07 — CÂMERA → CRC32C DE FRAME + CANAL σ
# ─────────────────────────────────────────────────────────────────────────────
seed_07_camera_crc() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 07: Câmera → CRC32C de frame → canal σ + diff score
Detecta adulteração/mudança de cena por diferença de hash entre frames.
Alimenta scanner CTI: bad_event se miss_score > threshold.
"""
import subprocess, json, time, os, tempfile

POLY = 0x82F63B78
CAMERA_ID = "0"  # 0=back, 1=front

def crc32c(data: bytes) -> int:
    crc = 0xFFFFFFFF
    for b in data:
        crc ^= b
        for _ in range(8):
            crc = (crc >> 1) ^ (POLY & -(crc & 1))
    return (~crc) & 0xFFFFFFFF

def capture_jpeg() -> bytes:
    path = tempfile.mktemp(suffix='.jpg')
    subprocess.run(['termux-camera-photo', '-c', CAMERA_ID, path],
                  check=True, capture_output=True, timeout=15)
    with open(path, 'rb') as f:
        data = f.read()
    os.unlink(path)
    return data

def jpeg_mean_byte(data: bytes) -> float:
    # Amostra cada 100 bytes para mean (proxy de brilho)
    sampled = data[::100]
    return sum(sampled) / len(sampled) if sampled else 0

def run():
    prev_crc = None
    prev_mean = None
    while True:
        try:
            img = capture_jpeg()
        except Exception as e:
            print(json.dumps({"error": str(e)}))
            time.sleep(5); continue

        cur_crc  = crc32c(img)
        cur_mean = jpeg_mean_byte(img)

        # miss_score: quanto o CRC mudou (branchless: 1 se diferente, 0 se igual)
        crc_changed = int(prev_crc is not None and cur_crc != prev_crc)
        mean_delta  = abs(cur_mean - prev_mean) if prev_mean is not None else 0.0

        # canal σ: CRC mapped to Q16 (low 16 bits)
        sigma_q16 = cur_crc & 0xFFFF

        bad_event = mean_delta > 20.0  # diferença de brilho > 20/255

        out = {
            "channel":    "sigma",
            "ts_ns":      time.time_ns(),
            "sigma_q16":  sigma_q16,
            "crc32c":     hex(cur_crc),
            "size_kb":    round(len(img) / 1024, 1),
            "mean_byte":  round(cur_mean, 2),
            "mean_delta": round(mean_delta, 2),
            "crc_changed":crc_changed,
            "bad_event":  bad_event,
        }
        print(json.dumps(out))
        if bad_event:
            subprocess.run(['termux-notification','--title','σ CHANGE DETECTED',
                '--content', f'Mean delta={mean_delta:.1f}'], capture_output=True)
        prev_crc  = cur_crc
        prev_mean = cur_mean
        time.sleep(10)

if __name__ == "__main__":
    run()
SEED
}

ir_tx() {

# Encode NEC (address, command)
nec_encode() {
  local addr=$1 cmd=$2
  local addr_inv=$(( addr ^ 0xFF ))
  local cmd_inv=$(( cmd  ^ 0xFF ))
  # Lead: 9000µs + 4500µs, cada bit: 560µs + (560 ou 1690)µs
  local pattern="9000,4500"
  for bit in $(python3 -c "
a,ai,c,ci=$addr,$addr_inv,$cmd,$cmd_inv
bits=[]
for v in [a,ai,c,ci]:
    for i in range(8):
        bits.append((v>>i)&1)
print(' '.join(str(b) for b in bits))
"); do
    if [ "$bit" = "1" ]; then
      pattern="${pattern},560,1690"
    else
      pattern="${pattern},560,560"
    fi
  done
  pattern="${pattern},560"
  echo "$pattern"
}

tv_power() {

ac_power() {

# ─────────────────────────────────────────────────────────────────────────────
# SEMENTE 12 — INTEGRADOR TOROIDAL (u,v,ψ,χ,ρ,δ,σ) → BITOMEGA
# O kernel central que decide FLOW ou NOISE em cada canal
# ─────────────────────────────────────────────────────────────────────────────
seed_12_toroidal_integrator() {
cat << 'SEED'
#!/usr/bin/env python3
"""
SEMENTE 12: Integrador Toroidal — BitOmega 10-state
Lê os 7 canais dos sensores (JSON linha por linha via pipes),
mantém estado T^7 com EMA (α=0.25), detecta atrator mais próximo
dentre 42, decide FLOW ou NOISE, emite evento Ω se coerência alta.

Integra: u(acc), v(gyr), ψ(mag), χ(audio), ρ(light), δ(gps), σ(camera)
"""
import json, math, time, sys, subprocess
from typing import Optional

Q16 = 65536
ALPHA = 0.25
N_ATTRACTORS = 42
N_DIM = 7

# Estados BitOmega (10 estados)
BO_STATES = ["INIT","FLOW","NOISE","COHERENT","INCOHERENT",
             "ATTRACTOR","TRANSITION","VOID","DELTA","OMEGA"]

def gen_attractors():
    """42 atratores em T^7 — mesmo algoritmo do GAIA-BBS."""
    attrs = []
    Q16_SPIRAL = 56756  # √3/2 × 65536
    Q16_PHI    = 105965 # φ × 65536
    for k in range(N_ATTRACTORS):
        sp = Q16
        row = []
        for i in range(N_DIM):
            base   = (k * Q16 // N_ATTRACTORS * sp) >> 16
            offset = (i * Q16 // N_DIM * Q16_PHI) >> 16
            row.append((base + offset) & 0xFFFF)
            sp = (sp * Q16_SPIRAL) >> 16
        attrs.append(row)
    return attrs

def torus_dist(a, b):
    s = 0
    for x, y in zip(a, b):
        d = abs(x - y)
        if d > Q16 // 2: d = Q16 - d
        s += d
    return s

def nearest_attractor(state, attractors):
    best_idx, best_dist = 0, 0xFFFFFFFF
    for i, a in enumerate(attractors):
        d = torus_dist(state, a)
        if d < best_dist:
            best_dist = d; best_idx = i
    return best_idx, best_dist

def ema_update(old, new, alpha=ALPHA):
    return int((1 - alpha) * old + alpha * new)

def coherence_score(state, attractor, max_dist):
    d = torus_dist(state, attractor)
    return 1.0 - d / max_dist

def run():
    attractors = gen_attractors()
    # Estado inicial: centro do toro
    state = [Q16 // 2] * N_DIM
    bo_state = "INIT"
    max_dist = N_DIM * Q16 // 2

    channel_map = {
        "u_v":   (0, 1),   # u=acc, v=gyr
        "psi":   (2,),
        "chi":   (3,),
        "rho":   (4,),
        "delta": (5,),
        "sigma": (6,),
        "f_omega": None,   # evento especial
    }

    omega_count = 0
    cycle = 0

    print("BitOmega integrador iniciado. Aguardando JSON de sensores em stdin...")
    for line in sys.stdin:
        try:
            obj = json.loads(line)
        except Exception:
            continue

        ch = obj.get("channel", "")
        ts = obj.get("ts_ns", time.time_ns())

        # Atualiza canais
        if ch == "u_v":
            state[0] = ema_update(state[0], obj.get("u_q16", state[0]))
            state[1] = ema_update(state[1], obj.get("v_q16", state[1]))
        elif ch == "psi":
            state[2] = ema_update(state[2], obj.get("psi_q16", state[2]))
        elif ch == "chi":
            state[3] = ema_update(state[3], obj.get("chi_q16", state[3]))
        elif ch == "rho":
            state[4] = ema_update(state[4], obj.get("rho_q16", state[4]))
        elif ch == "delta":
            state[5] = ema_update(state[5], obj.get("delta_q16", state[5]))
        elif ch == "sigma":
            state[6] = ema_update(state[6], obj.get("sigma_q16", state[6]))
        elif ch == "f_omega" and obj.get("event_OMEGA"):
            omega_count += 1

        # Encontra atrator
        att_idx, att_dist = nearest_attractor(state, attractors)
        coh = coherence_score(state, attractors[att_idx], max_dist)

        # Decide estado BitOmega
        if coh > 0.85:   bo_state = "OMEGA"
        elif coh > 0.70: bo_state = "COHERENT"
        elif coh > 0.50: bo_state = "FLOW"
        elif coh > 0.30: bo_state = "ATTRACTOR"
        elif coh > 0.15: bo_state = "TRANSITION"
        else:            bo_state = "NOISE"

        cycle += 1
        out = {
            "cycle":        cycle,
            "ts_ns":        ts,
            "state_T7":     state,
            "attractor":    att_idx,
            "att_dist":     att_dist,
            "coherence":    round(coh, 4),
            "bo_state":     bo_state,
            "omega_events": omega_count,
            "Phi_ethica":   round(1.0 - coh / 1.0, 4),
            "R_t_next":     round(coh * 1.61803 * 0.86602, 4),
        }
        print(json.dumps(out))

        # Notificação de estado Ω
        if bo_state == "OMEGA":
            subprocess.run(['termux-notification','--id','7777',
                '--title','Ω ESTADO COERENTE',
                '--content', f'Atrator {att_idx} | coh={coh:.3f}'],
                capture_output=True)

if __name__ == "__main__":
    run()
SEED
}
