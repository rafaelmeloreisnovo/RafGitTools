/* Gate ∅∅∅0001123 (img01/20) — alfabeto discreto Fibonacci */
/* Reduz a: 0,0,0,0,1,1,2,3 mod 8 */
static u32 raf_gate(u32 t){
  static const u32 S1[5]={0,1,1,2,3};   /* 01123 */
  static const u32 S2[6]={0,0,1,1,2,3}; /* 001123 */
  u32 a=S1[t%5u], b=S2[t%6u];
  return (a + 2u*b + (t%4u)) & 7u;       /* gate ∈ {0..7} → 8 direções */
}

 * §4 — CRC32C (integridade do catálogo)
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 crc32c(const u8 *p, u32 n){
  u32 c=0xFFFFFFFFu;
  for(u32 i=0;i<n;i++){
    c^=(u32)p[i];
    for(u32 j=0;j<8u;j++) c=(c>>1)^(0x82F63B78u&-(c&1u));
  }
  return ~c;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §5 — VOCAB + TOKENIZER

#define VOCAB_MAX 8192u
typedef struct { char s[32]; u32 id,freq; u8 used; } VE;
typedef struct { VE *e; u32 sz,cap,unk,bos,eos; } Vocab;

static Vocab *vocab_new(void){
  Vocab *v=(Vocab*)_a(sizeof(Vocab)); if(!v)return (void*)0;
  v->e=(VE*)_a(sizeof(VE)*VOCAB_MAX); if(!v->e)return (void*)0;
  _mset(v->e,0,sizeof(VE)*VOCAB_MAX);
  v->sz=4; v->cap=VOCAB_MAX; v->unk=1; v->bos=2; v->eos=3;
  const char *sp[]={"<pad>","<unk>","<bos>","<eos>",(void*)0};
  for(i32 i=0;sp[i];i++){
    u32 h=5381u; const char *s=sp[i]; while(*s)h=((h<<5)+h)^(u8)*s++;h%=VOCAB_MAX;
    u32 n=0; while(sp[i][n]&&n<31u){v->e[h].s[n]=sp[i][n];n++;} v->e[h].s[n]=0;
    v->e[h].id=(u32)i; v->e[h].used=1;
  }
  return v;
}
static u32 vocab_ins(Vocab *v,const char *s){
  if(v->sz>=v->cap-1u)return v->unk;
  u32 h=5381u; const char *p=s; while(*p)h=((h<<5)+h)^(u8)*p++;h%=v->cap;
  for(u32 pr=0;pr<v->cap;pr++){
    VE *e=&v->e[h];
    if(!e->used){u32 i=0;while(s[i]&&i<31u){e->s[i]=s[i];i++;}e->s[i]=0;e->id=v->sz++;e->freq=1;e->used=1;return e->id;}
    u32 eq=1; for(u32 i=0;i<32u;i++){if(e->s[i]!=s[i]){eq=0;break;}if(!e->s[i])break;}
    if(eq){e->freq++;return e->id;}
    h=(h+1u)%v->cap;
  }
  return v->unk;
}
static u32 vocab_get(const Vocab *v,const char *s){
  u32 h=5381u; const char *p=s; while(*p)h=((h<<5)+h)^(u8)*p++;h%=v->cap;
  for(u32 pr=0;pr<v->cap;pr++){
    const VE *e=&v->e[h]; if(!e->used)return v->unk;
    u32 eq=1; for(u32 i=0;i<32u;i++){if(e->s[i]!=s[i]){eq=0;break;}if(!e->s[i])break;}
    if(eq){ return e->id; }
    h=(h+1u)%v->cap;
  }
  return v->unk;
}

static u32 tokenize(Vocab *v,const char *text,u32 len,u32 *ids,u32 max){
  char buf[32]; u32 bp=0,n=0;
  #define EMIT() do{if(bp>0&&n<max){buf[bp]=0;ids[n++]=vocab_ins(v,buf);bp=0;}}while(0)
  for(u32 i=0;i<len&&n<max;i++){
    u8 c=(u8)text[i];
    if(c>='A'&&c<='Z')c+=32u;
    if(c==' '||c=='\t'||c=='\n'||c=='\r'){EMIT();continue;}
    if(c=='.'||c==','||c==';'||c==':'||c=='!'||c=='?'){
      EMIT(); if(n<max){buf[0]=(char)c;buf[1]=0;ids[n++]=vocab_ins(v,buf);}continue;
    }
    if(bp<31u)buf[bp++]=(char)c; else{EMIT();buf[bp++]=(char)c;}
  }
  EMIT();
  #undef EMIT
  return n;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §6 — MODELO GEOLM-RAFAELIA
 *   Bigram mínimo com embedding EMBED_DIM=64 + contrator λ no decay
 *   Xn+1 = U(PC(Xn))·Φ + εn  (img11)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define EMBED_DIM 64u

static f32 fast_exp(f32 x){
  if(x>10.f){ return 22026.f; }
  if(x<-10.f){ return 4.54e-5f; }
  f32 x2=x*x,x3=x2*x,x4=x3*x;
  return 1.f+x+x2*.5f+x3*.16667f+x4*.04167f;
}

typedef struct {
  f32 embed[VOCAB_MAX][EMBED_DIM];
  f32 head_W[EMBED_DIM][VOCAB_MAX];
  f32 head_b[VOCAB_MAX];
  u32 vocab_size;
  u32 step;
  f32 best_loss;
  /* RAFAELIA: centróide por câmara (6 câmaras × EMBED_DIM) */
  f32 cam_center[6][EMBED_DIM];
  u32 cam_count[6];
} RafModel;

static RafModel *model_new(u32 vocab_size){
  RafModel *m=(RafModel*)_a(sizeof(RafModel)); if(!m)return (void*)0;
  _mset(m,0,sizeof(RafModel));
  m->vocab_size=vocab_size; m->step=0; m->best_loss=1e9f;
  u32 rng=0x52414641u; /* "RAFA" */
  f32 s=1.f/_sqrtf((f32)EMBED_DIM);
  for(u32 v=0;v<vocab_size;v++){
    for(u32 d=0;d<EMBED_DIM;d++){
      rng=rng*1664525u+1013904223u;
      m->embed[v][d]=((f32)(i32)rng)/2147483648.f*s;
    }
  }
  s=1.f/_sqrtf((f32)EMBED_DIM);
  for(u32 i=0;i<EMBED_DIM;i++){
    for(u32 j=0;j<vocab_size;j++){
      rng=rng*1664525u+1013904223u;
      m->head_W[i][j]=((f32)(i32)rng)/2147483648.f*s;
    }
  }
  /* inicializa centróides de câmara com ruído pequeno */
  for(u32 k=0;k<6u;k++){
    for(u32 d=0;d<EMBED_DIM;d++){
      rng=rng*1664525u+1013904223u;
      m->cam_center[k][d]=((f32)(i32)rng)/2147483648.f*0.01f;
    }
    m->cam_count[k]=1u;
  }
  return m;
}

/* Decay de coerência via λ=√3/2 no embedding (img02)
 * Aplica contrator: embed[v] = embed[v] × λ + novo × (1-λ) */
static void raf_coherence_decay(RafModel *m, u32 vid, u32 step){
  if(vid>=m->vocab_size)return;
  /* apenas a cada 100 steps para não destruir o embedding */
  if(step%100u!=0u)return;
  f32 *e=m->embed[vid];
  for(u32 d=0;d<EMBED_DIM;d++) e[d]*=RAF_LAMBDA;
}

/* Câmara ativa para um token (associa token à câmara via gate) */
static u32 raf_cam(u32 token_id, u32 step){
  u32 g=raf_gate(step ^ (token_id*7u));
  return g % 6u; /* mapeia gate 0..7 para câmara 0..5 */
}

/* Update do centróide de câmara (EMA — img11) */
static void raf_cam_update(RafModel *m, u32 cam, u32 vid){
  if(cam>=6u||vid>=m->vocab_size)return;
  u32 n=m->cam_count[cam];
  f32 *c=m->cam_center[cam];
  f32 *e=m->embed[vid];
  for(u32 d=0;d<EMBED_DIM;d++)
    c[d]=((f32)n*c[d]+e[d])/(f32)(n+1u);
  if(n<0xFFFFFFFFu) m->cam_count[cam]++;
}

/* Forward + backprop bigram com temperatura modulada pelo gate */
static f32 model_step_raf(RafModel *m, u32 ctx, u32 tgt,
                           f32 *logits, f32 lr, u32 step){
  if(ctx>=m->vocab_size||tgt>=m->vocab_size)return 0.f;
  f32 *emb=m->embed[ctx];

  /* logits = head_W^T × emb + head_b */
  for(u32 j=0;j<m->vocab_size;j++){
    f32 s=m->head_b[j];
    for(u32 i=0;i<EMBED_DIM;i++) s+=m->head_W[i][j]*emb[i];
    logits[j]=s;
  }

  /* softmax */
  f32 mx=logits[0];
  for(u32 j=1;j<m->vocab_size;j++) if(logits[j]>mx)mx=logits[j];
  f32 sum=0.f;
  for(u32 j=0;j<m->vocab_size;j++){logits[j]=fast_exp(logits[j]-mx);sum+=logits[j];}
  f32 inv=sum>1e-9f?1.f/sum:0.f;
  for(u32 j=0;j<m->vocab_size;j++) logits[j]*=inv;

  f32 loss=-_logf(logits[tgt]+1e-9f);

  /* backward */
  static f32 dL[VOCAB_MAX];
  for(u32 j=0;j<m->vocab_size;j++) dL[j]=logits[j]-(j==tgt?1.f:0.f);
  for(u32 i=0;i<EMBED_DIM;i++){
    f32 ei=emb[i];
    for(u32 j=0;j<m->vocab_size;j++) m->head_W[i][j]-=lr*dL[j]*ei;
  }
  for(u32 j=0;j<m->vocab_size;j++) m->head_b[j]-=lr*dL[j];
  for(u32 i=0;i<EMBED_DIM;i++){
    f32 g=0.f;
    for(u32 j=0;j<m->vocab_size;j++) g+=m->head_W[i][j]*dL[j];
    emb[i]-=lr*g;
  }

  /* RAFAELIA: atualiza câmara e aplica decay de coerência */
  u32 cam=raf_cam(ctx, step);
  raf_cam_update(m, cam, ctx);
  raf_coherence_decay(m, ctx, step);

  m->step++;
  return loss;
}

/* Geração com temperatura modulada pelo gate RAFAELIA (img19/20)
 *   temp alta → mais ruído → exploração (câmara IMAG/TRANS)
 *   temp baixa → mais foco → convergência (câmara FOCO/STRUCT) */
static void model_generate_raf(RafModel *m, Vocab *v, u32 start, u32 n){
  static f32 logits[VOCAB_MAX];
  u32 cur=start, rng=(u32)_now_ns();
  _w("[RAF>] ");
  for(u32 i=0;i<n;i++){
    u32 gate=raf_gate(m->step+i);
    /* temperatura mapeada pela câmara do gate */
    static const f32 CAM_TEMP[8]={0.6f,0.8f,1.0f,0.7f,1.2f,0.9f,1.4f,0.5f};
    f32 temp=CAM_TEMP[gate%8u];

    for(u32 j=0;j<m->vocab_size;j++){
      f32 s=m->head_b[j];
      for(u32 d=0;d<EMBED_DIM;d++) s+=m->head_W[d][j]*m->embed[cur][d];
      logits[j]=s/temp;
    }
    f32 mx=logits[0]; for(u32 j=1;j<m->vocab_size;j++) if(logits[j]>mx)mx=logits[j];
    f32 sum=0.f; for(u32 j=0;j<m->vocab_size;j++){logits[j]=fast_exp(logits[j]-mx);sum+=logits[j];}

    /* sampling multinomial */
    rng=rng*1664525u+1013904223u;
    f32 r=((f32)(rng>>1))/(f32)0x7FFFFFFFu*sum;
    u32 sampled=0; f32 acc=0.f;
    for(u32 j=0;j<m->vocab_size;j++){acc+=logits[j];if(acc>=r){sampled=j;break;}}

    /* imprime token + direção RAFAELIA */
    const VE *ve=&v->e[0];
    for(u32 k=0;k<VOCAB_MAX;k++){
      if(v->e[k].used&&v->e[k].id==sampled){ve=&v->e[k];break;}
    }
    _w(ve->s); _w("/"); _w(RAF_DIR[gate%8u]); _w(" ");
    cur=sampled;
    if(cur==v->eos)break;
  }
  _w("\n");
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §7 — TRAINING LOOP
 * ═══════════════════════════════════════════════════════════════════════════ */
static void train_loop(RafModel *m, const u32 *ids, u32 n_ids,
                        u32 epochs, f32 lr){
  static f32 logits[VOCAB_MAX];
  _w("[train] vocab="); _pd(m->vocab_size);
  _w("  tokens="); _pd(n_ids);
  _w("  epochs="); _pd(epochs);
  u64 t0=_now_ns();
  for(u32 ep=0;ep<epochs;ep++){
    f32 total=0.f; u32 cnt=0;
    for(u32 i=0;i+1u<n_ids;i++){
      total+=model_step_raf(m,ids[i],ids[i+1],logits,lr,m->step);
      cnt++;
    }
    f32 loss=total/(f32)(cnt?cnt:1u);
    if(loss<m->best_loss) m->best_loss=loss;
    if(ep%10u==0u||ep==epochs-1u){
      u64 t1=_now_ns();
      u32 ms=(u32)((t1-t0)/1000000u);
      _w("  ep="); _pd(ep);
      _w(" loss="); _pf2(loss);
      _w(" ms="); _pd(ms);
      t0=t1;
    }
  }
}

/* ═══════════════════════════════════════════════════════════════════════════

 *   Metas: Arena 68.8ns, CRC32 0.114GB/s, T7 36.49ns/step
 * ═══════════════════════════════════════════════════════════════════════════ */
static void run_bench(RafModel *m, const u32 *ids, u32 n_ids){
  static f32 logits[VOCAB_MAX];
  _w("\n[bench] RAFAELIA_HEXGRID metas ARM32:\n");
  _w("  Arena alloc target : 68.8 ns/alloc\n");
  _w("  CRC32C target      : 0.114 GB/s\n");
  _w("  T7 toroidal target : 36.49 ns/step\n");
  _w("  Q16 kernel target  : 11.03 ns/op\n");

  /* bench model_step */
  u32 N=500u; if(n_ids<N+1u)N=n_ids>1u?n_ids-1u:1u;
  u64 t0=_now_ns();
  for(u32 i=0;i<N&&i+1u<n_ids;i++)
    model_step_raf(m,ids[i],ids[i+1],logits,0.f,m->step+i);
  u64 t1=_now_ns();
  u32 ns_per=(u32)((t1-t0)/(u64)N);
  _w("  model_step/token   : "); _pd(ns_per); _w(" ns\n");

  /* bench crc32c */
  static u8 crc_buf[4096];
  for(u32 i=0;i<4096u;i++) crc_buf[i]=(u8)(i*13u+7u);
  t0=_now_ns();
  volatile u32 sink=0;
  for(u32 i=0;i<256u;i++) sink^=crc32c(crc_buf,4096u);
  t1=_now_ns();
  u64 bytes=(u64)256u*4096u;
  u64 ns=(t1-t0);
  /* GB/s = bytes / ns */
  u32 gbps_x100=(u32)(bytes*100u/(ns?ns:1u));
  _w("  CRC32C measured    : 0."); _pd(gbps_x100); _w(" GB/s\n");
  (void)sink;
}

static void run_tests(void){
  _w("\n[tests] RAFAELIA GeoLM\n");

  /* B0: CRC32C */
  const u8 v[]={'1','2','3','4','5','6','7','8','9'};
  _T(crc32c(v,9u)==0xE3069283u,"B0:crc32c_NIST");

  /* B1: RAF_GATE determinístico */
  u32 g0=raf_gate(0),g1=raf_gate(1),g0b=raf_gate(0);
  _T(g0==g0b,"B1:raf_gate_deterministic");
  _T(g0<8u&&g1<8u,"B1:raf_gate_range");

  /* B2: câmaras */
  _T(raf_cam(42u,0u)<6u,"B2:raf_cam_range");

  /* B3: vocab */
  Vocab *tv=vocab_new();
  _T(tv!=(void*)0,"B3:vocab_new_ok");
  u32 id1=vocab_ins(tv,"verbo"),id2=vocab_ins(tv,"verbo");
  _T(id1==id2,"B3:vocab_dedup");
  u32 id3=vocab_get(tv,"verbo");
  _T(id3==id1,"B3:vocab_lookup");

  /* B4: tokenizer */
  static u32 tids[64];
  u32 nt=tokenize(tv,"deus e amor",11u,tids,64u);
  _T(nt==3u,"B4:tokenize_3words");

  /* B5: modelo novo */
  u32 gsave=G_ABUMP; /* sem arena_save disponível aqui, usa G_ABUMP direto */
  RafModel *tm=model_new(tv->sz);
  _T(tm!=(void*)0,"B5:model_new_ok");
  _T(tm->vocab_size==tv->sz,"B5:model_vocab_size");

  /* B6: model_step converge */
  static f32 lg[VOCAB_MAX];
  f32 l0=model_step_raf(tm,tids[0],tids[1],lg,0.1f,0u);
  f32 l1=model_step_raf(tm,tids[0],tids[1],lg,0.1f,1u);
  _T(l0>0.f,"B6:loss_positive");
  _T(l1<=l0+0.5f,"B6:loss_decreasing"); /* pode aumentar ligeiramente */

  /* B7: câmara EMA atualiza */
  u32 cam=raf_cam(tids[0],0u);
  f32 before=tm->cam_center[cam][0];
  raf_cam_update(tm,cam,tids[0]);
  _T(1u,"B7:cam_update_runs");

  /* B8: gate sequência não constante */
  u32 same=1;
  for(u32 i=1;i<16u;i++) if(raf_gate(i)!=raf_gate(0)){same=0;break;}
  _T(!same,"B8:gate_not_constant");

  /* B9: T7 Q16 spiral */
  u32 q=((u32)56756u*(u32)56756u)>>16u; /* √3/2 × √3/2 = 3/4 */
  _T(q>48000u&&q<50000u,"B9:q16_spiral_mul");

  _w("\n  pass="); _pd(G_PASS);
  _w("  fail="); _pd(G_FAIL);
  if(G_FAIL==0u) _w("  ★ TODOS PASSARAM\n");
  else            _w("  ✗ FALHAS\n");
}
