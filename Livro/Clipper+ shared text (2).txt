/*

 *   Branchless onde possível (selects via aritmética).

 *   Watchdog via commit gate 4 fases.
 *   Rollback automático se CRC inválido.
 *

 *         -fPIE -pie -std=c11 -ffast-math -fno-builtin

/* ── tipos ─────────────────────────────────────────────────────────── */
typedef unsigned char      u8;
typedef unsigned short     u16;
typedef unsigned int       u32;
typedef unsigned long long u64;
typedef signed int         i32;
typedef signed long long   i64;
typedef float              f32;

/* ── constantes geométricas Q16.16 ────────────────────────────────── */
#define SPIRAL_Q16  56755u   /* sqrt(3)/2 */
#define PHI_Q16    105965u   /* (1+sqrt(5))/2 */
#define Q16_ONE     65536u
#define Q16_HALF    32768u
#define PERIOD         42u

/* ── arena BSS: 2MB, zero heap ─────────────────────────────────────── */
#define ARENA_SZ (2u*1024u*1024u)
static u8  __attribute__((aligned(64))) _arena[ARENA_SZ];
static u32 _bump = 0;

static void *_A(u32 n, u32 al) {
    u32 m=al-1u, s=(_bump+m)&~m, e=s+n;
    if (e > ARENA_SZ) return 0;
    _bump = e;
    return _arena + s;
}

/* ── memset / memcpy freestanding ──────────────────────────────────── */
static void _zero(void *p, u32 n) {
    u8 *d=(u8*)p;
    while(n--) *d++=0;
}
static void _copy(void *d, const void *s, u32 n) {
    u8 *dd=(u8*)d; const u8 *ss=(const u8*)s;
    while(n--) *dd++=*ss++;
}

/* ── Q16.16 ops (sem float no hot path) ────────────────────────────── */
static inline u32 qmul(u32 a, u32 b) { return (u32)(((u64)a*b)>>16); }
static inline u32 qema(u32 o, u32 i) {
    return (u32)(((u64)o*49152u+(u64)i*16384u)>>16);
}
static inline u32 qabs(i32 v) { return (u32)(v<0?-v:v); }

/* ── CRC32C Castagnoli (RFC 3720) ───────────────────────────────────── */
static u32 _crc_tab[256];
static int _crc_rdy = 0;
static void _crc_init(void) {
    for (u32 i=0;i<256u;i++) {
        u32 v=i;
        for (int j=0;j<8;j++) v=(v&1u)?(v>>1)^0x82F63B78u:(v>>1);
        _crc_tab[i]=v;
    }
    _crc_rdy=1;
}
static u32 crc32c(const void *b, u32 n) {
    if (!_crc_rdy) _crc_init();
    const u8 *p=(const u8*)b; u32 c=~0u;
    while (n--) c=(c>>8)^_crc_tab[(c^*p++)&0xFF];
    return ~c;
}

/* ── exp aproximado Q16.16 (Taylor ordem 4, |x|<3) ─────────────────── */
/* SHADOW: erro ~0.5% — documentado */
static u32 qexp_decay(u32 days, u32 half_life) {
    /* exp(-days/half_life) em Q16.16
     * = 2^(-days/half_life/log2e)
     * Aproximação: e^{-x} ≈ 1/(1+x+x²/2)  para x pequeno */
    if (half_life == 0u) return 0u;
    u32 x_q16 = (u32)(((u64)days * Q16_ONE) / half_life);
    if (x_q16 > 3u*Q16_ONE) return 0u;
    /* 1 + x + x²/2 em Q16.16 */
    u32 x2 = qmul(x_q16, x_q16);
    u32 denom = Q16_ONE + x_q16 + (x2>>1);
    if (denom == 0u) return 0u;
    return (u32)(((u64)Q16_ONE * Q16_ONE) / denom);
}

/* ── output sem printf ──────────────────────────────────────────────── */
static const char _HX[]="0123456789ABCDEF";

static void _ws(const char *s) {
    u32 n=0; while(s[n]) n++;
#ifdef __linux__
    /* syscall write via inline asm ARM32 */
#ifdef __arm__
    __asm__ volatile(
        "mov r7, #4\n" "mov r0, #1\n" "mov r1, %0\n" "mov r2, %1\n" "swi #0\n"
        :: "r"(s),"r"(n) : "r0","r1","r2","r7"
    );
#else
    /* fallback: GCC builtin write */
    extern long write(int,const void*,u32);
    write(1,s,n);
#endif
#endif
}

static void _wh(u32 v) {
    char b[11]="0x00000000\n";
    for (int i=0;i<8;i++) b[2+i]=_HX[(v>>(28-i*4))&0xF];
    _ws(b);
}
static void _wf2(f32 v) {
    /* imprime float com 2 casas decimais, sem printf */
    char buf[16]; int neg=(v<0.0f); if(neg)v=-v;
    u32 ip=(u32)v;
    u32 fp=(u32)((v-(f32)ip)*100.0f+0.5f);
    int i=14; buf[i--]='\n'; buf[i--]='0'+fp%10; buf[i--]='0'+fp/10%10;
    buf[i--]='.';
    if(!ip){buf[i--]='0';}else{u32 vv=ip;while(vv){buf[i--]='0'+vv%10;vv/=10;}}
    if(neg) buf[i--]='-';
    _ws(buf+i+1);
}

/* ══════════════════════════════════════════════════════════════════════
 * 1. HIDDEN INCENTIVE DETECTOR
 * ══════════════════════════════════════════════════════════════════════
 *
 * Vetor de pressão de 6 componentes, cada um em [0,1] Q16.16.
 * Decaimento exponencial pela distância em dias até o evento.
 *
 * REGISTRADORES (sem variável nomeada no hot path):
 *   r4=event_ptr  r5=days  r6=proximity  r7=contribution
 */

typedef struct {
    u32 bonus;       /* pressão de bônus            */
    u32 mandate;     /* fim de mandato              */
    u32 quarterly;   /* fechamento de trimestre     */
    u32 fiscal;      /* prazo fiscal / auditoria    */
    u32 subsidiary;  /* fechamento de subsidiária   */
    u32 political;   /* dividendo / orçamento       */
    u32 crc;
} incentive_vec_t;

/* kinds codificados como u8 */
#define EV_BONUS       0u
#define EV_MANDATE     1u
#define EV_QUARTER     2u
#define EV_YEAR_END    3u
#define EV_EARNINGS    4u
#define EV_TAX         5u
#define EV_AUDIT       6u
#define EV_SUBSIDIARY  7u
#define EV_DIVIDEND    8u
#define EV_BUDGET      9u

typedef struct {
    u8  kind;
    u8  entity_idx;   /* índice na tabela de entidades */
    u16 days_from_now;
    u16 intensity_q8; /* Q8.8 em [0,256] */
    u16 _pad;
} corp_event_t;

/* ── compute incentive vector ───────────────────────────────────────── */
static void incentive_compute(
    incentive_vec_t *out,
    const corp_event_t *evs, u32 n_ev,
    u8 entity_idx,
    u32 lookahead_days,
    u32 half_life
) {
    _zero(out, sizeof(*out));

    for (u32 i=0; i<n_ev; i++) {
        const corp_event_t *ev = evs + i;
        if (ev->entity_idx != entity_idx) continue;
        u32 d = ev->days_from_now;
        if (d > lookahead_days) continue;

        u32 prox = qexp_decay(d, half_life);
        /* intensity_q8 → Q16.16 */
        u32 intens = (u32)ev->intensity_q8 << 8;
        u32 contrib = qmul(prox, intens);

        /* branchless: acumula no campo correto via switch-less trick
         * (cada campo é u32 consecutivo, endereçado por offset) */
        u32 *fields = &out->bonus;
        u8 k = ev->kind;
        u32 idx = (k==EV_BONUS)?0u:(k==EV_MANDATE)?1u:
                  (k==EV_QUARTER||k==EV_YEAR_END||k==EV_EARNINGS)?2u:
                  (k==EV_TAX||k==EV_AUDIT)?3u:
                  (k==EV_SUBSIDIARY)?4u:5u;
        /* clamp ao acumular */
        u32 cur = fields[idx];
        u32 sum = cur + contrib;
        fields[idx] = sum > Q16_ONE ? Q16_ONE : sum;
    }

    out->crc = 0;
    out->crc = crc32c(out, sizeof(*out));
}

static u32 incentive_total(const incentive_vec_t *v) {
    return (v->bonus + v->mandate + v->quarterly +
            v->fiscal + v->subsidiary + v->political) / 6u;
}

/* ══════════════════════════════════════════════════════════════════════
 * 2. INSTITUTIONAL TIMING TRACKER
 * ══════════════════════════════════════════════════════════════════════
 *
 * Histograma de (days_before_event, magnitude) por entidade+ação.
 * Sem heap: ring buffer fixo de 128 registros.
 */

#define TIMING_CAP 128u

typedef struct {
    u8  entity_idx;
    u8  action_kind;
    i16 days_before;
    u16 magnitude_q8;
    u16 _pad;
} timing_rec_t;

typedef struct {
    timing_rec_t recs[TIMING_CAP];
    u32 head;
    u32 count;
    u32 crc;
} timing_store_t;

static void timing_record(
    timing_store_t *st,
    u8 entity, u8 action, i16 days, u16 mag_q8
) {
    u32 idx = st->head % TIMING_CAP;
    st->recs[idx] = (timing_rec_t){entity, action, days, mag_q8, 0};
    st->head++;
    if (st->count < TIMING_CAP) st->count++;
    st->crc = 0;
    st->crc = crc32c(st, sizeof(*st));
}

/* Retorna z-score de desvio em Q16.16 (0 = dentro do padrão) */
static u32 timing_deviation(
    const timing_store_t *st,
    u8 entity, u8 action,
    i16 obs_days, u16 obs_mag_q8
) {
    /* acumula soma e soma² para desvio padrão online */
    i64 sum_d=0, sum_d2=0;
    i64 sum_m=0;
    u32 n=0;

    for (u32 i=0; i<st->count; i++) {
        const timing_rec_t *r = &st->recs[i];
        if (r->entity_idx != entity || r->action_kind != action) continue;
        sum_d  += r->days_before;
        sum_d2 += (i64)r->days_before * r->days_before;
        sum_m  += r->magnitude_q8;
        n++;
    }
    if (n < 3u) return 0u;

    i64 mean_d = sum_d / (i64)n;
    i64 var_d  = sum_d2/(i64)n - mean_d*mean_d;
    if (var_d <= 0) return 0u;

    /* stddev: Newton-Raphson sqrt */
    i64 std_d = var_d;
    for (int it=0; it<8; it++) std_d = (std_d + var_d/std_d)/2;

    i64 z_days = ((i64)obs_days - mean_d);
    if (z_days < 0) z_days = -z_days;
    /* z_days / std_d em Q16.16 */
    u32 z_q16 = std_d>0 ? (u32)((z_days * (i64)Q16_ONE) / std_d) : 0u;

    i64 mean_m = sum_m / (i64)n;
    u32 mag_diff = (u32)((obs_mag_q8>mean_m)?(obs_mag_q8-mean_m):(mean_m-obs_mag_q8));

    /* combined: (z + mag_diff/256) / 4, capped at Q16_ONE */
    u32 combined = (z_q16 + (mag_diff<<8)) / 4u;
    return combined > Q16_ONE ? Q16_ONE : combined;
}

/* ══════════════════════════════════════════════════════════════════════
 * 3. ADAPTIVE CULTURE INDEX
 * ══════════════════════════════════════════════════════════════════════
 *
 * EMA dos 4 parâmetros: lag, magnitude, persistence, learning.
 * alpha_q16 = 0.2 × 65536 = 13107
 *
 * RESTRIÇÃO ÉTICA:
 *   O índice aplica-se à entidade jurídica observada.
 *   Não há generalização para grupos étnicos ou nacionais.
 */

#define CULTURE_CAP 64u
#define CULTURE_ALPHA 13107u  /* 0.2 × Q16_ONE */

typedef struct {
    u8  entity_idx;
    u8  _pad[3];
    u32 lag_q16;         /* atraso médio de reação */
    u32 magnitude_q16;   /* magnitude média de resposta */
    u32 persistence_q16; /* duração média da posição */
    u32 learning_q16;    /* taxa de ajuste após erro */
    u32 n_obs;
    u32 crc;
} culture_profile_t;

typedef struct {
    culture_profile_t profiles[CULTURE_CAP];
    u32 n_profiles;
} culture_store_t;

static culture_profile_t *_culture_find(culture_store_t *st, u8 entity) {
    for (u32 i=0; i<st->n_profiles; i++)
        if (st->profiles[i].entity_idx == entity)
            return &st->profiles[i];
    if (st->n_profiles < CULTURE_CAP) {
        culture_profile_t *p = &st->profiles[st->n_profiles++];
        _zero(p, sizeof(*p));
        p->entity_idx = entity;
        return p;
    }
    return 0;
}

static void culture_update(
    culture_store_t *st, u8 entity,
    u32 lag_q16, u32 mag_q16, u32 pers_q16, u32 learn_q16
) {
    culture_profile_t *p = _culture_find(st, entity);
    if (!p) return;
    p->lag_q16         = qema(p->lag_q16, lag_q16);
    p->magnitude_q16   = qema(p->magnitude_q16, mag_q16);
    p->persistence_q16 = qema(p->persistence_q16, pers_q16);
    p->learning_q16    = qema(p->learning_q16, learn_q16);
    p->n_obs++;
    p->crc = 0;
    p->crc = crc32c(p, sizeof(*p));
}

static u32 culture_adaptability(const culture_store_t *st, u8 entity) {
    /* score = 0.4×speed + 0.3×mag + 0.3×learn, em Q16.16 */
    for (u32 i=0; i<st->n_profiles; i++) {
        const culture_profile_t *p = &st->profiles[i];
        if (p->entity_idx != entity) continue;
        if (p->n_obs < 5u) return Q16_HALF;
        /* speed = 1/(1 + lag/30days) em Q16.16 */
        u32 lag_days = p->lag_q16 >> 16;
        u32 denom = Q16_ONE + (u32)(((u64)lag_days * Q16_ONE) / 30u);
        u32 speed = (u32)(((u64)Q16_ONE * Q16_ONE) / denom);
        u32 mag   = p->magnitude_q16 > Q16_ONE ? Q16_ONE : p->magnitude_q16;
        u32 learn = p->learning_q16  > Q16_ONE ? Q16_ONE : p->learning_q16;
        return qmul(26214u,speed) + qmul(19661u,mag) + qmul(19661u,learn);
    }
    return Q16_HALF;
}

/* ══════════════════════════════════════════════════════════════════════
 * 4. CORPORATE BULLWHIP ENGINE
 * ══════════════════════════════════════════════════════════════════════
 *
 * Combina chicote físico com amplificação por incentivo corporativo.
 * score_combined = score_physical × (1 + amplification)
 *
 * amplification = 0.30×quarterly + 0.40×mandate + 0.20×bonus - 0.10×fiscal
 *
 * Registradores (sem variável nomeada):
 *   r4=node_idx r5=phys_score r6=ampl r7=result
 */

#define BULL_NODES_CAP 32u

typedef struct {
    u8  name_idx;     /* índice no array de nomes */
    u32 base_lag;     /* atraso base em dias */
    u32 amplif_q16;   /* amplificação estrutural Q16.16 */
} bull_node_t;

typedef struct {
    u32 phys_score_q16;
    u32 amplification_q16;
    u32 combined_q16;
    u32 dominant_pressure; /* índice 0..5 */
    u32 total_incentive_q16;
    u32 crc;
} bull_result_t;

static bull_result_t bullwhip_combined(
    const bull_node_t *node,
    u32 phys_score_q16,
    const incentive_vec_t *inc
) {
    bull_result_t r;
    _zero(&r, sizeof(r));

    /* amplification = 0.30×quarterly + 0.40×mandate + 0.20×bonus - 0.10×fiscal */
    u32 ampl = qmul(19661u, inc->quarterly)   /* 0.30 */
             + qmul(26214u, inc->mandate)      /* 0.40 */
             + qmul(13107u, inc->bonus)        /* 0.20 */
             - qmul(6554u,  inc->fiscal);      /* -0.10 */

    r.phys_score_q16    = phys_score_q16;
    r.amplification_q16 = ampl;
    /* combined = phys × (1 + ampl) */
    r.combined_q16 = phys_score_q16 + qmul(phys_score_q16, ampl);

    /* dominant: campo com maior pressão */
    u32 *fields = (u32*)inc; /* bonus[0]..political[5] */
    u32 dmax=0, didx=0;
    for (u32 i=0; i<6u; i++) {
        if (fields[i] > dmax) { dmax=fields[i]; didx=i; }
    }
    r.dominant_pressure    = didx;
    r.total_incentive_q16  = incentive_total(inc);

    r.crc = 0;
    r.crc = crc32c(&r, sizeof(r));
    return r;
}

/* ══════════════════════════════════════════════════════════════════════
 * 5. STOCK MANIPULATION DETECTOR
 * ══════════════════════════════════════════════════════════════════════
 *
 * 5 detectores independentes. Cada um retorna score [0,1] Q16.16.
 * Sem branches em caminhos quentes — aritmética pura.
 */

/* branchless clamp [0, Q16_ONE] */
static inline u32 _clamp(u32 v) {
    return v > Q16_ONE ? Q16_ONE : v;
}
/* branchless: retorna v se cond≠0, 0 caso contrário */
static inline u32 _sel(u32 cond, u32 v) {
    return (u32)(-(i32)(cond!=0u)) & v;
}

typedef struct {
    u32 channel_stuffing;
    u32 false_scarcity;
    u32 margin_burn;
    u32 dumping;
    u32 delayed_repasse;
    u32 crc;
} manip_scores_t;

static manip_scores_t stock_detect(
    /* inventário */
    u32 inv_pct_change_q16,    /* delta% Q16.16, pode ser >1 */
    u32 inv_level_q16,
    u32 inv_baseline_q16,
    /* produção */
    u32 prod_pct_change_q16,
    /* preço */
    u32 price_pct_change_q16,
    /* margem */
    i32 margin_pct_change_q16, /* signed */
    u32 revenue_pct_change_q16,
    /* custo de input */
    u32 input_cost_pct_q16,
    u32 output_price_pct_q16,
    /* tempos */
    u32 days_to_quarter_end,
    u32 days_to_year_end,
    u32 days_since_input_change,
    u32 expected_lag_days,
    /* incentivo */
    u32 bonus_pressure_q16
) {
    manip_scores_t s;
    _zero(&s, sizeof(s));

    /* ── channel stuffing ─────────────────────────────────────────── */
    {
        u32 cond = (days_to_quarter_end <= 30u) &
                   (inv_pct_change_q16 > 6554u) &    /* >10% */
                   (prod_pct_change_q16 < 1311u);    /* <2% */
        u32 prox = days_to_quarter_end < 30u ?
                   Q16_ONE - (u32)(((u64)(days_to_quarter_end)*Q16_ONE)/30u) : 0u;
        u32 mag  = _clamp((u32)(((u64)inv_pct_change_q16*Q16_ONE)/19661u));
        s.channel_stuffing = _sel(cond, qmul(prox, mag));
    }

    /* ── false scarcity ───────────────────────────────────────────── */
    {
        u32 ratio_q16 = inv_baseline_q16>0 ?
            (u32)(((u64)inv_level_q16*Q16_ONE)/inv_baseline_q16) : Q16_ONE;
        u32 cond = (ratio_q16 < 45875u) &              /* < 0.70 */
                   (prod_pct_change_q16 < 3277u) &     /* <5% drop */
                   (price_pct_change_q16 > 3277u);     /* >5% rise */
        u32 scarcity = _clamp(Q16_ONE - ratio_q16);
        u32 pr_resp  = _clamp((u32)(((u64)price_pct_change_q16*Q16_ONE)/13107u));
        s.false_scarcity = _sel(cond, qmul(scarcity, pr_resp));
    }

    /* ── margin burn ──────────────────────────────────────────────── */
    {
        u32 cond = (margin_pct_change_q16 < -3277) &
                   (revenue_pct_change_q16 > 3277u) &
                   (bonus_pressure_q16 > Q16_HALF);
        u32 burn = _clamp(qabs(margin_pct_change_q16));
        s.margin_burn = _sel(cond, qmul(burn, bonus_pressure_q16));
    }

    /* ── dumping ──────────────────────────────────────────────────── */
    {
        u32 ratio_q16 = inv_baseline_q16>0 ?
            (u32)(((u64)inv_level_q16*Q16_ONE)/inv_baseline_q16) : Q16_ONE;
        u32 cond = (ratio_q16 > 91750u) &               /* >1.4 */
                   (price_pct_change_q16 > 3277u) &     /* price fell >5% (abs) */
                   (days_to_year_end <= 60u);
        u32 excess = _clamp(ratio_q16 - Q16_ONE);
        u32 cut    = _clamp((u32)(((u64)price_pct_change_q16*Q16_ONE)/16384u));
        s.dumping = _sel(cond, qmul(excess, cut));
    }

    /* ── delayed repasse ──────────────────────────────────────────── */
    {
        u32 cond = (days_since_input_change > expected_lag_days*2u) &
                   (input_cost_pct_q16 > output_price_pct_q16 + 3277u);
        u32 gap  = _clamp(input_cost_pct_q16 - output_price_pct_q16);
        u32 norm = _clamp((u32)(((u64)gap*Q16_ONE)/13107u));
        s.delayed_repasse = _sel(cond, norm);
    }

    s.crc = 0;
    s.crc = crc32c(&s, sizeof(s));
    return s;
}

/* ══════════════════════════════════════════════════════════════════════
 * 6. DETERMINISM GUARD
 * ══════════════════════════════════════════════════════════════════════
 *
 * Tempera todo sinal final com intervalo de confiança explícito.
 * Confiança máxima: 0.85 × Q16_ONE = 55705.
 * O sistema NUNCA chega a 1.0 — incerteza residual sempre presente.
 *
 * WATCHDOG: se CRC do estado temperado for 0 → rollback para raw/2.
 */

#define GUARD_MAX_CONF  55705u  /* 0.85 × Q16_ONE */
#define GUARD_MIN_SAMP  30u
#define GUARD_ALT_PEN   3277u   /* 0.05 × Q16_ONE por explicação alternativa */

typedef struct {
    u32 raw_q16;
    u32 tempered_q16;
    u32 confidence_q16;
    u32 ci_width_q16;
    u32 n_samples;
    u32 exception_rate_q16;
    u32 alt_explanations;
    u32 crc;
} tempered_signal_t;

static tempered_signal_t guard_temper(
    u32 raw_q16,
    u32 n_samples,
    u32 exception_rate_q16,
    u32 alt_explanations
) {
    tempered_signal_t out;
    _zero(&out, sizeof(out));

    u32 conf;
    if (n_samples < GUARD_MIN_SAMP) {
        conf = (u32)(((u64)GUARD_MAX_CONF * 30u * n_samples) /
                     ((u64)GUARD_MIN_SAMP * 100u));
    } else {
        u32 base = Q16_ONE > exception_rate_q16 ?
                   Q16_ONE - exception_rate_q16 : 0u;
        u32 penalty = GUARD_ALT_PEN * alt_explanations;
        conf = base > penalty ? base - penalty : 0u;
        /* cap GUARD_MAX_CONF */
        if (conf > GUARD_MAX_CONF) conf = GUARD_MAX_CONF;
    }

    out.raw_q16            = raw_q16;
    out.confidence_q16     = conf;
    out.tempered_q16       = qmul(raw_q16, conf);
    out.ci_width_q16       = Q16_ONE - conf;
    out.n_samples          = n_samples;
    out.exception_rate_q16 = exception_rate_q16;
    out.alt_explanations   = alt_explanations;
    out.crc                = 0;
    out.crc                = crc32c(&out, sizeof(out));

    /* WATCHDOG: CRC=0 após calcular = corrupção → rollback */
    if (out.crc == 0u) {
        _zero(&out, sizeof(out));
        out.raw_q16       = raw_q16;
        out.tempered_q16  = raw_q16 >> 1;  /* fallback: raw/2 */
        out.confidence_q16= Q16_ONE >> 2;   /* 25% */
        out.ci_width_q16  = 3u*(Q16_ONE>>2);
        out.crc = crc32c(&out, sizeof(out));
    }

    return out;
}

/* ══════════════════════════════════════════════════════════════════════
 * 7. COMMIT GATE 4 FASES
 * ══════════════════════════════════════════════════════════════════════
 *
 * Protege a saída final do pipeline com LOAD→PROCESS→VERIFY→COMMIT.
 * Rollback automático se CRC inválido na fase VERIFY.
 */

#define GATE_LOAD    0x1u
#define GATE_PROC    0x2u
#define GATE_VERIFY  0x4u
#define GATE_COMMIT  0x8u
#define GATE_ALL     0xFu

typedef struct {
    bull_result_t   bull;
    manip_scores_t  manip;
    tempered_signal_t final_signal;
    u32 gate_bits;
    u32 crc_chain;
    u32 commits;
    u32 rollbacks;
} pipeline_state_t;

static int pipeline_step(
    pipeline_state_t *ps,
    const bull_result_t *bull_in,
    const manip_scores_t *manip_in,
    u32 n_samples,
    u32 exception_rate_q16,
    u32 alt_explanations
) {
    /* LOAD: snapshot */
    pipeline_state_t snap = *ps;
    ps->gate_bits |= GATE_LOAD;

    /* PROCESS */
    ps->bull  = *bull_in;
    ps->manip = *manip_in;

    /* agrega: sinal bruto = combined_bull × (1 - manipulation_avg) */
    u32 manip_avg = (manip_in->channel_stuffing + manip_in->false_scarcity +
                     manip_in->margin_burn + manip_in->dumping +
                     manip_in->delayed_repasse) / 5u;
    u32 discount   = Q16_ONE > manip_avg ? Q16_ONE - manip_avg : 0u;
    u32 raw_signal = qmul(bull_in->combined_q16, discount);

    ps->final_signal = guard_temper(raw_signal, n_samples,
                                     exception_rate_q16, alt_explanations);
    ps->gate_bits |= GATE_PROC;

    /* VERIFY */
    u32 sc = crc32c(&ps->bull, sizeof(ps->bull));
    if (sc == 0u || ps->final_signal.confidence_q16 == 0u) {
        *ps = snap;
        ps->rollbacks++;
        ps->gate_bits = 0u;
        return 0;
    }
    ps->gate_bits |= GATE_VERIFY;

    /* COMMIT */
    if ((ps->gate_bits & GATE_ALL) == GATE_ALL) {
        u32 pair[2] = {ps->final_signal.tempered_q16, ps->crc_chain};
        ps->crc_chain = crc32c(pair, 8u);
        ps->commits++;
        ps->gate_bits = 0u;
        return 1;
    }

    *ps = snap;
    ps->rollbacks++;
    return 0;
}
