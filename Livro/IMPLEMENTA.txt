#pragma once
/* SHA3-256 (Keccak-f[1600]) — implementação do zero, sem deps
 * Constantes e algoritmo: FIPS 202 / domínio público
 * Rate=1088 bits(136B), Capacity=512 bits, Output=256 bits */
typedef unsigned long long u64;
typedef unsigned char      u8;
typedef unsigned int       u32;
/* Constantes de round (FIPS 202 Table 1) */
static const u64 _K_RC[24]={
    0x0000000000000001ULL,0x0000000000008082ULL,0x800000000000808AULL,
    0x8000000080008000ULL,0x000000000000808BULL,0x0000000080000001ULL,
    0x8000000080008081ULL,0x8000000000008009ULL,0x000000000000008AULL,
    0x0000000000000088ULL,0x0000000080008009ULL,0x000000008000000AULL,
    0x000000008000808BULL,0x800000000000008BULL,0x8000000000008089ULL,
    0x8000000000008003ULL,0x8000000000008002ULL,0x8000000000000080ULL,
    0x000000000000800AULL,0x800000008000000AULL,0x8000000080008081ULL,
    0x8000000000008080ULL,0x0000000080000001ULL,0x8000000080008008ULL
};
/* Offsets de rotação rho (FIPS 202 Table 2) */
static const int _K_RO[25]={
     0, 1,62,28,27,36,44, 6,55,20, 3,10,43,25,39,
    41,45,15,21, 8,18, 2,61,56,14
};
/* Permutação pi */
static const int _K_PI[25]={
     0,10,20, 5,15,16, 1,11,21, 6,7,17, 2,12,22,
    23, 8,18, 3,13,14,24, 9,19, 4
};
#define ROL64(x,n) (((x)<<(n))|((x)>>(64-(n))))
static void keccak_f(u64 st[25]){
    u64 t[25],C[5],D[5];
    int r,x,y,i;
    for(r=0;r<24;r++){
        /* theta */
        for(x=0;x<5;x++) C[x]=st[x]^st[x+5]^st[x+10]^st[x+15]^st[x+20];
        for(x=0;x<5;x++){D[x]=C[(x+4)%5]^ROL64(C[(x+1)%5],1);}
        for(i=0;i<25;i++) st[i]^=D[i%5];
        /* rho+pi */
        for(i=0;i<25;i++) t[_K_PI[i]]=ROL64(st[i],_K_RO[i]);
        /* chi */
        for(y=0;y<5;y++) for(x=0;x<5;x++)
            st[y*5+x]=t[y*5+x]^((~t[y*5+(x+1)%5])&t[y*5+(x+2)%5]);
        /* iota */
        st[0]^=_K_RC[r];
    }
}
#define SHA3_RATE 136 /* 1088/8 bytes */
typedef struct{ u64 st[25]; u8 buf[SHA3_RATE]; u32 pos; } SHA3_ctx;
static void sha3_init(SHA3_ctx*c){
    int i; for(i=0;i<25;i++) c->st[i]=0;
    c->pos=0;
}
static void sha3_update(SHA3_ctx*c,const u8*in,u32 n){
    u32 i;
    for(i=0;i<n;i++){
        c->buf[c->pos++]^=(u8)in[i]; /* XOR byte na posição */
        /* Quando rate completo: absorve */
        if(c->pos==SHA3_RATE){
            u32 j; for(j=0;j<SHA3_RATE;j++) ((u8*)c->st)[j]^=c->buf[j];
            keccak_f(c->st);
            for(j=0;j<SHA3_RATE;j++) c->buf[j]=0;
            c->pos=0;
        }
    }
}
static void sha3_final(SHA3_ctx*c,u8 out[32]){
    /* Padding SHA3: 0x06 na posição atual, 0x80 no último byte do rate */
    u32 i;
    u8 tmp[SHA3_RATE]; for(i=0;i<SHA3_RATE;i++) tmp[i]=0;
    tmp[c->pos]=0x06; tmp[SHA3_RATE-1]|=0x80;
    for(i=0;i<SHA3_RATE;i++) ((u8*)c->st)[i]^=tmp[i];
    keccak_f(c->st);
    for(i=0;i<32;i++) out[i]=((u8*)c->st)[i];
}
/* API única: sha3_256(data, n, out[32]) */
static void sha3_256(const u8*data,u32 n,u8 out[32]){
    SHA3_ctx c; sha3_init(&c); sha3_update(&c,data,n); sha3_final(&c,out);
}
/* Converte digest 32 bytes → string hex 64 chars + null */
static void sha3_hex(const u8 d[32],char hex[65]){
    static const char h[]="0123456789abcdef";
    u32 i; for(i=0;i<32;i++){hex[i*2]=h[d[i]>>4];hex[i*2+1]=h[d[i]&0xF];}
    hex[64]=0;
}

#pragma once
/* Entropia de Shannon H(X) = -Σ p_i * log2(p_i)
 * Implementação sem libm: log2 via bit trick + refinamento Newton-Raphson
 * Resultado em Q16.16 (bits × 65536) */
typedef unsigned int  _sh_u32;
typedef unsigned char _sh_u8;
/* log2 inteiro: posição do bit mais significativo */
static __attribute__((always_inline)) inline
_sh_u32 _sh_ilog2(_sh_u32 v){
    _sh_u32 r=0;
    if(v>=0x10000){r+=16;v>>=16;}
    if(v>=0x100  ){r+= 8;v>>= 8;}
    if(v>=0x10   ){r+= 4;v>>= 4;}
    if(v>=0x4    ){r+= 2;v>>= 2;}
    if(v>=0x2    ){r+= 1;}
    return r;
}
/* log2(v) em Q16.16 via bit trick + 4 bits de fração por lookup */
static _sh_u32 _sh_log2_q16(_sh_u32 v){
    if(!v) return 0;
    _sh_u32 i=_sh_ilog2(v);   /* parte inteira */
    _sh_u32 f=0;
    if(i>=1){
        /* fração: normaliza v para [1,2) multiplicando por 2^(16-i) */
        _sh_u32 norm;
        if(i<=16) norm=(v<<(16-i));
        else       norm=(v>>(i-16));
        /* norm está em Q16.16 no intervalo [65536, 131072) */
        /* Aproximação linear da fração de log2: f ≈ (norm-65536)/65536 */
        /* Refinamento: usa série Taylor log2(1+x) ≈ x/ln2 - x²/(2ln2) */
        _sh_u32 x = norm - 65536u; /* x em [0, 65536) */
        /* ln2 ≈ 45426 em Q16.16 (0.6931...*65536) */
        f = ((_sh_u32)(((unsigned long long)x * 94548u) >> 16u)); /* x/ln2 Q16 */
        _sh_u32 x2 = (_sh_u32)(((unsigned long long)x*x)>>16u);
        _sh_u32 corr=(_sh_u32)(((unsigned long long)x2*47274u)>>16u);
        if(f>corr) f-=corr; else f=0;
    }
    return (i<<16u) | (f&0xFFFFu);
}
/* Entropia de Shannon em Q16.16 bits
 * freq[256]: frequência absoluta de cada símbolo (contagem)
 * n: total de símbolos (soma de freq)
 * Retorna H(X)*65536 (bits × Q16) */
static _sh_u32 shannon_entropy_q16(const _sh_u32 freq[256], _sh_u32 n){
    if(!n) return 0;
    unsigned long long H=0;
    _sh_u32 i;
    for(i=0;i<256u;i++){
        if(!freq[i]) continue;
        /* p_i = freq[i]/n em Q32: p_i_q32 = (freq[i]<<16)/n */
        _sh_u32 p_q16=(_sh_u32)(((unsigned long long)freq[i]<<16u)/n);
        if(!p_q16) continue;
        _sh_u32 log2p=_sh_log2_q16(p_q16); /* log2(p) em Q16 — será negativo: */
        /* Como p<1, log2(p)<0. Calculamos log2(n/freq[i]) = log2(n)-log2(freq[i]) */
        _sh_u32 log2n=_sh_log2_q16(n);
        _sh_u32 log2f=_sh_log2_q16(freq[i]);
        _sh_u32 neg_log2p=(log2n>log2f)?(log2n-log2f):0;
        /* H += p_i * log2(1/p_i) = freq[i]/n * (log2n - log2freq[i]) */
        H += ((unsigned long long)freq[i] * neg_log2p) / n;
    }
    return (_sh_u32)(H & 0xFFFFFFFFu);
}
/* Calcula frequências e entropia de um buffer em uma chamada */
static _sh_u32 shannon_buf_q16(const _sh_u8*buf, _sh_u32 n){
    _sh_u32 freq[256]={0}, i;
    for(i=0;i<n;i++) freq[buf[i]]++;
    return shannon_entropy_q16(freq,n);
}
/* Normaliza para [0..1] Q16: H_max = log2(256)=8 bits */
static _sh_u32 shannon_normalized_q16(const _sh_u8*buf,_sh_u32 n){
    _sh_u32 h=shannon_buf_q16(buf,n);
    /* Divide por 8 (log2(256)) = shift right 3 + ajuste */
    return h>>3u; /* H/8 em Q16 → [0, 65536] */
}

#pragma once
/* raf_replay.h — Replay: re-executa operações a partir de um log imutável
 * O rollback volta ao estado anterior. O replay re-aplica operações sobre
 * qualquer estado base, permitindo auditoria, fork e convergência.
 * Cada entrada do log: operação + payload + CRC32C para integridade */
typedef unsigned int  _rp_u32;
typedef unsigned char _rp_u8;
typedef unsigned long long _rp_u64;

#define RP_MAX_ENTRIES 1024u
#define RP_PAYLOAD_SZ  64u
#define RP_OP_TORUS    0x01u /* Passo toroidal */
#define RP_OP_EMA      0x02u /* Atualização EMA */
#define RP_OP_COMMIT   0x03u /* Commit gate */
#define RP_OP_MARK     0x04u /* Checkpoint de arena */
#define RP_OP_ROLLBACK 0x05u /* Rollback executado */
#define RP_OP_CRC      0x06u /* Verificação CRC */
#define RP_OP_GENESIS  0x10u /* Genesis block */

typedef struct __attribute__((packed)) {
    _rp_u32 seq;                  /* número sequencial (monotônico) */
    _rp_u32 op;                   /* código da operação */
    _rp_u64 t_ns;                 /* timestamp nanosegundos */
    _rp_u8  payload[RP_PAYLOAD_SZ]; /* dados da operação */
    _rp_u32 payload_len;          /* bytes válidos em payload */
    _rp_u32 crc;                  /* CRC32C de todos os campos acima */
    _rp_u32 prev_crc;             /* CRC da entrada anterior (chain) */
} ReplayEntry;

typedef struct {
    ReplayEntry entries[RP_MAX_ENTRIES];
    _rp_u32 count;      /* entradas gravadas */
    _rp_u32 play_pos;   /* posição de reprodução atual */
    _rp_u32 genesis_crc;/* CRC do genesis block */
    _rp_u32 last_crc;   /* CRC da última entrada */
    _rp_u32 n_corrupted;/* entradas com CRC inválido */
} ReplayLog;

static ReplayLog G_REPLAY;

/* CRC32C inline (poly Castagnoli) */
static __attribute__((always_inline)) inline
_rp_u32 _rp_crc(const _rp_u8*b,_rp_u32 n){
    _rp_u32 c=~0u;
    while(n--){c^=*b++;for(int j=0;j<8;j++)c=(c>>1)^(0x82F63B78u&-(_rp_u32)(c&1u));}
    return~c;
}

/* Grava uma entrada no log */
static int rp_record(ReplayLog*L,_rp_u32 op,const _rp_u8*payload,_rp_u32 plen,_rp_u64 t_ns){
    if(L->count>=RP_MAX_ENTRIES) return -1;
    ReplayEntry*e=&L->entries[L->count];
    e->seq=L->count;
    e->op=op;
    e->t_ns=t_ns;
    e->payload_len=(plen>RP_PAYLOAD_SZ?RP_PAYLOAD_SZ:plen);
    _rp_u32 i; for(i=0;i<RP_PAYLOAD_SZ;i++) e->payload[i]=0;
    for(i=0;i<e->payload_len;i++) e->payload[i]=payload[i];
    e->prev_crc=L->last_crc;
    e->crc=0;
    e->crc=_rp_crc((_rp_u8*)e,((_rp_u32)sizeof(ReplayEntry)-sizeof(_rp_u32)));
    L->last_crc=e->crc;
    if(L->count==0) L->genesis_crc=e->crc;
    L->count++;
    return 0;
}

/* Verifica integridade completa do log */
static _rp_u32 rp_verify(ReplayLog*L){
    _rp_u32 bad=0,i;
    _rp_u32 prev=0;
    for(i=0;i<L->count;i++){
        ReplayEntry*e=&L->entries[i];
        _rp_u32 saved=e->crc; e->crc=0;
        _rp_u32 c=_rp_crc((_rp_u8*)e,((_rp_u32)sizeof(ReplayEntry)-sizeof(_rp_u32)));
        e->crc=saved;
        if(c!=saved||e->prev_crc!=prev) bad++;
        prev=saved;
    }
    L->n_corrupted=bad;
    return bad;
}

/* Replay: chama callback para cada operação a partir de from_seq */
typedef void (*rp_handler)(const ReplayEntry*,void*ctx);
static _rp_u32 rp_replay(ReplayLog*L,_rp_u32 from_seq,rp_handler fn,void*ctx){
    _rp_u32 applied=0,i;
    for(i=0;i<L->count;i++){
        if(L->entries[i].seq<from_seq) continue;
        fn(&L->entries[i],ctx);
        applied++;
    }
    return applied;
}

/* Valida genesis: entry[0] deve existir e ter CRC correto */
static int rp_validate_genesis(const ReplayLog*L){
    if(!L->count) return -1;
    const ReplayEntry*e=&L->entries[0];
    if(e->op!=RP_OP_GENESIS) return -2;
    if(e->seq!=0) return -3;
    if(e->prev_crc!=0) return -4; /* genesis não tem predecessor */
    ReplayEntry tmp=*e; tmp.crc=0;
    _rp_u32 c=_rp_crc((_rp_u8*)&tmp,((_rp_u32)sizeof(ReplayEntry)-sizeof(_rp_u32)));
    return (c==e->crc)?0:-5;
}

/* Inicializa log com genesis block */
static void rp_init(ReplayLog*L,const _rp_u8*seed,_rp_u32 slen,_rp_u64 t_ns){
    _rp_u32 i; for(i=0;i<RP_MAX_ENTRIES;i++){
        _rp_u8*b=(_rp_u8*)&L->entries[i];
        _rp_u32 sz=(_rp_u32)sizeof(ReplayEntry);
        while(sz--) *b++=0;
    }
    L->count=0; L->play_pos=0; L->genesis_crc=0;
    L->last_crc=0; L->n_corrupted=0;
    rp_record(L,RP_OP_GENESIS,seed,slen,t_ns);
}

#pragma once
/* raf_genesis.h — Genesis block: ancora de confiança da hashchain
 * O genesis é o único bloco sem predecessor verificável.
 * Sua integridade depende de: conteúdo fixo + SHA3-256 + timestamp.
 * Qualquer hashchain válida DEVE começar com um genesis verificado. */
#include "raf_sha3.h"
#include "raf_replay.h"
typedef unsigned int  _gn_u32;
typedef unsigned char _gn_u8;
typedef unsigned long long _gn_u64;

/* Conteúdo canônico do genesis RAFAELIA */
static const char _GN_CONTENT[] =
    "RAFAELIA-GENESIS:v1.0:Omega=Amor:DeltaRafaelVerboOmega:"
    "bitraf64=ADELTABOMEGADELTATTPHIIIBOMEGADELTASIGMASIGMA:"
    "F*=23.158:D_H=1.347:n_c=7:period=42";

typedef struct {
    _gn_u8  sha3[32];       /* SHA3-256 do conteúdo canônico */
    _gn_u8  sha3_hex[65];   /* versão hex para display */
    _gn_u64 timestamp_ns;   /* quando foi criado */
    _gn_u32 content_len;    /* comprimento do conteúdo */
    _gn_u32 verified;       /* 1 se verificado com sucesso */
} GenesisBlock;

static GenesisBlock G_GENESIS;

static void genesis_create(_gn_u64 t_ns){
    G_GENESIS.timestamp_ns=(t_ns?t_ns:0ULL);
    G_GENESIS.content_len=(_gn_u32)sizeof(_GN_CONTENT)-1u;
    sha3_256((_gn_u8*)_GN_CONTENT,G_GENESIS.content_len,G_GENESIS.sha3);
    sha3_hex(G_GENESIS.sha3,(_gn_u8(*)[65])G_GENESIS.sha3_hex);
    G_GENESIS.verified=1u;
}

/* Verifica: re-calcula SHA3 e compara byte a byte */
static int genesis_verify(void){
    if(!G_GENESIS.verified) return -1;
    _gn_u8 check[32];
    sha3_256((_gn_u8*)_GN_CONTENT,G_GENESIS.content_len,check);
    _gn_u32 i,diff=0;
    for(i=0;i<32u;i++) diff|=(_gn_u32)(G_GENESIS.sha3[i]^check[i]);
    return diff?-2:0;
}

/* Verifica que o primeiro entry do ReplayLog é o genesis correto */
static int genesis_anchor_log(ReplayLog*L){
    int gv=genesis_verify();
    if(gv) return gv;
    int rv=rp_validate_genesis(L);
    if(rv) return rv;
    /* Payload do entry[0] deve conter os primeiros 32 bytes do SHA3 */
    const ReplayEntry*e=&L->entries[0];
    if(e->payload_len<32u) return -10;
    _gn_u32 i,diff=0;
    for(i=0;i<32u;i++) diff|=(_gn_u32)(e->payload[i]^G_GENESIS.sha3[i]);
    return diff?-11:0;
}

# B3: Multicore clone()
cat > "$D/rafaelia_b3.S" << 'F'

@ 4 workers: 1 por Cortex-A53 · CRC32C de 64KB por worker
@ Syscalls: clone=120 wait4=114 gettimeofday=78
.equ SYS_EXIT,  1; .equ SYS_WRITE,4; .equ SYS_CLONE,120; .equ SYS_WAIT4,114

.section .rodata
.align 6
msg_b3: .ascii "B3: MULTICORE+CRC32SW\n"; .equ msg_b3_len,.-msg_b3
msg_us: .ascii "US="; .equ msg_us_len,.-msg_us

g_crc_rdy3: .word 0
g_tv_start: .word 0,0
g_tv_end:   .word 0,0
g_hex_b3:   .byte 0,0,0,0,0,0,0,0,0x0A,0
@ Buffer de trabalho: preenchido com padrão 0xA5
g_work_buf: .space WORK_BUF_SZ,0xA5
.section .text
.syntax unified
.thumb
.type ws3,%function
ws3: push{lr};mov r2,r1;mov r1,r0;mov r0,#1;mov r7,#SYS_WRITE;swi #0;pop{pc}
.type ph3,%function
ph3:
    push{r4,r5,lr};mov r4,r0;ldr r5,=g_hex_b3;ldr r2,=hex_b3;mov r1,#28
.p3: lsr r0,r4,r1;and r0,r0,#0xF;ldrb r0,[r2,r0];strb r0,[r5],#1;subs r1,r1,#4;bge .p3
    mov r0,#0x0A;strb r0,[r5];ldr r0,=g_hex_b3;mov r1,#9;bl ws3;pop{r4,r5,pc}
.type crc_init3,%function
crc_init3:
    push{r4,r5,r6,lr};ldr r5,=g_crc_tab3;mov r4,#0
.ci3: cmp r4,#256;bge .ci3d
    mov r5,r4;mov r6,#8
.ci3i: lsr r0,r5,#1;and r1,r5,#1;rsb r1,r1,#0;ldr r2,=CRC32_POLY;and r2,r2,r1;eor r5,r0,r2;subs r6,r6,#1;bne .ci3i
    ldr r0,=g_crc_tab3;str r5,[r0,r4,lsl #2];add r4,r4,#1;b .ci3
.ci3d: ldr r0,=g_crc_rdy3;mov r1,#1;str r1,[r0];pop{r4,r5,r6,pc}
@ worker_fn: r0=worker_id. Calcula CRC do g_work_buf e armazena em g_crc_results[id]
.type worker_fn,%function
worker_fn:
    push{r4,r5,r6,r7,lr}
    mov r7,r0          @ worker id
    ldr r4,=g_crc_tab3; mvn r5,#0
    ldr r6,=g_work_buf; mov r2,#WORK_BUF_SZ
.wl: cmp r2,#0;beq .wd
    ldrb r0,[r6],#1;eor r0,r0,r5;and r0,r0,#0xFF
    ldr r0,[r4,r0,lsl #2];lsr r5,r5,#8;eor r5,r5,r0;sub r2,r2,#1;b .wl
.wd: mvn r0,r5
    ldr r4,=g_crc_results;str r0,[r4,r7,lsl #2]
    mov r7,#SYS_EXIT;mov r0,#0;swi #0   @ worker exit
    pop{r4,r5,r6,r7,pc}
.align 2
.global _start
.thumb_func
_start:
    ldr r0,=msg_b3;mov r1,#msg_b3_len;bl ws3
    bl crc_init3
    @ Aloca 4 stacks de worker via mmap2
    mov r8,#0  @ worker counter
.spawn_loop:
    cmp r8,#N_WORKERS;bge .wait_loop
    @ mmap2: 1 stack por worker
    mov r0,#0;mov r1,#WORKER_STACK_SZ;mov r2,#PROT_RW
    mov r3,#MAP_ANON;mov r4,#-1;mov r5,#0;mov r7,#SYS_MMAP2;swi #0
    cmn r0,#1;beq .b3_fail
    @ stack cresce para baixo: r0 = topo do stack = base + size
    add r9,r0,#WORKER_STACK_SZ
    @ clone: flags, stack_ptr
    ldr r0,=0xF00  @ CLONE_VM|CLONE_FS|CLONE_FILES|CLONE_SIGHAND
    mov r1,r9      @ stack top
    mov r7,#SYS_CLONE;swi #0
    @ No filho (r0==0): executa worker_fn
    cmp r0,#0;bne .parent
    mov r0,r8;bl worker_fn @ nunca retorna
.parent:
    add r8,r8,#1;b .spawn_loop
.wait_loop:
    @ wait4 para cada worker
    mov r9,#N_WORKERS
.wl2: cmp r9,#0;beq .wl_done
    mov r0,#-1;mov r1,#0;mov r2,#0;mov r3,#0
    mov r7,#SYS_WAIT4;swi #0;sub r9,r9,#1;b .wl2
.wl_done:
    @ XOR de todos os CRCs como invariante distribuída
    ldr r4,=g_crc_results
    ldr r0,[r4];ldr r1,[r4,#4];eor r0,r0,r1
    ldr r1,[r4,#8];eor r0,r0,r1
    ldr r1,[r4,#12];eor r0,r0,r1
    push{r0}
    ldr r0,=msg_crc;mov r1,#msg_crc_len;bl ws3
    pop{r0};bl ph3
    ldr r0,=msg_b3d;mov r1,#msg_b3d_len;bl ws3
    mov r7,#SYS_EXIT;mov r0,#0;swi #0
.b3_fail: mvn r0,#0;mov r7,#SYS_EXIT;swi #0
F
ok "rafaelia_b3.S ($(wc -l < $D/rafaelia_b3.S)L)"

# B4: Senoides + camadas
cat > "$D/rafaelia_b4.S" << 'F'
@ rafaelia_b4.S — SENOIDES + 7 CAMADAS + SOBREPOSIÇÃO
@ sin_q16 via Taylor: x - x³/6 + x⁵/120 · 7 layers freq harmônicas de π/21
@ Pesos com decaimento exp: w_i ≈ (√2/2)^i · NEON vmull.s32 + vshrn.s64
.equ SYS_EXIT,  1; .equ SYS_WRITE, 4
.equ Q16_ONE,   65536; .equ Q16_PI, 205887; .equ Q16_2PI, 411774
.equ Q16_INV6,  10923; .equ Q16_INV120, 546
.equ PERIOD,    42; .equ N_LAYERS, 7
.section .rodata
.align 6
msg_b4: .ascii "B4: SENOIDE+CAMADAS\n"; .equ msg_b4_len,.-msg_b4
msg_lay:.ascii "LAYER="; .equ msg_lay_len,.-msg_lay
msg_ovl:.ascii "OVERLAP="; .equ msg_ovl_len,.-msg_ovl
msg_raf:.ascii "RAFAELIA:COMPLETE\n"; .equ msg_raf_len,.-msg_raf
@ Frequências harmônicas: freq_i = i * PI/21 em Q16
@ PI/21 ≈ 9804 Q16. freq[0]=0(DC), freq[1..6]=9804..58824
layer_freqs: .word 0,9804,19608,29412,39216,49020,58824
@ Pesos decaimento: w_i = (46341/65536)^i Q16 ≈ (√2/2)^i
layer_wts:   .word 65536,46341,32768,23170,16384,11585,8192
hex_b4: .ascii "0123456789ABCDEF"
.section .data
.align 6
layer_phases: .word 0,0,0,0,0,0,0  @ fase acumulada por camada
layer_vals:   .word 0,0,0,0,0,0,0  @ valor atual por camada (EMA)
g_overlap:    .word 0
g_hex_b4:     .byte 0,0,0,0,0,0,0,0,0x0A,0
.section .text
.syntax unified
.thumb
.type ws4,%function
ws4: push{lr};mov r2,r1;mov r1,r0;mov r0,#1;mov r7,#SYS_WRITE;swi #0;pop{pc}
.type ph4,%function
ph4:
    push{r4,r5,lr};mov r4,r0;ldr r5,=g_hex_b4;ldr r2,=hex_b4;mov r1,#28
.p4: lsr r0,r4,r1;and r0,r0,#0xF;ldrb r0,[r2,r0];strb r0,[r5],#1;subs r1,r1,#4;bge .p4
    mov r0,#0x0A;strb r0,[r5];ldr r0,=g_hex_b4;mov r1,#9;bl ws4;pop{r4,r5,pc}
@ sin_q16: r0=angle_q16 → r0=sin_q16 (Taylor x - x³/6 + x⁵/120)
.type sin_q16,%function
sin_q16:
    push{r4,r5,r6,lr}
    @ Normaliza para [0, 2PI)
.sn1: ldr r1,=Q16_2PI;cmp r0,r1;subge r0,r0,r1;bge .sn1
    @ Fold para [0, PI]: se >= PI → neg, x -= PI
    mov r4,#0  @ neg flag
    ldr r1,=Q16_PI;cmp r0,r1;subge r0,r0,r1;movge r4,#1
    @ Taylor: x³/6
    @ x2 = x*x >> 16
    smull r1,r2,r0,r0; lsr r1,r1,#16; orr r5,r1,r2,lsl #16  @ x2 Q16
    @ x3 = x2*x >> 16
    smull r1,r2,r5,r0; lsr r1,r1,#16; orr r6,r1,r2,lsl #16  @ x3 Q16
    @ t1 = x3 * INV6 >> 16
    ldr r1,=Q16_INV6; smull r1,r2,r6,r1; lsr r1,r1,#16; orr r1,r1,r2,lsl #16
    @ x5 = x3*x2 >> 16
    smull r2,r3,r6,r5; lsr r2,r2,#16; orr r2,r2,r3,lsl #16
    @ t2 = x5 * INV120 >> 16
    ldr r3,=Q16_INV120; smull r3,r5,r2,r3; lsr r3,r3,#16; orr r3,r3,r5,lsl #16
    @ sin ≈ x - t1 + t2
    sub r0,r0,r1; add r0,r0,r3
    @ clamp [0, 65535]
    cmp r0,#0; movlt r0,#0; ldr r1,=65535; cmp r0,r1; movgt r0,r1
    @ aplica sinal
    cmp r4,#1; rsbge r0,r0,#0
    pop{r4,r5,r6,pc}
.align 2
.global _start
.thumb_func
_start:
    ldr r0,=msg_b4;mov r1,#msg_b4_len;bl ws4
    @ 42 ciclos
    mov r9,#0
.b4_loop:
    cmp r9,#PERIOD;bge .b4_done
    mov r10,#0  @ overlap acum
    @ 7 camadas
    mov r8,#0
.lay_loop:
    cmp r8,#N_LAYERS;bge .lay_done
    @ Carrega freq e fase da camada
    ldr r4,=layer_freqs;ldr r0,[r4,r8,lsl #2]
    ldr r4,=layer_phases;ldr r1,[r4,r8,lsl #2]
    add r0,r0,r1   @ angulo = fase + freq*cycle → aqui fase acumulada
    str r0,[r4,r8,lsl #2]   @ armazena nova fase
    bl sin_q16      @ r0 = sin(fase)
    @ Aplica peso: w*sin >> 16
    ldr r4,=layer_wts;ldr r1,[r4,r8,lsl #2]
    smull r2,r3,r0,r1;lsr r2,r2,#16;orr r0,r2,r3,lsl #16
    @ EMA com valor anterior
    ldr r4,=layer_vals;ldr r1,[r4,r8,lsl #2]
    @ new = 0.75*old + 0.25*sin_weighted
    ldr r2,=49152;smull r5,r6,r1,r2;lsr r5,r5,#16;orr r1,r5,r6,lsl #16
    ldr r2,=16384;smull r5,r6,r0,r2;lsr r5,r5,#16;orr r0,r5,r6,lsl #16
    add r0,r0,r1
    str r0,[r4,r8,lsl #2]
    @ Acumula overlap via XOR
    eor r10,r10,r0
    add r8,r8,#1;b .lay_loop
.lay_done:
    ldr r4,=g_overlap;str r10,[r4]
    add r9,r9,#1;b .b4_loop
.b4_done:
    @ Imprime 7 valores de camada
    ldr r4,=layer_vals; mov r8,#0
.print_lay:
    cmp r8,#N_LAYERS;bge .print_done
    ldr r0,=msg_lay;mov r1,#msg_lay_len;bl ws4
    ldr r0,[r4,r8,lsl #2];bl ph4
    add r8,r8,#1;b .print_lay
.print_done:
    ldr r0,=msg_ovl;mov r1,#msg_ovl_len;bl ws4
    ldr r0,=g_overlap;ldr r0,[r0];bl ph4
    ldr r0,=msg_raf;mov r1,#msg_raf_len;bl ws4
    mov r7,#SYS_EXIT;mov r0,#0;swi #0
F
ok "rafaelia_b4.S ($(wc -l < $D/rafaelia_b4.S)L)"
