// T01_B: LEXER puro C bare-metal — branchless, ZERO libc, tabela única
static const unsigned char CLASS[256] = {
    [' ']=1, ['\n']=1, ['\t']=1,
    ['a']=2,['b']=2,['c']=2,['d']=2,['e']=2,['f']=2,['g']=2,['h']=2,
    ['i']=2,['j']=2,['k']=2,['l']=2,['m']=2,['n']=2,['o']=2,['p']=2,

    ['y']=2,['z']=2,
    ['0']=3,['1']=3,['2']=3,['3']=3,['4']=3,['5']=3,['6']=3,['7']=3,
    ['8']=3,['9']=3,
    [':']=4,['.']=4,[',']=4,[';']=4,
};
typedef struct { unsigned offset, len; unsigned char klass; } Tok;
unsigned lex(const char* s, unsigned n, Tok* out, unsigned max){
    unsigned i=0, k=0; unsigned char cur=0;
    while(i<n && k<max){
        unsigned char c = CLASS[(unsigned char)s[i]];
        if(c!=cur){
            if(cur && cur!=1 && k<max){ out[k].len = i - out[k].offset; k++; }
            if(c && c!=1){ out[k].offset = i; out[k].klass = c; }
            cur = c;
        }
        i++;
    }
    return k;
}

seed_T01_C_signed_token_chain() { cat << 'SEED'
# T01_C: cada TOKEN assinado por CRC32C encadeado (audit chain)
import struct
POLY = 0x82F63B78
def crc32c(data, init=0xFFFFFFFF):
    c = init
    for b in data:
        c ^= b
        for _ in range(8): c = (c>>1) ^ (POLY & -(c&1))
    return ~c & 0xFFFFFFFF
def tokenize_signed(text):
    prev = 0xFFFFFFFF
    tokens = []
    for w in text.split():
        tok = w.encode('utf-8')
        chain = crc32c(tok + struct.pack('<I', prev))
        tokens.append({"word": w, "crc": hex(chain)})
        prev = chain
    return tokens, prev
SEED
}

seed_T01_D_polyglot_voting() { cat << 'SEED'
# T01_D: tokenizadores ESPECIALIZADOS por língua, votam por confiança
def tokenize_polyglot(text):
    candidates = []
    if any('\u0590' <= c <= '\u05FF' for c in text):
        candidates.append({"lang":"he","tokens":text.split(),"conf":0.95})
    if any('\u0370' <= c <= '\u03FF' for c in text):
        candidates.append({"lang":"el","tokens":text.split(),"conf":0.92})
    if all(ord(c) < 0x180 or c.isspace() for c in text):
        candidates.append({"lang":"la","tokens":text.split(),"conf":0.80})
    candidates.append({"lang":"pt","tokens":text.split(),"conf":0.70})
    return max(candidates, key=lambda c: c['conf'])
SEED
}

# T01_E: token em SUPERPOSIÇÃO até colapsar no compile-time
import numpy as np
class QuantumToken:
    def __init__(self, word, meanings):
        self.word = word; self.meanings = meanings
        n = len(meanings)
        self.amp = np.ones(n, dtype=complex) / np.sqrt(n)
    def observe(self, context_vec):
        probs = np.abs(self.amp)**2
        for i, m in enumerate(self.meanings):
            probs[i] *= 1.0 + np.dot(context_vec, m.get('vec',[0]*8))
        probs /= probs.sum() + 1e-9
        chosen = np.random.choice(len(probs), p=probs)
        self.amp = np.zeros_like(self.amp); self.amp[chosen] = 1.0
        return self.meanings[chosen]
