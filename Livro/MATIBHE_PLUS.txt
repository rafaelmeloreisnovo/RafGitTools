#pragma once
/* raf_sha3.h — Keccak-f[1600] real, SHA3-256 conforme NIST FIPS 202.
 * Sem stubs · sem djb2 · sem deps · branchless onde possível.
 * Estado: 25 lanes de 64-bit (5x5 matriz). 24 rounds. */
#include "raf_arch.h"

#define SHA3_RATE_256    136u   /* bytes (1088 bits) */
#define SHA3_DIGEST_256  32u    /* bytes (256 bits) */

static const u64 SHA3_RC[24] = {
    0x0000000000000001ULL, 0x0000000000008082ULL,
    0x800000000000808AULL, 0x8000000080008000ULL,
    0x000000000000808BULL, 0x0000000080000001ULL,
    0x8000000080008081ULL, 0x8000000000008009ULL,
    0x000000000000008AULL, 0x0000000000000088ULL,
    0x0000000080008009ULL, 0x000000008000000AULL,
    0x000000008000808BULL, 0x800000000000008BULL,
    0x8000000000008089ULL, 0x8000000000008003ULL,
    0x8000000000008002ULL, 0x8000000000000080ULL,
    0x000000000000800AULL, 0x800000008000000AULL,
    0x8000000080008081ULL, 0x8000000000008080ULL,
    0x0000000080000001ULL, 0x8000000080008008ULL
};

static const u32 SHA3_ROT[24] = {
     1,  3,  6, 10, 15, 21, 28, 36,
    45, 55,  2, 14, 27, 41, 56,  8,
    25, 43, 62, 18, 39, 61, 20, 44
};

static const u32 SHA3_PI[24] = {
    10,  7, 11, 17, 18,  3,  5, 16,
     8, 21, 24,  4, 15, 23, 19, 13,
    12,  2, 20, 14, 22,  9,  6,  1
};

static inline u64 _rotl64(u64 x, u32 n) {
    return (x << n) | (x >> (64u - n));
}

/* Keccak-f[1600] — 24 rounds */
static void _keccak_f(u64 st[25]) {
    for (u32 r = 0; r < 24u; r++) {
        /* θ (theta) */
        u64 bc[5];
        for (u32 i = 0; i < 5u; i++)
            bc[i] = st[i] ^ st[i+5] ^ st[i+10] ^ st[i+15] ^ st[i+20];
        for (u32 i = 0; i < 5u; i++) {
            u64 t = bc[(i+4u)%5u] ^ _rotl64(bc[(i+1u)%5u], 1);
            for (u32 j = 0; j < 25u; j += 5u) st[j+i] ^= t;
        }
        /* ρ + π (rho + pi) */
        u64 t = st[1];
        for (u32 i = 0; i < 24u; i++) {
            u32 j = SHA3_PI[i];
            u64 tmp = st[j];
            st[j] = _rotl64(t, SHA3_ROT[i]);
            t = tmp;
        }
        /* χ (chi) */
        for (u32 j = 0; j < 25u; j += 5u) {
            u64 a0 = st[j+0], a1 = st[j+1], a2 = st[j+2],
                a3 = st[j+3], a4 = st[j+4];
            st[j+0] = a0 ^ ((~a1) & a2);
            st[j+1] = a1 ^ ((~a2) & a3);
            st[j+2] = a2 ^ ((~a3) & a4);
            st[j+3] = a3 ^ ((~a4) & a0);
            st[j+4] = a4 ^ ((~a0) & a1);
        }
        /* ι (iota) */
        st[0] ^= SHA3_RC[r];
    }
}

/* SHA3-256: absorb input, squeeze 32 bytes */
static void raf_sha3_256(const u8 *in, u32 len, u8 out[32]) {
    u64 st[25];
    for (u32 i = 0; i < 25u; i++) st[i] = 0ULL;
    u32 off = 0u;
    /* Absorb full blocks */
    while (len - off >= SHA3_RATE_256) {
        u8 *stb = (u8*)st;
        for (u32 i = 0; i < SHA3_RATE_256; i++) stb[i] ^= in[off + i];
        _keccak_f(st);
        off += SHA3_RATE_256;
    }
    /* Padding: 0x06 || 0x00... || 0x80 */
    u32 rem = len - off;
    u8 block[SHA3_RATE_256];
    for (u32 i = 0; i < SHA3_RATE_256; i++) block[i] = 0u;
    for (u32 i = 0; i < rem; i++) block[i] = in[off + i];
    block[rem] = 0x06u;
    block[SHA3_RATE_256 - 1u] |= 0x80u;
    u8 *stb = (u8*)st;
    for (u32 i = 0; i < SHA3_RATE_256; i++) stb[i] ^= block[i];
    _keccak_f(st);
    /* Squeeze 32 bytes */
    u8 *src = (u8*)st;
    for (u32 i = 0; i < 32u; i++) out[i] = src[i];
}

#pragma once
/* raf_blake3.h — BLAKE3 real (compress fn + chunk hashing)
 * Implementação completa para input <= 1 chunk (1024B).
 * Para inputs maiores, usa Merkle binary tree (não implementado aqui:
 * suficiente para nosso runtime de até 1KB). */
#include "raf_arch.h"

static const u32 BLAKE3_IV[8] = {
    0x6A09E667u, 0xBB67AE85u, 0x3C6EF372u, 0xA54FF53Au,
    0x510E527Fu, 0x9B05688Cu, 0x1F83D9ABu, 0x5BE0CD19u
};

static const u8 BLAKE3_PERM[16] = {
    2, 6, 3, 10, 7, 0, 4, 13, 1, 11, 12, 5, 9, 14, 15, 8
};

/* Flags */
#define B3_CHUNK_START  1u
#define B3_CHUNK_END    2u
#define B3_PARENT       4u
#define B3_ROOT         8u

static inline u32 _b3_rotr(u32 x, u32 n) {
    return (x >> n) | (x << (32u - n));
}

/* G function — mistura 4 palavras */
static inline void _b3_g(u32 *st, u32 a, u32 b, u32 c, u32 d, u32 mx, u32 my) {
    st[a] = st[a] + st[b] + mx;
    st[d] = _b3_rotr(st[d] ^ st[a], 16);
    st[c] = st[c] + st[d];
    st[b] = _b3_rotr(st[b] ^ st[c], 12);
    st[a] = st[a] + st[b] + my;
    st[d] = _b3_rotr(st[d] ^ st[a], 8);
    st[c] = st[c] + st[d];
    st[b] = _b3_rotr(st[b] ^ st[c], 7);
}

/* Round: 4 G colunas + 4 G diagonais */
static inline void _b3_round(u32 *st, const u32 *m) {
    _b3_g(st, 0, 4,  8, 12, m[ 0], m[ 1]);
    _b3_g(st, 1, 5,  9, 13, m[ 2], m[ 3]);
    _b3_g(st, 2, 6, 10, 14, m[ 4], m[ 5]);
    _b3_g(st, 3, 7, 11, 15, m[ 6], m[ 7]);
    _b3_g(st, 0, 5, 10, 15, m[ 8], m[ 9]);
    _b3_g(st, 1, 6, 11, 12, m[10], m[11]);
    _b3_g(st, 2, 7,  8, 13, m[12], m[13]);
    _b3_g(st, 3, 4,  9, 14, m[14], m[15]);
}

/* Compress: 64 bytes de entrada → 32 bytes de saída */
static void _b3_compress(const u32 cv[8], const u32 block_words[16],
                         u64 counter, u32 block_len, u32 flags,
                         u32 out[16]) {
    u32 st[16];
    for (u32 i = 0; i < 8u; i++) st[i] = cv[i];
    st[ 8] = BLAKE3_IV[0];
    st[ 9] = BLAKE3_IV[1];
    st[10] = BLAKE3_IV[2];
    st[11] = BLAKE3_IV[3];
    st[12] = (u32)(counter & 0xFFFFFFFFu);
    st[13] = (u32)(counter >> 32);
    st[14] = block_len;
    st[15] = flags;

    u32 m[16];
    for (u32 i = 0; i < 16u; i++) m[i] = block_words[i];

    /* 7 rounds com permutação */
    for (u32 r = 0; r < 7u; r++) {
        _b3_round(st, m);
        if (r < 6u) {
            u32 m_new[16];
            for (u32 i = 0; i < 16u; i++) m_new[i] = m[BLAKE3_PERM[i]];
            for (u32 i = 0; i < 16u; i++) m[i] = m_new[i];
        }
    }

    /* output_block: feed-forward */
    for (u32 i = 0; i < 8u; i++) {
        out[i]   = st[i] ^ st[i + 8u];
        out[i+8] = st[i + 8u] ^ cv[i];
    }
}

/* Hash de um buffer pequeno (<=1024B) gera 32B output */
static void raf_blake3(const u8 *in, u32 len, u8 out[32]) {
    u32 cv[8];
    for (u32 i = 0; i < 8u; i++) cv[i] = BLAKE3_IV[i];

    u32 off = 0u;
    u32 blocks_in_chunk = 0u;
    while (len - off > 64u) {
        u32 m[16];
        for (u32 i = 0; i < 16u; i++) {
            u32 j = off + i * 4u;
            m[i] = (u32)in[j] | ((u32)in[j+1] << 8) |
                   ((u32)in[j+2] << 16) | ((u32)in[j+3] << 24);
        }
        u32 flags = 0u;
        if (blocks_in_chunk == 0u) flags |= B3_CHUNK_START;
        u32 outw[16];
        _b3_compress(cv, m, 0ULL, 64u, flags, outw);
        for (u32 i = 0; i < 8u; i++) cv[i] = outw[i];
        off += 64u;
        blocks_in_chunk++;
    }

    /* Bloco final */
    u32 rem = len - off;
    u8 last_block[64];
    for (u32 i = 0; i < 64u; i++) last_block[i] = 0u;
    for (u32 i = 0; i < rem; i++) last_block[i] = in[off + i];
    u32 m[16];
    for (u32 i = 0; i < 16u; i++) {
        u32 j = i * 4u;
        m[i] = (u32)last_block[j] | ((u32)last_block[j+1] << 8) |
               ((u32)last_block[j+2] << 16) | ((u32)last_block[j+3] << 24);
    }
    u32 flags = B3_CHUNK_END | B3_ROOT;
    if (blocks_in_chunk == 0u) flags |= B3_CHUNK_START;
    u32 outw[16];
    _b3_compress(cv, m, 0ULL, rem, flags, outw);
    for (u32 i = 0; i < 8u; i++) {
        out[i*4 + 0] = (u8)(outw[i] & 0xFFu);
        out[i*4 + 1] = (u8)((outw[i] >> 8) & 0xFFu);
        out[i*4 + 2] = (u8)((outw[i] >> 16) & 0xFFu);
        out[i*4 + 3] = (u8)((outw[i] >> 24) & 0xFFu);
    }
}

#pragma once
/* raf_fft.h — FFT radix-2 Cooley-Tukey em Q16.16, sem float, sem libm.
 * Twiddle precomputado por Taylor. Bit-reversal. In-place. */
#include "raf_arch.h"

#define FFT_LOG_MAX_N 10u
#define FFT_MAX_N     (1u << FFT_LOG_MAX_N)  /* 1024 */
#define FFT_Q16_ONE   65536
#define FFT_PI_Q16    205887  /* π · 65536 */

typedef struct {
    i32 re, im;
} cq16;

/* Taylor sin/cos em Q16.16 — domínio [-π, π] */
static i32 _fft_sin(i32 x_q16) {
    /* reduzir para [-π, π] */
    while (x_q16 >  FFT_PI_Q16) x_q16 -= 2*FFT_PI_Q16;
    while (x_q16 < -FFT_PI_Q16) x_q16 += 2*FFT_PI_Q16;
    /* sin(x) ≈ x - x³/6 + x⁵/120 - x⁷/5040 */
    i64 x = x_q16;
    i64 x2 = (x * x) >> 16;
    i64 x3 = (x2 * x) >> 16;
    i64 x5 = (x3 * x2) >> 16;
    i64 x7 = (x5 * x2) >> 16;
    i64 r = x - x3 / 6 + x5 / 120 - x7 / 5040;
    return (i32)r;
}
static i32 _fft_cos(i32 x_q16) {
    return _fft_sin(x_q16 + (FFT_PI_Q16 / 2));
}

/* Bit-reverse de n bits */
static u32 _fft_brev(u32 x, u32 logN) {
    u32 r = 0u;
    for (u32 i = 0; i < logN; i++) {
        r = (r << 1) | (x & 1u);
        x >>= 1;
    }
    return r;
}

/* FFT in-place, N deve ser potência de 2, N <= FFT_MAX_N */
static void raf_fft_q16(cq16 *x, u32 N, u32 logN, u32 inverse) {
    /* bit-reversal */
    for (u32 i = 0; i < N; i++) {
        u32 j = _fft_brev(i, logN);
        if (j > i) {
            cq16 t = x[i]; x[i] = x[j]; x[j] = t;
        }
    }
    /* butterflies */
    for (u32 s = 1u; s <= logN; s++) {
        u32 m = 1u << s;
        u32 mh = m >> 1;
        /* twiddle angle: -2π/m (forward) ou +2π/m (inverse) */
        i32 base_angle = inverse ?
            (2 * FFT_PI_Q16 / (i32)m) :
            -(2 * FFT_PI_Q16 / (i32)m);
        for (u32 k = 0; k < N; k += m) {
            for (u32 j = 0; j < mh; j++) {
                i32 ang = base_angle * (i32)j;
                i32 wr = _fft_cos(ang);
                i32 wi = _fft_sin(ang);
                cq16 a = x[k + j];
                cq16 b = x[k + j + mh];
                /* t = w · b */
                i32 tr = (i32)(((i64)wr * b.re - (i64)wi * b.im) >> 16);
                i32 ti = (i32)(((i64)wr * b.im + (i64)wi * b.re) >> 16);
                x[k + j].re        = a.re + tr;
                x[k + j].im        = a.im + ti;
                x[k + j + mh].re   = a.re - tr;
                x[k + j + mh].im   = a.im - ti;
            }
        }
    }
    /* normalização inversa */
    if (inverse) {
        for (u32 i = 0; i < N; i++) {
            x[i].re /= (i32)N;
            x[i].im /= (i32)N;
        }
    }
}

/* Magnitude espectral aproximada |X| = sqrt(re² + im²) Q16 */
static u32 raf_fft_mag(const cq16 *X, u32 k) {
    i64 re = X[k].re;
    i64 im = X[k].im;
    i64 sq = (re * re + im * im) >> 16;
    /* Newton sqrt em Q16 */
    if (sq <= 0) return 0u;
    i64 r = sq;
    for (u32 i = 0; i < 16u; i++) {
        if (r == 0) break;
        r = (r + sq / r) >> 1;
    }
    return (u32)r;
}

/* raf_btree.h — B-tree ordem 4 (até 3 chaves, 4 filhos por nó)
 * arena estática de 256 nós · sem malloc · sem deps */
#include "raf_arch.h"

#define BT_ORDER    4u
#define BT_MAX_K    (BT_ORDER - 1u)   /* 3 chaves */
#define BT_N_NODES  256u

typedef struct {
    u32 keys[BT_MAX_K];
    u32 vals[BT_MAX_K];
    u32 child[BT_ORDER];   /* índices na arena, 0xFFFFFFFF = NULL */
    u32 n;                 /* número de chaves */
    u32 leaf;              /* 1 se folha */
    u32 in_use;
    u32 reserved;
} BTNode;

typedef struct {
    BTNode nodes[BT_N_NODES];
    u32 root;
    u32 next_free;
    u32 size;
    u32 inserts, lookups, splits;
} BTree;

#define BT_NULL 0xFFFFFFFFu

static u32 _bt_alloc(BTree *t) {
    for (u32 i = t->next_free; i < BT_N_NODES; i++) {
        if (!t->nodes[i].in_use) {
            t->nodes[i].in_use = 1u;
            t->nodes[i].n = 0u;
            t->nodes[i].leaf = 1u;
            for (u32 k = 0; k < BT_ORDER; k++) t->nodes[i].child[k] = BT_NULL;
            t->next_free = i + 1u;
            return i;
        }
    }
    return BT_NULL;
}

static void bt_init(BTree *t) {
    _mset(t, 0, sizeof(*t));
    t->root = _bt_alloc(t);
}

/* Search */
static u32 bt_get(BTree *t, u32 key, u32 *val) {
    t->lookups++;
    u32 idx = t->root;
    while (idx != BT_NULL) {
        BTNode *n = &t->nodes[idx];
        u32 i = 0u;
        while (i < n->n && key > n->keys[i]) i++;
        if (i < n->n && key == n->keys[i]) {
            *val = n->vals[i];
            return 1u;
        }
        if (n->leaf) return 0u;
        idx = n->child[i];
    }
    return 0u;
}

/* Split child: child y at index i of parent x is full → split */
static void _bt_split(BTree *t, u32 x_idx, u32 i) {
    t->splits++;
    BTNode *x = &t->nodes[x_idx];
    u32 y_idx = x->child[i];
    BTNode *y = &t->nodes[y_idx];
    u32 z_idx = _bt_alloc(t);
    BTNode *z = &t->nodes[z_idx];
    z->leaf = y->leaf;
    u32 t_mid = BT_MAX_K / 2u;
    z->n = BT_MAX_K - t_mid - 1u;
    for (u32 j = 0; j < z->n; j++) {
        z->keys[j] = y->keys[j + t_mid + 1u];
        z->vals[j] = y->vals[j + t_mid + 1u];
    }
    if (!y->leaf) {
        for (u32 j = 0; j <= z->n; j++)
            z->child[j] = y->child[j + t_mid + 1u];
    }
    y->n = t_mid;
    for (u32 j = x->n; j > i; j--) x->child[j+1] = x->child[j];
    x->child[i+1] = z_idx;
    for (u32 j = x->n; j > i; j--) {
        x->keys[j] = x->keys[j-1];
        x->vals[j] = x->vals[j-1];
    }
    x->keys[i] = y->keys[t_mid];
    x->vals[i] = y->vals[t_mid];
    x->n++;
}

static void _bt_insert_nonfull(BTree *t, u32 x_idx, u32 key, u32 val) {
    BTNode *x = &t->nodes[x_idx];
    i32 i = (i32)x->n - 1;
    if (x->leaf) {
        while (i >= 0 && key < x->keys[i]) {
            x->keys[i+1] = x->keys[i];
            x->vals[i+1] = x->vals[i];
            i--;
        }
        x->keys[i+1] = key;
        x->vals[i+1] = val;
        x->n++;
    } else {
        while (i >= 0 && key < x->keys[i]) i--;
        i++;
        if (t->nodes[x->child[i]].n == BT_MAX_K) {
            _bt_split(t, x_idx, (u32)i);
            x = &t->nodes[x_idx];
            if (key > x->keys[i]) i++;
        }
        _bt_insert_nonfull(t, x->child[i], key, val);
    }
}

static u32 bt_put(BTree *t, u32 key, u32 val) {
    t->inserts++;
    u32 r = t->root;
    if (t->nodes[r].n == BT_MAX_K) {
        u32 s_idx = _bt_alloc(t);
        if (s_idx == BT_NULL) return 0u;
        BTNode *s = &t->nodes[s_idx];
        s->leaf = 0u;
        s->n = 0u;
        s->child[0] = r;
        t->root = s_idx;
        _bt_split(t, s_idx, 0);
        _bt_insert_nonfull(t, s_idx, key, val);
    } else {
        _bt_insert_nonfull(t, r, key, val);
    }
    t->size++;
    return 1u;
}

#pragma once
/* raf_phonetic.h — extração de features fonéticas para T⁷ via Goertzel.
 * Input: i16 PCM mono. Output: estado T⁷ Q16.
 * 7 extratores: F0 (Goertzel), 4 bandas espectrais, variância pitch,
 * razão pico/médio, regularidade, silêncio, correlação temporal. */
#include "raf_arch.h"
#include "raf_fft.h"

#define PH_SR        8000u
#define PH_N_WIN     32u
#define PH_WIN_SIZE  256u

/* Goertzel magnitude para freq target (Hz) na janela size */
static u32 _ph_goertzel(const i16 *x, u32 n, u32 sample_rate, u32 target_hz) {
    /* k = round(N · f / fs) */
    u32 k = (target_hz * n + sample_rate/2u) / sample_rate;
    /* ω = 2π · k / N → 2 cos(ω) Q16 */
    i32 omega_q16 = (i32)((u64)(2u * FFT_PI_Q16) * k / n);
    i32 coeff = 2 * _fft_cos(omega_q16);
    /* q0 = coeff·q1 - q2 + sample */
    i64 q1 = 0, q2 = 0;
    for (u32 i = 0; i < n; i++) {
        i64 q0 = (((i64)coeff * q1) >> 16) - q2 + (i64)x[i];
        q2 = q1; q1 = q0;
    }
    /* mag² = q1² + q2² - q1·q2·coeff/2^16 */
    i64 mag2 = q1*q1 + q2*q2 - ((i64)coeff * q1 * q2 >> 16);
    if (mag2 < 0) mag2 = -mag2;
    /* sqrt Newton */
    i64 r = mag2;
    for (u32 i = 0; i < 16u; i++) {
        if (r == 0) break;
        r = (r + mag2/r) >> 1;
    }
    return (u32)(r & 0xFFFFFFFFu);
}

/* RMS de um buffer i16 em Q16 */
static u32 _ph_rms(const i16 *x, u32 n) {
    i64 sum = 0;
    for (u32 i = 0; i < n; i++) sum += (i64)x[i] * x[i];
    sum /= (i64)n;
    /* sqrt Newton */
    i64 r = sum + 1;
    for (u32 i = 0; i < 16u; i++) {
        if (r == 0) break;
        r = (r + sum/r) >> 1;
    }
    return (u32)r;
}

/* Estado T⁷ extraído de áudio */
typedef struct {
    u32 s[7];  /* Q16 */
} PhoneticT7;

static void raf_phonetic_extract(const i16 *pcm, u32 n_samples, PhoneticT7 *out) {
    /* === Extrator 1: F0 via Goertzel (varredura 80-300Hz, step 20Hz) === */
    u32 best_f = 80u;
    u32 best_mag = 0u;
    for (u32 f = 80u; f <= 300u; f += 20u) {
        u32 mag = _ph_goertzel(pcm, n_samples < 512u ? n_samples : 512u,
                               PH_SR, f);
        if (mag > best_mag) { best_mag = mag; best_f = f; }
    }
    out->s[0] = (best_f * 65536u) / 300u;  /* normalizado [0, Q16] */

    /* === Extrator 2: razão energia banda baixa (0-500Hz) === */
    u32 e_low = 0u, e_high = 0u;
    for (u32 f = 100u; f <= 500u; f += 100u)
        e_low += _ph_goertzel(pcm, n_samples < 512u ? n_samples : 512u, PH_SR, f);
    for (u32 f = 1000u; f <= 3000u; f += 500u)
        e_high += _ph_goertzel(pcm, n_samples < 512u ? n_samples : 512u, PH_SR, f);
    u32 e_total = e_low + e_high + 1u;
    out->s[1] = (e_low * 65536u) / e_total;

    /* === Extrator 3: centroide espectral aproximado === */
    u64 sum_wf = 0, sum_e = 0;
    for (u32 f = 200u; f <= 3000u; f += 200u) {
        u32 m = _ph_goertzel(pcm, n_samples < 256u ? n_samples : 256u, PH_SR, f);
        sum_wf += (u64)m * f;
        sum_e += m;
    }
    if (sum_e == 0) sum_e = 1;
    u32 centroid = (u32)(sum_wf / sum_e);
    out->s[2] = (centroid * 65536u) / 3000u;

    /* === Extrator 4: RMS médio === */
    out->s[3] = _ph_rms(pcm, n_samples) & 0xFFFFu;

    /* === Extrator 5: zero-crossing rate === */
    u32 zc = 0u;
    for (u32 i = 1; i < n_samples; i++) {
        if ((pcm[i-1] >= 0) != (pcm[i] >= 0)) zc++;
    }
    out->s[4] = (zc * 65536u) / n_samples;

    /* === Extrator 6: variância de energia inter-janela === */
    u32 win_e[PH_N_WIN];
    u32 step = n_samples / PH_N_WIN;
    if (step == 0) step = 1;
    u64 mean_e = 0;
    u32 n_win = 0;
    for (u32 w = 0; w < PH_N_WIN; w++) {
        u32 off = w * step;
        if (off + step > n_samples) break;
        win_e[w] = _ph_rms(pcm + off, step);
        mean_e += win_e[w];
        n_win++;
    }
    if (n_win == 0) n_win = 1;
    mean_e /= n_win;
    u64 var = 0;
    for (u32 w = 0; w < n_win; w++) {
        i64 d = (i64)win_e[w] - (i64)mean_e;
        var += (u64)(d * d);
    }
    var /= n_win;
    out->s[5] = (u32)(var & 0xFFFFu);

    /* === Extrator 7: correlação temporal E[t]·E[t+1] === */
    u64 corr = 0;
    for (u32 w = 1; w < n_win; w++) corr += (u64)win_e[w-1] * win_e[w];
    if (n_win > 1) corr /= (n_win - 1u);
    out->s[6] = (u32)(corr & 0xFFFFu);
}

static u32 _me_dist(const u32 *x, const u32 *y) {
    u32 s = 0u;
    for (u32 i = 0; i < 7u; i++) {
        u32 d = (x[i] > y[i]) ? (x[i] - y[i]) : (y[i] - x[i]);
        u32 dw = ME_Q16 - d;
        u32 mask = -(u32)(dw < d);
        d = (d & ~mask) | (dw & mask);
        s += d;
    }
    return s;
}
static u32 me_project(const u32 *x, const u32 attr[42][7], u32 *out) {
    u32 best_k = 0u, best_d = 0xFFFFFFFFu;
    for (u32 k = 0; k < 42u; k++) {
        u32 d = _me_dist(x, attr[k]);
        if (d < best_d) { best_d = d; best_k = k; }
    }
    for (u32 i = 0; i < 7u; i++) out[i] = attr[best_k][i];
    return best_k;
}
static u64 me_noise(u32 seed) {
    u64 h = 0xCBF29CE484222325ULL;
    for (u32 i = 0; i < 4u; i++) {
        h ^= (seed >> (i*8u)) & 0xFFu;
        h *= 0x100000001B3ULL;
    }
    return h;
}
static u32 me_step(const u32 *X, const u32 attr[42][7], u32 step, u32 *X_next) {
    u32 proj[7];
    u32 k = me_project(X, attr, proj);
    /* rotação 3x2D */
    u32 cs = ME_SPIRAL, sn = ME_SPIRAL >> 1;
    for (u32 i = 0; i < 3u; i++) {
        u32 a = proj[i*2], b = proj[i*2+1];
        u32 na = (u32)(((u64)a*cs) >> 16) - (u32)(((u64)b*sn) >> 16);
        u32 nb = (u32)(((u64)a*sn) >> 16) + (u32)(((u64)b*cs) >> 16);
        proj[i*2]   = na & 0xFFFFu;
        proj[i*2+1] = nb & 0xFFFFu;
    }
    u32 acc = 0u;
    for (u32 i = 0; i < 6u; i++) acc = (acc + proj[i]) & 0xFFFFu;
    proj[6] = acc;
    u64 eps = me_noise(step);
    u32 phi_lo = ME_PHI & 0xFFFFu;
    for (u32 i = 0; i < 7u; i++) {
        u32 sc = (u32)(((u64)proj[i] * phi_lo) >> 16);
        u32 nc = (u32)(eps >> (i*9u)) & 0xFFFFu;
        X_next[i] = (sc + (nc >> 4u)) & 0xFFFFu;
    }
    return k;
}

#pragma once
/* raf_vm.h — VM 10-opcode com dispatch table mutável (sem switch)
 * Estado: registradores R[16] u32, memória estática 4KB, stack 256 entradas.
 * Cada opcode é uma função; tabela substituível em runtime. */
#include "raf_arch.h"

#define VM_N_REGS    16u
#define VM_MEM_SIZE  4096u
#define VM_STK_SIZE  256u

enum {
    OP_HALT  = 0u,
    OP_NOP   = 1u,
    OP_LOAD  = 2u,
    OP_STORE = 3u,
    OP_ADD   = 4u,
    OP_SUB   = 5u,
    OP_XOR   = 6u,
    OP_PUSH  = 7u,
    OP_POP   = 8u,
    OP_JMP   = 9u,
    VM_N_OPS = 10u
};

typedef struct {
    u32 op, a, b, c;
} VInstr;

typedef struct VM VM;
typedef u32 (*op_fn_t)(VM *vm, const VInstr *ins);

struct VM {
    u32 R[VM_N_REGS];
    u8  M[VM_MEM_SIZE];
    u32 S[VM_STK_SIZE];
    u32 sp;
    u32 pc;
    u32 halted;
    u32 instr_count;
    op_fn_t handlers[VM_N_OPS];
};

static u32 _op_halt(VM *vm, const VInstr *i) {
    (void)i; vm->halted = 1u; return 3u;
}
static u32 _op_nop(VM *vm, const VInstr *i) {
    (void)vm; (void)i; return 1u;
}
static u32 _op_load(VM *vm, const VInstr *i) {
    /* R[a] = M[b] | (M[b+1]<<8) | (M[b+2]<<16) | (M[b+3]<<24) */
    if (i->a >= VM_N_REGS || i->b + 4u > VM_MEM_SIZE) return 0u;
    vm->R[i->a] = (u32)vm->M[i->b] | ((u32)vm->M[i->b+1] << 8) |
                  ((u32)vm->M[i->b+2] << 16) | ((u32)vm->M[i->b+3] << 24);
    return 1u;
}
static u32 _op_store(VM *vm, const VInstr *i) {
    if (i->a >= VM_N_REGS || i->b + 4u > VM_MEM_SIZE) return 0u;
    u32 v = vm->R[i->a];
    vm->M[i->b+0] = (u8)(v & 0xFFu);
    vm->M[i->b+1] = (u8)((v >> 8) & 0xFFu);
    vm->M[i->b+2] = (u8)((v >> 16) & 0xFFu);
    vm->M[i->b+3] = (u8)((v >> 24) & 0xFFu);
    return 1u;
}
static u32 _op_add(VM *vm, const VInstr *i) {
    if (i->a >= VM_N_REGS || i->b >= VM_N_REGS || i->c >= VM_N_REGS) return 0u;
    vm->R[i->a] = vm->R[i->b] + vm->R[i->c];
    return 1u;
}
static u32 _op_sub(VM *vm, const VInstr *i) {
    if (i->a >= VM_N_REGS || i->b >= VM_N_REGS || i->c >= VM_N_REGS) return 0u;
    vm->R[i->a] = vm->R[i->b] - vm->R[i->c];
    return 1u;
}
static u32 _op_xor(VM *vm, const VInstr *i) {
    if (i->a >= VM_N_REGS || i->b >= VM_N_REGS || i->c >= VM_N_REGS) return 0u;
    vm->R[i->a] = vm->R[i->b] ^ vm->R[i->c];
    return 1u;
}
static u32 _op_push(VM *vm, const VInstr *i) {
    if (i->a >= VM_N_REGS || vm->sp >= VM_STK_SIZE) return 0u;
    vm->S[vm->sp++] = vm->R[i->a];
    return 1u;
}
static u32 _op_pop(VM *vm, const VInstr *i) {
    if (i->a >= VM_N_REGS || vm->sp == 0u) return 0u;
    vm->R[i->a] = vm->S[--vm->sp];
    return 1u;
}
static u32 _op_jmp(VM *vm, const VInstr *i) {
    vm->pc = i->a;
    return 2u;  /* 2 = pc já foi modificado, não incrementar */
}

static void vm_init(VM *vm) {
    _mset(vm, 0, sizeof(*vm));
    vm->handlers[OP_HALT]  = _op_halt;
    vm->handlers[OP_NOP]   = _op_nop;
    vm->handlers[OP_LOAD]  = _op_load;
    vm->handlers[OP_STORE] = _op_store;
    vm->handlers[OP_ADD]   = _op_add;
    vm->handlers[OP_SUB]   = _op_sub;
    vm->handlers[OP_XOR]   = _op_xor;
    vm->handlers[OP_PUSH]  = _op_push;
    vm->handlers[OP_POP]   = _op_pop;
    vm->handlers[OP_JMP]   = _op_jmp;
}

static u32 vm_run(VM *vm, const VInstr *prog, u32 n_prog, u32 max_steps) {
    u32 steps = 0u;
    while (!vm->halted && vm->pc < n_prog && steps < max_steps) {
        const VInstr *ins = &prog[vm->pc];
        if (ins->op >= VM_N_OPS) return 0u;
        u32 r = vm->handlers[ins->op](vm, ins);
        if (r == 0u) return 0u;
        if (r == 3u) { vm->instr_count++; steps++; return 1u; }
        if (r != 2u) vm->pc++;
        vm->instr_count++;
        steps++;
    }
    return 1u;
}

/* Substitui handler em runtime (mutabilidade) */
static u32 vm_swap_handler(VM *vm, u32 op, op_fn_t fn) {
    if (op >= VM_N_OPS) return 0u;
    vm->handlers[op] = fn;
    return 1u;
}
