/* ──────────────── Hex print ──────────────── */
static void _ph(u32 v){
  static const char H[]="0123456789abcdef";
  char b[12]; b[0]='0'; b[1]='x';
  for(u32 i=0;i<8u;i++) b[2u+i] = H[(v >> (28u - i*4u)) & 0xFu];
  b[10]='\n'; _wn(b, 11);
}
static void _pd(u32 v){
  char b[16]; i32 i=15; b[i--]=0;
  if(!v){ b[i--]='0'; } else { while(v){ b[i--]='0'+(char)(v%10u); v/=10u; } }
  _w(b+i+1); _w("\n");
}

/* ──────────────── VM state ──────────────── */
#define STACK_SIZE 64
#define STR_AREA   4096

typedef struct {
  /* Stack de operandos: ponteiros para strings na string area */
  const char* stack[STACK_SIZE];
  i32 sp;
  /* String area: copia strings de bytecode aqui */
  u8 strs[STR_AREA];
  u32 str_top;
  /* Registros */
  const char* reg_god;
  const char* reg_spirit;
  const char* reg_logos;
  /* Estado de execução */
  u32 pc;
  u32 cycle;
  u32 sealed;
} VmState;

/* ──────────────── Helpers ──────────────── */
static const char* vm_intern(VmState* vm, const u8* src, u32 len){
  if(vm->str_top + len + 1 > STR_AREA) return "<OOM>";
  u8* dst = &vm->strs[vm->str_top];
  for(u32 i=0;i<len;i++) dst[i] = src[i];
  dst[len] = 0;
  vm->str_top += len + 1;
  return (const char*)dst;
}

static void vm_push(VmState* vm, const char* s){
  if(vm->sp < STACK_SIZE) vm->stack[vm->sp++] = s;
}
static const char* vm_pop(VmState* vm){
  return (vm->sp > 0) ? vm->stack[--vm->sp] : "<empty>";
}

/* ──────────────── Main dispatcher ──────────────── */
static void vm_execute(VmState* vm, const u8* bc, u32 n){
  vm->pc = 0;
  vm->cycle = 0;
  while(vm->pc < n && !vm->sealed){
    u8 op = bc[vm->pc];
    vm->cycle++;
    switch(op){
      case OP_NOP:        vm->pc++; break;
      case OP_IN_BEGIN:   rt_in_begin(vm); vm->pc++; break;
      case OP_CREATE:     rt_create(vm); vm->pc++; break;
      case OP_SPEAK:      rt_speak(vm); vm->pc++; break;
      case OP_LET_BE:     rt_let_be(vm); vm->pc++; break;
      case OP_SEPARATE:   rt_separate(vm); vm->pc++; break;
      case OP_NAME:       rt_name(vm); vm->pc++; break;
      case OP_LINK_AND:   rt_link_and(vm); vm->pc++; break;
      case OP_BLESS:      rt_bless(vm); vm->pc++; break;
      case OP_REST:       rt_rest(vm); vm->pc++; break;
      case OP_CALL_GOD:   rt_call_god(vm); vm->pc++; break;
      case OP_CALL_SPIRIT:rt_call_spirit(vm); vm->pc++; break;
      case OP_CALL_LOGOS: rt_call_logos(vm); vm->pc++; break;
      case OP_PRINT:      rt_print(vm); vm->pc++; break;
      case OP_SEAL_VERSE: rt_seal_verse(vm); break;
      case OP_PUSH_OBJ: {
        u32 j = vm->pc + 1;
        while(j < n && bc[j] != 0) j++;
        rt_push_obj(vm, bc + vm->pc + 1, j - vm->pc - 1);
        vm->pc = j + 1;
        break;
      }
      case OP_POP: {
        vm_pop(vm); vm->pc++; break;
      }
      default:
        _w("  [op "); _ph((u32)op); _w("]\n");
        vm->pc++;
        break;
    }
  }
}

/* ──────────────── Hex decoder (stdin) ──────────────── */
static u8 hex_nibble(char c){
  if(c >= '0' && c <= '9') return (u8)(c - '0');
  if(c >= 'a' && c <= 'f') return (u8)(c - 'a' + 10);
  if(c >= 'A' && c <= 'F') return (u8)(c - 'A' + 10);
  return 0xFF;
}
static u32 decode_hex(const u8* in, u32 n, u8* out, u32 max){
  u32 k = 0, i = 0;
  while(i + 1 < n && k < max){
    u8 hi = hex_nibble((char)in[i]);
    u8 lo = hex_nibble((char)in[i+1]);
    if(hi == 0xFF || lo == 0xFF){ i++; continue; }
    out[k++] = (u8)((hi << 4) | lo);
    i += 2;
  }
  return k;
}

# PARTE 3 — Pipeline integrado: bibliaCorpus → vm_runtime
# ═══════════════════════════════════════════════════════════════════════════

emit_pipeline_integration() { cat << 'PIPE_CODE'
#!/usr/bin/env bash
# pipeline_run.sh — pega bytecode_hex de um verso e executa na VM
#
# Uso:
#   bash pipeline_run.sh V01
#   bash pipeline_run.sh V09 V11 V12
#
# Requer: bibliaCorpus.txt + vm_run compilado no PATH ou $D/vm_run
D="${HOME}/.rafaelia/vm"
[ -x "$D/vm_run" ] || { echo "VM não compilada. Rode: bash vm_runtime.txt build"; exit 1; }
source "${HOME}/.rafaelia/bibliaCorpus.txt" 2>/dev/null || source /mnt/user-data/outputs/bibliaCorpus.txt

for v in "$@"; do
  echo ""
  echo "═══ Executando $v ═══"
  fn="verse_${v}_text"
  if declare -f "$fn" > /dev/null 2>&1; then
    BC=$("$fn" | grep -oP '"bytecode_hex":\s*"\K[^"]+')
    if [ -n "$BC" ]; then
      echo "  bytecode: $BC"
      echo "$BC" | xxd -r -p | "$D/vm_run"
    else
      echo "  [aviso] $v sem bytecode_hex"
    fi
  fi
done
PIPE_CODE
}

# ═══════════════════════════════════════════════════════════════════════════
# PARTE 4 — Build script (compila e roda)
# ═══════════════════════════════════════════════════════════════════════════

cmd_build() {
  D="${HOME}/.rafaelia/vm"
  mkdir -p "$D"
  CC=clang; command -v clang >/dev/null 2>&1 || CC=gcc
  ARCH=$(uname -m)
  case "$ARCH" in
    aarch64) AF="-march=armv8-a" ;;
    armv7*)  AF="-march=armv7-a -mfpu=neon-vfpv4 -mfloat-abi=softfp" ;;
    x86_64)  AF="-march=x86-64" ;;
    *)       AF="" ;;
  esac

  echo "[build] gerando vm_runtime.c..."
  emit_vm_runtime_c > "$D/vm_runtime.c"
  echo "  ✓ $D/vm_runtime.c ($(wc -l < $D/vm_runtime.c) linhas)"

  echo "[build] gerando gematria_full.py..."
  emit_gematria_full_py > "$D/gematria_full.py"
  chmod +x "$D/gematria_full.py"
  echo "  ✓ $D/gematria_full.py"

  echo "[build] gerando pipeline_run.sh..."
  emit_pipeline_integration > "$D/pipeline_run.sh"
  chmod +x "$D/pipeline_run.sh"
  echo "  ✓ $D/pipeline_run.sh"

  echo "[build] Modo A: libc..."
  if $CC -O2 $AF -Wall -Wno-unused-function -Wno-unused-variable \
         -Wno-unused-parameter "$D/vm_runtime.c" -o "$D/vm_run" 2>"$D/build.log"; then
    SZ=$(ls -lh "$D/vm_run" | awk '{print $5}')
    echo "  ✓ $D/vm_run ($SZ)"
  else
    echo "  ✗ build falhou — log em $D/build.log"
    head -20 "$D/build.log"
    return 1
  fi

  echo "[build] Modo B: freestanding..."
  if $CC -O1 -fPIE -fno-stack-protector -fno-builtin -nostdlib \
         -DFREESTANDING -e _start $AF \
         -Wno-unused-function -Wno-unused-variable -Wno-unused-parameter \
         "$D/vm_runtime.c" -o "$D/vm_run_fs" 2>"$D/build_fs.log"; then
    SZ=$(ls -lh "$D/vm_run_fs" | awk '{print $5}')
    echo "  ✓ $D/vm_run_fs ($SZ)"
  else
    echo "  ⚠ freestanding requer arch específica"
  fi

  echo ""
  echo "Build completo. Comandos:"
  echo "  $D/vm_run V01           — Gn 1:1"
  echo "  $D/vm_run V02           — Gn 1:3"
  echo "  $D/vm_run V09           — Jo 1:1"
  echo "  $D/vm_run V12           — Ap 1:8"
  echo "  $D/vm_run all           — todos"
  echo "  echo 'HEX' | xxd -r -p | $D/vm_run    — stdin"
  echo "  python3 $D/gematria_full.py            — F_gap fechado"
}

cmd_test() {
  D="${HOME}/.rafaelia/vm"
  [ -x "$D/vm_run" ] || { echo "Rode 'build' primeiro."; return 1; }

  echo ""
  echo "═══ TESTE 1: Gn 1:1 (V01) ═══"
  "$D/vm_run" V01

  echo ""
  echo "═══ TESTE 2: Gn 1:3 (V02) — Haja luz ═══"
  "$D/vm_run" V02

  echo ""
  echo "═══ TESTE 3: Jo 1:1 (V09) — Logos eterno ═══"
  "$D/vm_run" V09

  echo ""
  echo "═══ TESTE 4: Ap 1:8 (V12) — Alfa e Ômega ═══"

  echo ""
  echo "═══ TESTE 5: stdin com bytecode arbitrário ═══"
  echo "1011f00400ff" | xxd -r -p | "$D/vm_run"

  echo ""
  echo "═══ TESTE 6: F_gap — gematria grega completa ═══"
  python3 "$D/gematria_full.py" 2>&1 | head -30 || echo "(python3 indisponível)"
}
