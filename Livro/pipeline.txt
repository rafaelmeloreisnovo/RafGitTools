/* ═══════════════════════════════════════════════════════════════════════════
 * §1 CONSTANTES RAFAELIA
 * ═══════════════════════════════════════════════════════════════════════════ */
#define RAF_LAMBDA   0.866025f   /* √3/2 — contrator de coerência          */
#define RAF_PHI      1.618034f   /* φ    — razão áurea                     */
#define EMBED_DIM    64u
#define VOCAB_MAX    8192u

/* Nomes das 8 direções cognitivas (img20: atenção octogonal) */
static const char *RAF_DIR[8] = {
    "FOCO","PERCEP","MEMOR","INTUI","EMOC","VOL","IMAG","EXEC"
};

/* Nomes dos 6 clusters geolm */
static const char *CLUST_NAME[6] = {
    "TOROIDE","MATEMA","VVM","CICLO","SIMBOLIC","APLICADO"
};

/* Temperaturas base por direção/gate (img01: câmaras de estado)
 *   FOCO/EXEC  → baixa (convergência)
 *   IMAG/EMOC  → alta  (exploração)  */
static const f32 GATE_TEMP[8] = {
    0.55f,  /* 0 FOCO     — convergência máxima    */
    0.80f,  /* 1 PERCEP   — observação              */
    0.70f,  /* 2 MEMOR    — recuperação estável     */
    0.90f,  /* 3 INTUI    — salto moderado          */
    1.30f,  /* 4 EMOC     — exploração emocional    */
    1.00f,  /* 5 VOL      — equilíbrio              */
    1.50f,  /* 6 IMAG     — criatividade máxima     */
    0.50f   /* 7 EXEC     — execução determinística */
};

/* Modificadores de temperatura por cluster visual
 *   TOROIDE   → mais estruturado  (-10%)
 *   MATEMA    → preciso           (-5%)
 *   SIMBOLIC  → mais livre        (+20%)
 *   CICLO     → equilibrado       (0%)    */
static const f32 CLUST_MOD[6] = {
    0.90f,  /* 0 TOROIDE  */
    0.95f,  /* 1 MATEMA   */
    0.85f,  /* 2 VVM      */
    1.00f,  /* 3 CICLO    */
    1.20f,  /* 4 SIMBOLIC */
    1.05f   /* 5 APLICADO */
};

/* ═══════════════════════════════════════════════════════════════════════════
 * §2 GATE ENGINE — fibo_gate_step (gbs3_color_c.txt)
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 raf_gate(u32 t) {
    static const u32 S1[5] = {0,1,1,2,3};   /* 01123  */
    static const u32 S2[6] = {0,0,1,1,2,3}; /* 001123 */
    u32 a = S1[t % 5u];
    u32 b = S2[t % 6u];
    return (a + 2u*b + (t % 4u)) & 7u;       /* ∈ {0..7} */
}

/* Câmara RAFAELIA mapeada do gate (img01: 6 estados)
 * gate 0..7 → câmara 0..5 com semântica preservada */
static u32 gate_to_cam(u32 gate) {
    static const u32 MAP[8] = {0,1,2,3,4,5,4,0};
    return MAP[gate % 8u];
}

static const char *CAM_NAME[6] = {
    "Create","Struct","Trans","Polar","Integ","Harm"
};

/* ═══════════════════════════════════════════════════════════════════════════
 * §3 DELTA-P ENGINE — lê triad_trace.csv (gbs3)
 *   ΔP = P(stable|peaks) − P(stable|non-peaks)
 *   ΔP > 0.2  → picos Fibonacci altamente estáveis
 *   ΔP ≈ 0    → sem sinal
 *   ΔP < 0    → picos menos estáveis que baseline
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
    f32  delta_p;
    u32  n_rows, n_peaks, n_stable;
    f32  p_peaks, p_non;
    u8   loaded;
} TraceStats;

static void trim_nl(char *s) {
    size_t n = strlen(s);
    while(n && (s[n-1]=='\n'||s[n-1]=='\r')) s[--n] = 0;
}
static int col_idx(char cols[][64], int nc, const char *name) {
    for(int i=0;i<nc;i++) if(!strcmp(cols[i],name)) return i;
    return -1;
}
static int parse_csv_hdr(const char *line, char cols[][64], int max) {
    int c=0; const char *p=line;
    while(*p && c<max) {
        char buf[64]={0}; int bi=0;
        while(*p && *p!=',' && *p!='\n' && *p!='\r') {
            if(bi<63) buf[bi++]=*p; p++;
        }
        /* trim */
        int s=0; while(buf[s]==' ') s++;
        memmove(buf,buf+s,strlen(buf+s)+1);
        int j=(int)strlen(buf);
        while(j>0 && buf[j-1]==' ') buf[--j]=0;
        strncpy(cols[c],buf,63); cols[c][63]=0; c++;
        if(*p==',') p++;
    }
    return c;
}
static int split_row(char *line, char *cells[], int max) {
    int c=0; char *p=line;
    while(*p && c<max) {
        cells[c++]=p;
        while(*p && *p!=',') p++;
        if(*p==',') { *p=0; p++; }
    }
    return c;
}

static TraceStats load_trace(const char *path) {
    TraceStats ts; memset(&ts,0,sizeof(ts));
    FILE *fp=fopen(path,"rb"); if(!fp) return ts;

    char line[8192];
    if(!fgets(line,sizeof(line),fp)) { fclose(fp); return ts; }
    trim_nl(line);
    char cols[128][64]; int nc=parse_csv_hdr(line,cols,128);

    int iesc=col_idx(cols,nc,"escaped");
    int ista=col_idx(cols,nc,"stable_any");
    int igp =col_idx(cols,nc,"gate_in_peaks");
    int igate=col_idx(cols,nc,"gate_9to1");
    if(igate<0) igate=col_idx(cols,nc,"gate");

    if(iesc<0||ista<0||igp<0) { fclose(fp); return ts; }

    long pt=0,ps=0,nt=0,ns=0,rows=0;
    while(fgets(line,sizeof(line),fp)) {
        trim_nl(line); if(!line[0]) continue;
        char *cells[256]={0}; split_row(line,cells,256);
        int st  = (cells[ista])?atoi(cells[ista]):0;
        int gp  = (cells[igp]) ?atoi(cells[igp]) :0;
        int is_peak=gp;
        if(!is_peak && igate>=0 && cells[igate]) {
            int g=atoi(cells[igate]);
            if(g==3||g==4||g==8) is_peak=1;
        }
        if(is_peak){ pt++; ps+=st; } else { nt++; ns+=st; }
        rows++;
    }
    fclose(fp);

    ts.n_rows   = (u32)rows;
    ts.n_peaks  = (u32)pt;
    ts.n_stable = (u32)(ps+ns);
    ts.p_peaks  = pt>0 ? (f32)ps/(f32)pt : 0.f;
    ts.p_non    = nt>0 ? (f32)ns/(f32)nt : 0.f;
    ts.delta_p  = ts.p_peaks - ts.p_non;
    ts.loaded   = 1;
    return ts;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §4 CLUSTER CONTEXT — lê geolm_catalog.bin (geolm.txt)
 *   Retorna cluster do último entry catalogado
 * ═══════════════════════════════════════════════════════════════════════════ */
#define CAT_MAGIC 0xC47A1032u

typedef struct __attribute__((packed)) {
    u32 magic, entry_crc;
    u64 timestamp_ns, phash;
    u32 feat[7], encoded[7];
    u32 cluster, dist_to_centroid;
    u32 width, height, file_crc, tag_len;
    char fname[128], tag[64];
} CatEntry;

typedef struct {
    u32  cluster;        /* cluster do entry mais recente */
    u32  dist;           /* distância ao centróide */
    u32  n_entries;      /* total de entradas no catálogo */
    u32  cluster_hist[6];/* histograma: quantos entries por cluster */
    u8   loaded;
    char last_tag[64];
    char last_fname[128];
} CatalogCtx;

static CatalogCtx load_catalog(const char *path) {
    CatalogCtx ctx; memset(&ctx,0,sizeof(ctx)); ctx.cluster=0;
    FILE *fp=fopen(path,"rb"); if(!fp) return ctx;
    CatEntry e;
    u32 last_ts_hi=0, last_ts_lo=0;
    while(fread(&e,sizeof(CatEntry),1,fp)==1) {
        if(e.magic!=CAT_MAGIC) continue;
        if(e.cluster<6u) ctx.cluster_hist[e.cluster]++;
        ctx.n_entries++;
        u32 thi=(u32)(e.timestamp_ns>>32);
        u32 tlo=(u32)(e.timestamp_ns);
        if(thi>last_ts_hi||(thi==last_ts_hi&&tlo>last_ts_lo)) {
            last_ts_hi=thi; last_ts_lo=tlo;
            ctx.cluster = e.cluster<6u ? e.cluster : 0u;
            ctx.dist    = e.dist_to_centroid;
            strncpy(ctx.last_tag,  e.tag,  63); ctx.last_tag[63]=0;
            strncpy(ctx.last_fname,e.fname,127); ctx.last_fname[127]=0;
        }
    }
    fclose(fp);
    ctx.loaded = ctx.n_entries > 0;
    return ctx;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §5 TEMP_MAP — fórmula de temperatura unificada
 *   T_base  = GATE_TEMP[gate]
 *   T_delta = T_base × (1 − ΔP×k_dp)  [ΔP alto → foco]
 *   T_clust = T_delta × CLUST_MOD[c]
 *   T_final = clamp(T_clust, 0.3, 2.0)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define TEMP_MIN 0.30f
#define TEMP_MAX 2.00f
#define K_DP     0.50f   /* sensibilidade ao ΔP */

static f32 compute_temp(f32 delta_p, u32 gate, u32 cluster) {
    f32 t = GATE_TEMP[gate % 8u];
    /* ΔP calibra: mais estável → mais focado (temp menor) */
    t *= (1.f - delta_p * K_DP);
    /* cluster visual ajusta */
    t *= CLUST_MOD[cluster % 6u];
    /* clamp */
    if(t < TEMP_MIN) t = TEMP_MIN;
    if(t > TEMP_MAX) t = TEMP_MAX;
    return t;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §6 VOCAB + TOKENIZER (compacto, reusado de iaRAF)
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { char s[32]; u32 id,freq; u8 used; } VE;
typedef struct { VE *e; u32 sz,cap,unk,bos,eos; } Vocab;

static u8 G_VMEM[sizeof(VE)*VOCAB_MAX];
static Vocab G_VOCAB;

static void vocab_init(void) {
    G_VOCAB.e=(VE*)G_VMEM; G_VOCAB.sz=4; G_VOCAB.cap=VOCAB_MAX;
    G_VOCAB.unk=1; G_VOCAB.bos=2; G_VOCAB.eos=3;
    memset(G_VMEM,0,sizeof(G_VMEM));
    const char *sp[]={"<pad>","<unk>","<bos>","<eos>",NULL};
    for(int i=0;sp[i];i++){
        u32 h=5381u; const char *s=sp[i];
        while(*s) h=((h<<5)+h)^(u8)*s++;
        h%=VOCAB_MAX;
        strncpy(G_VOCAB.e[h].s,sp[i],31); G_VOCAB.e[h].id=(u32)i; G_VOCAB.e[h].used=1;
    }
}
static u32 vocab_ins(const char *s) {
    Vocab *v=&G_VOCAB;
    if(v->sz>=v->cap-1) return v->unk;
    u32 h=5381u; const char *p=s; while(*p) h=((h<<5)+h)^(u8)*p++; h%=v->cap;
    for(u32 pr=0;pr<v->cap;pr++) {
        VE *e=&v->e[h];
        if(!e->used){strncpy(e->s,s,31);e->id=v->sz++;e->freq=1;e->used=1;return e->id;}
        if(!strncmp(e->s,s,31)){e->freq++;return e->id;}
        h=(h+1)%v->cap;
    }
    return v->unk;
}
static u32 vocab_get(const char *s) {
    Vocab *v=&G_VOCAB;
    u32 h=5381u; const char *p=s; while(*p) h=((h<<5)+h)^(u8)*p++; h%=v->cap;
    for(u32 pr=0;pr<v->cap;pr++) {
        VE *e=&v->e[h];
        if(!e->used) return v->unk;
        if(!strncmp(e->s,s,31)) return e->id;
        h=(h+1)%v->cap;
    }
    return v->unk;
}
static const char *vocab_str(u32 id) {
    for(u32 i=0;i<VOCAB_MAX;i++)
        if(G_VOCAB.e[i].used && G_VOCAB.e[i].id==id) return G_VOCAB.e[i].s;
    return "<unk>";
}
static u32 tokenize(const char *text, u32 len, u32 *ids, u32 max) {
    char buf[32]; u32 bp=0, n=0;
    #define EMIT() do{if(bp>0&&n<max){buf[bp]=0;ids[n++]=vocab_ins(buf);bp=0;}}while(0)
    for(u32 i=0;i<len&&n<max;i++){
        u8 c=(u8)text[i]; if(c>='A'&&c<='Z') c+=32u;
        if(c==' '||c=='\t'||c=='\n'||c=='\r'){EMIT();continue;}
        if(c=='.'||c==','||c==';'||c==':'||c=='!'||c=='?'){
            EMIT();if(n<max){buf[0]=(char)c;buf[1]=0;ids[n++]=vocab_ins(buf);}continue;
        }
        if(bp<31) buf[bp++]=(char)c; else{EMIT();buf[bp++]=(char)c;}
    }
    EMIT();
    #undef EMIT
    return n;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §7 MODEL (bigram + embedding decay λ)
 * ═══════════════════════════════════════════════════════════════════════════ */
static f32 G_EMB[VOCAB_MAX][EMBED_DIM];
static f32 G_HW[EMBED_DIM][VOCAB_MAX];
static f32 G_HB[VOCAB_MAX];
static u32 G_VSIZ=0, G_STEP=0;
static f32 G_BEST_LOSS=1e9f;

static f32 fast_exp(f32 x){
    if(x>10.f){return 22026.f;}
    if(x<-10.f){return 4.54e-5f;}
    f32 x2=x*x,x3=x2*x,x4=x3*x;
    return 1.f+x+x2*.5f+x3*.16667f+x4*.04167f;
}

static void model_init(u32 vsz) {
    G_VSIZ=vsz; G_STEP=0; G_BEST_LOSS=1e9f;
    u32 rng=0x52414641u;
    f32 s=1.f/sqrtf((f32)EMBED_DIM);
    for(u32 v=0;v<vsz;v++)
        for(u32 d=0;d<EMBED_DIM;d++){
            rng=rng*1664525u+1013904223u;
            G_EMB[v][d]=((f32)(i32)rng)/2147483648.f*s;
        }
    for(u32 i=0;i<EMBED_DIM;i++)
        for(u32 j=0;j<vsz;j++){
            rng=rng*1664525u+1013904223u;
            G_HW[i][j]=((f32)(i32)rng)/2147483648.f*s;
        }
    memset(G_HB,0,vsz*sizeof(f32));
}

static f32 model_train_step(u32 ctx, u32 tgt, f32 *logits, f32 lr) {
    if(ctx>=G_VSIZ||tgt>=G_VSIZ) return 0.f;
    f32 *emb=G_EMB[ctx];
    for(u32 j=0;j<G_VSIZ;j++){
        f32 s=G_HB[j];
        for(u32 i=0;i<EMBED_DIM;i++) s+=G_HW[i][j]*emb[i];
        logits[j]=s;
    }
    f32 mx=logits[0]; for(u32 j=1;j<G_VSIZ;j++) if(logits[j]>mx) mx=logits[j];
    f32 sum=0.f;
    for(u32 j=0;j<G_VSIZ;j++){logits[j]=fast_exp(logits[j]-mx);sum+=logits[j];}
    f32 inv=sum>1e-9f?1.f/sum:0.f;
    for(u32 j=0;j<G_VSIZ;j++) logits[j]*=inv;
    f32 loss=-logf(logits[tgt]+1e-9f);
    static f32 dL[VOCAB_MAX];
    for(u32 j=0;j<G_VSIZ;j++) dL[j]=logits[j]-(j==tgt?1.f:0.f);
    for(u32 i=0;i<EMBED_DIM;i++){
        f32 ei=emb[i];
        for(u32 j=0;j<G_VSIZ;j++) G_HW[i][j]-=lr*dL[j]*ei;
    }
    for(u32 j=0;j<G_VSIZ;j++) G_HB[j]-=lr*dL[j];
    for(u32 i=0;i<EMBED_DIM;i++){
        f32 g=0.f;
        for(u32 j=0;j<G_VSIZ;j++) g+=G_HW[i][j]*dL[j];
        emb[i]-=lr*g;
    }
    /* λ-decay a cada 100 steps */
    if(G_STEP%100u==0u) for(u32 d=0;d<EMBED_DIM;d++) emb[d]*=RAF_LAMBDA;
    G_STEP++;
    return loss;
}

static u32 model_sample(u32 cur, f32 temp, u32 *rng) {
    static f32 logits[VOCAB_MAX];
    for(u32 j=0;j<G_VSIZ;j++){
        f32 s=G_HB[j];
        for(u32 d=0;d<EMBED_DIM;d++) s+=G_HW[d][j]*G_EMB[cur][d];
        logits[j]=s/temp;
    }
    f32 mx=logits[0]; for(u32 j=1;j<G_VSIZ;j++) if(logits[j]>mx) mx=logits[j];
    f32 sum=0.f;
    for(u32 j=0;j<G_VSIZ;j++){logits[j]=fast_exp(logits[j]-mx);sum+=logits[j];}
    *rng=(*rng)*1664525u+1013904223u;
    f32 r=((f32)(*rng>>1))/(f32)0x7FFFFFFFu*sum;
    f32 acc=0.f;
    for(u32 j=0;j<G_VSIZ;j++){acc+=logits[j];if(acc>=r) return j;}
    return 0u;
}

typedef struct {

static void print_header(void) {
    printf("tick | token          | gate/dir   | cam    | cluster  | ΔP     | temp\n");
    printf("-----|----------------|------------|--------|----------|--------|------\n");
}
static void print_out(const PipelineOut *o) {
    printf("%4u | %-14s | %u/%-8s | %-6s | %-8s | %+.4f | %.3f\n",
        o->tick,
        o->token_str,
        o->gate, RAF_DIR[o->gate%8u],
        CAM_NAME[o->cam],
        CLUST_NAME[o->cluster%6u],
        o->delta_p,
        o->temp);
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §9 PIPELINE RUNNER
 * ═══════════════════════════════════════════════════════════════════════════ */
static void run_pipeline(const char *start_word,
                          u32 n_tokens,
                          const TraceStats *ts,
                          const CatalogCtx *cc) {
    u32 cur = vocab_get(start_word);
    if(cur == G_VOCAB.unk) {
        printf("[pipeline] '%s' not in vocab, using bos\n", start_word);
        cur = G_VOCAB.bos;
    }

    u32 rng = (u32)time(NULL);
    print_header();

    for(u32 t=0; t<n_tokens; t++) {
        u32 gate    = raf_gate(G_STEP + t);
        u32 cam     = gate_to_cam(gate);
        u32 cluster = cc->loaded ? cc->cluster : 0u;
        f32 dp      = ts->loaded ? ts->delta_p : 0.0f;
        f32 temp    = compute_temp(dp, gate, cluster);
        u32 next    = model_sample(cur, temp, &rng);

        PipelineOut o = {
            .tick       = t,
            .token_id   = next,
            .token_str  = vocab_str(next),
            .gate       = gate,
            .cam        = cam,
            .cluster    = cluster,
            .delta_p    = dp,
            .temp       = temp
        };
        print_out(&o);
        cur = next;
        if(cur == G_VOCAB.eos) break;
    }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §10 TESTES B0-B11
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 G_PASS=0, G_FAIL=0;
static void _T(u32 ok, const char *name) {
    if(ok){ G_PASS++; printf("  [PASS] %s\n",name); }
    else  { G_FAIL++; printf("  [FAIL] %s\n",name); }
}
static u32 crc32c_test(const u8 *p, u32 n){
    u32 c=0xFFFFFFFFu;
    for(u32 i=0;i<n;i++){c^=(u32)p[i];for(u32 j=0;j<8u;j++)c=(c>>1)^(0x82F63B78u&-(c&1u));}
    return ~c;
}

static void run_tests(void) {
    printf("\n[tests] RAFAELIA UNIFIED PIPELINE B0-B11\n");

    /* B0: crc32c */
    const u8 v9[]={'1','2','3','4','5','6','7','8','9'};
    _T(crc32c_test(v9,9)==0xE3069283u, "B0:crc32c_NIST");

    /* B1: raf_gate determinístico e range */
    _T(raf_gate(0)==raf_gate(0),        "B1:gate_deterministic");
    _T(raf_gate(7)<8u,                  "B1:gate_range");

    /* B2: gate_to_cam */
    _T(gate_to_cam(0)<6u,               "B2:cam_range");
    _T(gate_to_cam(6)==4u,              "B2:cam_map_correct"); /* IMAG→Integ */

    /* B3: compute_temp — ΔP alto reduz temp */
    f32 t0=compute_temp(0.0f, 0u, 0u);  /* gate=FOCO, cluster=TOROIDE */
    f32 t1=compute_temp(0.5f, 0u, 0u);
    _T(t1 < t0,                         "B3:temp_dp_reduces");
    _T(t0 >= TEMP_MIN && t0 <= TEMP_MAX,"B3:temp_in_range");

    /* B4: compute_temp — cluster SIMBOLIC aumenta temp */
    f32 tc=compute_temp(0.0f, 5u, 4u);  /* gate=VOL, cluster=SIMBOLIC */
    f32 td=compute_temp(0.0f, 5u, 2u);  /* gate=VOL, cluster=VVM */
    _T(tc > td,                         "B4:clust_simbolic_higher_temp");

    /* B5: vocab */
    vocab_init();
    u32 id1=vocab_ins("verbo"), id2=vocab_ins("verbo");
    _T(id1==id2,                        "B5:vocab_dedup");
    _T(vocab_get("verbo")==id1,         "B5:vocab_lookup");

    /* B6: tokenizer */
    static u32 tids[64];
    u32 nt=tokenize("deus e amor",11u,tids,64u);
    _T(nt==3u,                          "B6:tokenize_3words");

    /* B7: model init */
    model_init(G_VOCAB.sz);
    _T(G_VSIZ==G_VOCAB.sz,              "B7:model_vocab_size");

    /* B8: model_train_step converge */
    static f32 lg[VOCAB_MAX];
    f32 l0=model_train_step(tids[0],tids[1],lg,0.1f);
    f32 l1=model_train_step(tids[0],tids[1],lg,0.1f);
    _T(l0>0.f,                          "B8:loss_positive");
    _T(l1<=l0+0.5f,                     "B8:loss_non_increasing");

    /* B9: model_sample retorna id válido */
    u32 rng=12345u;
    u32 sam=model_sample(tids[0], 1.0f, &rng);
    _T(sam<G_VSIZ,                      "B9:sample_valid_id");

    /* B10: gate sequência não constante */
    u32 same=1;
    for(u32 i=1;i<32u;i++) if(raf_gate(i)!=raf_gate(0)){same=0;break;}
    _T(!same,                           "B10:gate_not_constant");

    /* B11: pipeline end-to-end sem crash (3 tokens) */
    TraceStats ts_null; memset(&ts_null,0,sizeof(ts_null)); ts_null.delta_p=0.1f;
    CatalogCtx cc_null; memset(&cc_null,0,sizeof(cc_null)); cc_null.cluster=0;
    /* redireciona stdout temporariamente */
    FILE *devnull=fopen("/dev/null","w");
    if(devnull){
        FILE *saved=stdout; stdout=devnull;
        run_pipeline("deus",3u,&ts_null,&cc_null);
        stdout=saved; fclose(devnull);
        _T(1u,                          "B11:pipeline_no_crash");
    } else {
        _T(0u,                          "B11:pipeline_no_crash");
    }

    printf("\n  pass=%u  fail=%u  %s\n",
           G_PASS, G_FAIL, G_FAIL==0?"★ TODOS PASSARAM":"✗ FALHAS");
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §11 BENCHMARK
 * ═══════════════════════════════════════════════════════════════════════════ */
static void run_bench(const u32 *ids, u32 n_ids) {
    printf("\n[bench] pipeline ARM32 metas:\n");
    printf("  gate_step target : < 5 ns/op\n");
    printf("  compute_temp     : < 10 ns/op\n");
    printf("  model_sample     : < 50 µs/token\n");

    struct timespec t0,t1;
    #define NS(a,b) ((b.tv_sec-a.tv_sec)*1000000000LL+(b.tv_nsec-a.tv_nsec))

    /* gate */
    volatile u32 sink=0;
    clock_gettime(CLOCK_MONOTONIC,&t0);
    for(u32 i=0;i<100000u;i++) sink^=raf_gate(i);
    clock_gettime(CLOCK_MONOTONIC,&t1);
    printf("  gate_step/op     : %lld ns\n", NS(t0,t1)/100000LL);

    /* compute_temp */
    volatile f32 fsink=0.f;
    clock_gettime(CLOCK_MONOTONIC,&t0);
    for(u32 i=0;i<100000u;i++) fsink+=compute_temp(0.1f,i%8u,i%6u);
    clock_gettime(CLOCK_MONOTONIC,&t1);
    printf("  compute_temp/op  : %lld ns\n", NS(t0,t1)/100000LL);

    /* model_sample */
    u32 rng=42u, n=500u;
    if(n_ids<n) n=n_ids>0?n_ids:1u;
    clock_gettime(CLOCK_MONOTONIC,&t0);
    for(u32 i=0;i<n;i++) sink^=model_sample(ids[i%n_ids],0.8f,&rng);
    clock_gettime(CLOCK_MONOTONIC,&t1);
    printf("  model_sample/tok : %lld µs\n", NS(t0,t1)/(n*1000LL));
    (void)sink; (void)fsink;
    #undef NS
}
