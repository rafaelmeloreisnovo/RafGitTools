/* ═══════════════════════════════════════════════════════════════════════════
 * §4 — CRC32C BRANCHLESS
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 crc32c(const u8 *p, u32 n){
  u32 c = 0xFFFFFFFFu;
  for(u32 i=0;i<n;i++){
    c ^= (u32)p[i];
    for(u32 j=0;j<8u;j++) c = (c>>1) ^ (0x82F63B78u & -(c&1u));
  }
  return ~c;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §5 — ARENA com CANARY + MEMORY POISON
 * ═══════════════════════════════════════════════════════════════════════════ */
#define ARENA_SZ      (32u*1024u*1024u)
#define CANARY_MAGIC  0xDEADC0DEu
#define POISON_BYTE   0xDEu
#define MAX_ALLOCS    2048u

typedef struct { u32 offset, size, canary_pre, canary_post; } AllocRec;
static u8 G_ARENA[ARENA_SZ] __attribute__((aligned(64)));
static u32 G_ABUMP = 0u;
static AllocRec G_ALLOC_RECS[MAX_ALLOCS];
static u32 G_N_ALLOCS = 0u;

static void *raf_alloc(u32 n){
  if(n == 0) return (void*)0;
  u32 s = (G_ABUMP + 63u) & ~63u;
  u32 e = s + 4u + n + 4u;
  if(e > ARENA_SZ || G_N_ALLOCS >= MAX_ALLOCS) return (void*)0;
  u32 *pre  = (u32*)(G_ARENA + s);
  u32 *post = (u32*)(G_ARENA + s + 4u + n);
  *pre = CANARY_MAGIC; *post = CANARY_MAGIC;
  G_ALLOC_RECS[G_N_ALLOCS].offset = s;
  G_ALLOC_RECS[G_N_ALLOCS].size = n;
  G_ALLOC_RECS[G_N_ALLOCS].canary_pre = s;
  G_ALLOC_RECS[G_N_ALLOCS].canary_post = s + 4u + n;
  G_N_ALLOCS++; G_ABUMP = e;
  u8 *data = G_ARENA + s + 4u;
  for(u32 i=0;i<n;i++) data[i] = POISON_BYTE;
  return data;
}

static RafErr raf_check_canaries(void){
  for(u32 i=0; i<G_N_ALLOCS; i++){
    u32 *pre  = (u32*)(G_ARENA + G_ALLOC_RECS[i].canary_pre);
    u32 *post = (u32*)(G_ARENA + G_ALLOC_RECS[i].canary_post);
    if(*pre != CANARY_MAGIC || *post != CANARY_MAGIC) return RAF_ERR_CANARY;
  }
  return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §6 — ATOMIC + BARRIERS
 * ═══════════════════════════════════════════════════════════════════════════ */
static void raf_barrier(void){
  __asm__ volatile("" ::: "memory");
#if defined(__aarch64__)
  __asm__ volatile("dmb ish" ::: "memory");
#elif defined(__arm__)
  __asm__ volatile("dmb" ::: "memory");
#elif defined(__x86_64__) || defined(__i386__)
  __asm__ volatile("mfence" ::: "memory");
#endif
}
static u32 raf_cas32(volatile u32 *p, u32 e, u32 d){
  return __sync_bool_compare_and_swap(p,e,d) ? 1u : 0u;
}
static u32 raf_atomic_add(volatile u32 *p, u32 v){ return __sync_fetch_and_add(p,v); }
static u64 raf_atomic_add64(volatile u64 *p, u64 v){ return __sync_fetch_and_add(p,v); }
static u32 raf_load_acquire(volatile u32 *p){ u32 v=*p; raf_barrier(); return v; }
static void raf_store_release(volatile u32 *p, u32 v){ raf_barrier(); *p = v; }

/* Cache prefetch hint (T11) — branchless, sem efeito se arch não suporta */
static void raf_prefetch(const void *p){
#if defined(__GNUC__) || defined(__clang__)
  __builtin_prefetch(p, 0, 3);  /* read, high temporal locality */
#else
  (void)p;
#endif
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §7 — SAFE MATH
 * ═══════════════════════════════════════════════════════════════════════════ */
static RafErr raf_add_u32(u32 a, u32 b, u32 *out){
  u32 r = a + b; if(r < a) return RAF_ERR_OVERFLOW;
  *out = r; return RAF_OK;
}
static RafErr raf_mul_u32(u32 a, u32 b, u32 *out){
  if(!a||!b){ *out = 0; return RAF_OK; }
  u64 r = (u64)a*(u64)b;
  if(r > 0xFFFFFFFFull) return RAF_ERR_OVERFLOW;
  *out = (u32)r; return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §8 — TIMING-SAFE COMPARE
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 raf_ct_eq(const void *a, const void *b, u32 n){
  const u8 *x=a, *y=b; u32 d=0;
  for(u32 i=0;i<n;i++) d |= (u32)(x[i]^y[i]);
  return (d - 1u) >> 31;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §9 — STRING SAFE
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 raf_strlcpy(char *dst, const char *src, u32 dst_size){
  u32 src_len = _slen(src);
  if(dst_size == 0) return src_len;
  u32 copy = (src_len < dst_size - 1u) ? src_len : dst_size - 1u;
  _mcpy(dst, src, copy); dst[copy] = 0;
  return src_len;
}

  w->kicked = 0u; w->pet_count = 0u;

  if(elapsed > w->timeout_ns){ raf_store_release(&w->kicked, 1u); return RAF_ERR_WATCHDOG; }
  return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §12 — CIRCUIT BREAKER (U05)
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef enum { CB_CLOSED=0, CB_OPEN=1, CB_HALF_OPEN=2 } CBState;
typedef struct {
  volatile u32 state, fail_count;
  volatile u64 opened_at_ns;
  u32 fail_threshold; u64 cooldown_ns;
} CircuitBreaker;
static void raf_cb_init(CircuitBreaker *cb, u32 threshold, u32 cooldown_ms){
  cb->state = CB_CLOSED; cb->fail_count = 0; cb->opened_at_ns = 0;
  cb->fail_threshold = threshold; cb->cooldown_ns = (u64)cooldown_ms*1000000ULL;
}
static RafErr raf_cb_allow(CircuitBreaker *cb){
  u32 s = raf_load_acquire(&cb->state);
  if(s == CB_OPEN){
    if(_now_ns() - cb->opened_at_ns >= cb->cooldown_ns){
      raf_cas32(&cb->state, CB_OPEN, CB_HALF_OPEN); return RAF_OK;
    }
    return RAF_ERR_BREAKER;
  }
  return RAF_OK;
}
static void raf_cb_success(CircuitBreaker *cb){
  cb->fail_count = 0; raf_store_release(&cb->state, CB_CLOSED);
}
static void raf_cb_failure(CircuitBreaker *cb){
  u32 fc = raf_atomic_add(&cb->fail_count, 1u) + 1u;
  if(fc >= cb->fail_threshold){
    cb->opened_at_ns = _now_ns();
    raf_store_release(&cb->state, CB_OPEN);
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §13 — ENTROPY POOL
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { u64 pool[8]; u32 mix_count; } EntropyPool;
static void raf_entropy_mix(EntropyPool *p, u64 sample){
  u32 idx = p->mix_count & 7u;
  u64 v = p->pool[idx] ^ sample;
  v = (v<<13)|(v>>51); v *= 0x9E3779B97F4A7C15ULL;
  p->pool[idx] = v; p->pool[(idx+1)&7u] ^= (v >> 17);
  p->mix_count++;
}
static void raf_entropy_seed(EntropyPool *p){
  _mset(p, 0, sizeof(*p));
  raf_entropy_mix(p, _now_ns());
  u64 sa; raf_entropy_mix(p, (u64)(unsigned long)&sa);
  raf_entropy_mix(p, (u64)(unsigned long)&raf_entropy_seed);
  raf_entropy_mix(p, _now_ns());
  raf_entropy_mix(p, (u64)(unsigned long)G_ARENA);
}
static u64 raf_entropy_extract(EntropyPool *p){
  u64 r = 0;
  for(u32 i=0;i<8u;i++){
    r ^= p->pool[i]; r = (r<<7)|(r>>57); r *= 0xBF58476D1CE4E5B9ULL;
  }
  raf_entropy_mix(p, r); return r;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §14 — REPLAY PROTECTION
 * ═══════════════════════════════════════════════════════════════════════════ */
static volatile u64 G_NONCE_COUNTER = 1;
static volatile u64 G_LAST_SEEN_NONCE = 0;
static u64 raf_nonce_get(void){ return raf_atomic_add64(&G_NONCE_COUNTER, 1ULL); }

    if(nonce <= last) return RAF_ERR_REPLAY;
    if(__sync_bool_compare_and_swap(&G_LAST_SEEN_NONCE, last, nonce)) return RAF_OK;
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §16 — HISTOGRAM
 * ═══════════════════════════════════════════════════════════════════════════ */
#define HIST_BUCKETS 32u
typedef struct {
  volatile u32 buckets[HIST_BUCKETS];
  volatile u64 sum_ns;
  volatile u32 count;
} Histogram;
static u32 raf_log2_u32(u32 v){
  if(v == 0) return 0;
  u32 r = 0; while(v >>= 1) r++; return r;
}
static void raf_hist_record(Histogram *h, u64 value_ns){
  u32 idx = raf_log2_u32((u32)(value_ns > 0xFFFFFFFFull ? 0xFFFFFFFFu : value_ns));
  if(idx >= HIST_BUCKETS) idx = HIST_BUCKETS - 1u;
  raf_atomic_add(&h->buckets[idx], 1u);
  raf_atomic_add64(&h->sum_ns, value_ns);
  raf_atomic_add(&h->count, 1u);
}
static u64 raf_hist_percentile(const Histogram *h, u32 percent){
  u32 total = h->count;
  if(total == 0) return 0;
  u64 target = ((u64)total * (u64)percent) / 100ULL;
  u64 cum = 0;
  for(u32 i=0;i<HIST_BUCKETS;i++){
    cum += h->buckets[i];
    if(cum >= target) return (1ULL << i) + (1ULL << (i > 0 ? i-1 : 0));
  }
  return 1ULL << (HIST_BUCKETS - 1u);
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §17 — POOL ALLOCATOR (U10)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define POOL_OBJ_SIZE 256u
#define POOL_N_OBJS   64u
typedef struct {
  u8 slots[POOL_N_OBJS][POOL_OBJ_SIZE] __attribute__((aligned(64)));
  u32 free_list[POOL_N_OBJS];
  volatile u32 free_head;
  u32 init_done;
} ObjPool;
static ObjPool G_POOL;
static void raf_pool_init(ObjPool *p){
  for(u32 i=0; i<POOL_N_OBJS-1; i++) p->free_list[i] = i + 1;
  p->free_list[POOL_N_OBJS-1] = 0xFFFFFFFFu;
  p->free_head = 0; p->init_done = 1;
}
static void *raf_pool_get(ObjPool *p){
  if(!p->init_done) raf_pool_init(p);
  for(;;){
    u32 head = raf_load_acquire(&p->free_head);
    if(head == 0xFFFFFFFFu) return (void*)0;
    u32 next = p->free_list[head];
    if(raf_cas32(&p->free_head, head, next)){
      _mset(p->slots[head], POISON_BYTE, POOL_OBJ_SIZE);
      return p->slots[head];
    }
  }
}
static void raf_pool_put(ObjPool *p, void *obj){
  i64 diff = (u8*)obj - (u8*)p->slots;
  if(diff < 0 || diff >= (i64)(POOL_N_OBJS * POOL_OBJ_SIZE)) return;
  u32 idx = (u32)(diff / POOL_OBJ_SIZE);
  _mset(p->slots[idx], POISON_BYTE, POOL_OBJ_SIZE);
  for(;;){
    u32 head = raf_load_acquire(&p->free_head);
    p->free_list[idx] = head;
    if(raf_cas32(&p->free_head, head, idx)) return;
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §18 — BLOOM FILTER (U11)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define BLOOM_BITS 8192u
#define BLOOM_HASHES 3u
typedef struct { u8 bits[BLOOM_BITS/8u]; volatile u32 count; } Bloom;
static u32 raf_bloom_hash(const u8 *data, u32 len, u32 seed){
  u32 h = 0x811C9DC5u ^ seed;
  for(u32 i=0;i<len;i++){ h ^= (u32)data[i]; h *= 0x01000193u; }
  return h % BLOOM_BITS;
}
static void raf_bloom_add(Bloom *b, const u8 *data, u32 len){
  for(u32 s=0; s<BLOOM_HASHES; s++){
    u32 bit = raf_bloom_hash(data, len, s * 0xDEADBEEFu);
    b->bits[bit/8u] |= (u8)(1u << (bit & 7u));
  }
  raf_atomic_add(&b->count, 1u);
}
static u32 raf_bloom_contains(const Bloom *b, const u8 *data, u32 len){
  for(u32 s=0; s<BLOOM_HASHES; s++){
    u32 bit = raf_bloom_hash(data, len, s * 0xDEADBEEFu);
    if(!(b->bits[bit/8u] & (u8)(1u << (bit & 7u)))) return 0u;
  }
  return 1u;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §19 — STRUCTURED LOG + AUDIT HMAC CHAIN
 * ═══════════════════════════════════════════════════════════════════════════ */
#define LOG_RING 256u
#define LOG_MSG_MAX 64u
#define LOG_FIELDS_MAX 4u
#define LOG_FK_MAX 16u
#define LOG_FV_MAX 32u
#define LOG_MAGIC 0x4C4F4733u  /* LOG3 */

typedef enum { LVL_TRACE=0, LVL_DEBUG, LVL_INFO, LVL_WARN, LVL_ERROR, LVL_FATAL } LogLevel;
typedef struct { char key[LOG_FK_MAX]; char val[LOG_FV_MAX]; } LogField;
typedef struct {
  u32 magic; u64 ts_ns; u32 level; CorrId corr_id;
  char msg[LOG_MSG_MAX];
  u32 n_fields; LogField fields[LOG_FIELDS_MAX];
  u32 crc;
} LogEntry;
static LogEntry G_LOG[LOG_RING];
static volatile u32 G_LOG_HEAD = 0;
static volatile u32 G_LOG_COUNT = 0;

static u32 G_AUDIT_KEY[8] = {
  0x52414641u, 0xDEADBEEFu, 0xCAFEBABEu, 0x12345678u,
  0xFEEDFACEu, 0xBADC0DEDu, 0xABCDEF01u, 0x23456789u
};
static volatile u32 G_AUDIT_CHAIN = 0xFFFFFFFFu;

static u32 raf_audit_hmac(const u8 *data, u32 len, u32 prev){
  u32 h = prev;
  for(u32 i=0;i<len;i++){
    u8 b = data[i] ^ ((u8)G_AUDIT_KEY[i & 7u]);
    h ^= (u32)b;
    for(u32 j=0;j<8u;j++) h = (h>>1) ^ (0x82F63B78u & -(h&1u));
  }
  return ~h;
}
static u32 raf_audit_chain(void){ return G_AUDIT_CHAIN; }
static u32 raf_audit_verify(void){
  u32 chain = 0xFFFFFFFFu;
  u32 n = G_LOG_COUNT < LOG_RING ? G_LOG_COUNT : LOG_RING;
  u32 ok = 0;
  for(u32 i=0; i<n; i++){
    if(G_LOG[i].magic != LOG_MAGIC) continue;
    u32 expected = raf_audit_hmac((const u8*)&G_LOG[i], sizeof(LogEntry) - 4u, chain);
    if(expected == G_LOG[i].crc) ok++;
    chain = expected;
  }
  return ok;
}
static void raf_log_with(LogLevel lvl, const CorrId *cid, const char *msg,
                          const char *k1, const char *v1, const char *k2, const char *v2){
  u32 idx;
  for(;;){
    u32 old = raf_load_acquire(&G_LOG_HEAD);
    u32 new_h = (old + 1u) % LOG_RING;
    if(raf_cas32(&G_LOG_HEAD, old, new_h)){ idx = old; break; }
  }
  LogEntry *e = &G_LOG[idx];
  _mset(e, 0, sizeof(*e));
  e->magic = LOG_MAGIC; e->ts_ns = _now_ns(); e->level = (u32)lvl;
  if(cid) e->corr_id = *cid;
  raf_strlcpy(e->msg, msg, LOG_MSG_MAX);
  if(k1 && v1){
    raf_strlcpy(e->fields[0].key, k1, LOG_FK_MAX);
    raf_strlcpy(e->fields[0].val, v1, LOG_FV_MAX);
    e->n_fields = 1;
  }
  if(k2 && v2){
    raf_strlcpy(e->fields[1].key, k2, LOG_FK_MAX);
    raf_strlcpy(e->fields[1].val, v2, LOG_FV_MAX);
    e->n_fields = 2;
  }
  u32 prev = raf_load_acquire(&G_AUDIT_CHAIN);
  e->crc = raf_audit_hmac((const u8*)e, sizeof(LogEntry) - 4u, prev);
  raf_store_release(&G_AUDIT_CHAIN, e->crc);
  raf_atomic_add(&G_LOG_COUNT, 1u);
}
static void raf_log(LogLevel lvl, const char *msg){
  raf_log_with(lvl, (void*)0, msg, (void*)0, (void*)0, (void*)0, (void*)0);
}

typedef struct {
  u32 capacity, refill_per_sec;
  volatile u32 tokens; volatile u64 last_refill_ns;
} RateLimiter;
static void raf_rl_init(RateLimiter *rl, u32 cap, u32 refill){
  rl->capacity = cap; rl->refill_per_sec = refill;
  rl->tokens = cap; rl->last_refill_ns = _now_ns();
}
static RafErr raf_rl_try(RateLimiter *rl){
  u64 now = _now_ns();
  u64 elapsed = now - rl->last_refill_ns;
  u32 add = (u32)((elapsed * (u64)rl->refill_per_sec) / 1000000000ULL);
  if(add > 0){
    u32 nt = rl->tokens + add;
    if(nt > rl->capacity) nt = rl->capacity;
    rl->tokens = nt; rl->last_refill_ns = now;
  }
  for(;;){
    u32 cur = raf_load_acquire(&rl->tokens);
    if(cur == 0) return RAF_ERR_RATELIM;
    if(raf_cas32(&rl->tokens, cur, cur - 1)) return RAF_OK;
  }
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §25 — [TOTAL T04] 7-WAY ARCHITECTURAL DIRECTIONS
 *   Cada token/operação é mapeado em 7 direções semânticas independentes:
 *     1. FORMAL/AXIOMÁTICA  — invariantes lógicos
 *     2. COMPUTACIONAL      — cost matrix, complexidade
 *     3. GEOMÉTRICA/TOPOL   — distância no toro T^7
 *     4. INFORMACIONAL      — entropia, surprise
 *     5. LINGUÍSTICA        — lexema, fonema
 *     6. SISTÊMICA          — flow, dependências
 *     7. ÉTICA/VALIDAÇÃO    — coerência com Φ_ethica
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef enum {
  DIR_FORMAL = 0, DIR_COMPUT = 1, DIR_GEOM = 2, DIR_INFO = 3,
  DIR_LING = 4, DIR_SYS = 5, DIR_ETHICA = 6, DIR_N = 7
} DirIdx;

typedef struct {
  i32 scores[DIR_N];  /* Q16: cada direção contribui um score */
  u32 magic;          /* 0x44495237 = "DIR7" */
} DirVector;

#define DIR7_MAGIC 0x44495237u

static void raf_dir7_init(DirVector *d){
  _mset(d, 0, sizeof(*d));
  d->magic = DIR7_MAGIC;
}
/* Combina dois vetores com pesos w1, w2 em Q16 (ambos somam 1) */
static void raf_dir7_combine(DirVector *out, const DirVector *a,
                              const DirVector *b, i32 w1_q16, i32 w2_q16){
  for(u32 i=0;i<DIR_N;i++){
    i64 v = ((i64)a->scores[i] * w1_q16 + (i64)b->scores[i] * w2_q16) >> 16;
    out->scores[i] = (i32)v;
  }
  out->magic = DIR7_MAGIC;
}
/* Soma absoluta de divergências entre dois vetores */
static u32 raf_dir7_divergence(const DirVector *a, const DirVector *b){
  u32 sum = 0;
  for(u32 i=0;i<DIR_N;i++){
    i32 d = a->scores[i] - b->scores[i];
    sum += (u32)(d < 0 ? -d : d);
  }
  return sum;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §26 — [TOTAL T03] RELATIONAL TENSOR 8×5×7×3 = 840
 *
 * Tensor multidirecional do mapa visual (Image 8 — atenção octogonal):
 *   A: 8 direções (FOCO/PERCEP/MEMOR/INTUI/EMOC/VOL/IMAG/EXEC) × 5 estados
 *   B: 7 direções semânticas (§25) × 3 cadeias (A/B/C — Image 1)
 *
 * Cardinalidades:
 *   |A| = 8×5 = 40
 *   |B| = 7×3 = 21
 *   |pares_A| = C(40,2) = 780
 *   |pares_B| = C(21,2) = 210
 *   |cruzamento| = 40 × 21 = 840
 *   |pares_de_pares| = 780 × 210 = 163800
 *
 * Bloco 2×2 adjacente em A: 7 × 4 = 28
 * Bloco 2×2 adjacente em B: 6 × 2 = 12
 * Permutações por bloco: 4! = 24
 *
 *   A_2x2_adj_perm = 28 × 24 = 672
 *   B_2x2_adj_perm = 12 × 24 = 288
 * ═══════════════════════════════════════════════════════════════════════════ */
#define AREL_A_DIRS 8u   /* 8 direções de atenção octogonal */
#define AREL_A_STATES 5u
#define AREL_B_DIRS 7u
#define AREL_B_CHAINS 3u
#define AREL_A_TOTAL (AREL_A_DIRS * AREL_A_STATES)        /* 40 */
#define AREL_B_TOTAL (AREL_B_DIRS * AREL_B_CHAINS)        /* 21 */
#define AREL_CROSS   (AREL_A_TOTAL * AREL_B_TOTAL)        /* 840 */

typedef struct {
  i16 axb[AREL_A_TOTAL][AREL_B_TOTAL];  /* tensor Q8.8 */
  u32 magic;                              /* 0x54454E73 = "TENs" */
  u32 crc;
} RelTensor;

#define ARTENS_MAGIC 0x54454E73u

static u32 raf_tensor_idx_a(u32 dir, u32 state){
  return dir * AREL_A_STATES + state;
}
static u32 raf_tensor_idx_b(u32 dir, u32 chain){
  return dir * AREL_B_CHAINS + chain;
}
static void raf_tensor_init(RelTensor *t){
  _mset(t, 0, sizeof(*t));
  t->magic = ARTENS_MAGIC;
  t->crc = crc32c((const u8*)t->axb, sizeof(t->axb));
}
static RafErr raf_tensor_set(RelTensor *t, u32 a, u32 b, i16 v){
  if(a >= AREL_A_TOTAL || b >= AREL_B_TOTAL) return RAF_ERR_RANGE;
  t->axb[a][b] = v;
  t->crc = crc32c((const u8*)t->axb, sizeof(t->axb));
  return RAF_OK;
}
static i16 raf_tensor_get(const RelTensor *t, u32 a, u32 b){
  if(a >= AREL_A_TOTAL || b >= AREL_B_TOTAL) return 0;
  return t->axb[a][b];
}
static u32 raf_tensor_verify_crc(const RelTensor *t){
  return crc32c((const u8*)t->axb, sizeof(t->axb)) == t->crc;
}

/* Operador final do mapa (Image 8 sintético):
 *   Rel(x,y) = w1·correlação + w2·informação_mutua + w3·entropia
 *            + w4·geometria + w5·causalidade + w6·campo
 * Aqui versão Q16 simplificada para inteiros: combinação pesada
 * de 6 componentes acumulados sobre o tensor. */
static i32 raf_tensor_rel_op(const RelTensor *t, u32 a, u32 b,
                              i32 w1, i32 w2, i32 w3,
                              i32 w4, i32 w5, i32 w6){
  i16 v = raf_tensor_get(t, a, b);
  /* Correlação = valor direto */
  i64 corr = (i64)v * w1;
  /* IM aprox = abs(v) */
  i64 mi   = (i64)(v < 0 ? -v : v) * w2;
  /* Entropia aprox = log2(|v|+1) deslocado */
  i64 ent  = (i64)raf_log2_u32((u32)(v<0?-v:v) + 1u) * w3;
  /* Geometria = distância no eixo A (mod 8) */
  u32 a_dir = a / AREL_A_STATES;
  i64 geom = (i64)((a_dir + AREL_A_DIRS/2u) % AREL_A_DIRS) * w4;
  /* Causalidade = b_chain (cadeia A/B/C tem ordens distintas) */
  u32 b_chain = b % AREL_B_CHAINS;
  i64 caus = (i64)(b_chain + 1) * w5;
  /* Campo = soma de vizinhos imediatos */
  i64 field = 0;
  if(a + 1 < AREL_A_TOTAL) field += t->axb[a+1][b];
  if(b + 1 < AREL_B_TOTAL) field += t->axb[a][b+1];
  field *= w6;
  /* Combina em Q16 */
  i64 total = (corr + mi + ent + geom + caus + field) >> 16;
  return (i32)total;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §27 — [TOTAL T01] PHONETIC PHONEME TABLE multilíngue
 *
 * Cada fonema = símbolo IPA-like + entrada de língua + features articulatórios.
 * Línguas suportadas: PT, EN, HE (hebraico), AR (aramaico), EL (grego),
 *                     JP (japonês), CN (mandarim).
 * Feature bits (8 flags em u8):
 *   bit0 voiced
 *   bit1 nasal
 *   bit2 stop
 *   bit3 fricative
 *   bit4 approximant
 *   bit5 vowel
 *   bit6 high (vowel) / strident (cons)
 *   bit7 long
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef enum {
  LANG_PT = 0, LANG_EN, LANG_HE, LANG_AR, LANG_EL, LANG_JP, LANG_CN, LANG_N
} LangIdx;

#define PHON_VOICED      0x01
#define PHON_NASAL       0x02
#define PHON_STOP        0x04
#define PHON_FRICATIVE   0x08
#define PHON_APPROX      0x10
#define PHON_VOWEL       0x20
#define PHON_HIGH        0x40
#define PHON_LONG        0x80

typedef struct {
  char  ipa[8];        /* símbolo IPA (UTF-8 cabe; truncado para 7 bytes + 0) */
  u8    lang;          /* LangIdx */
  u8    features;      /* bitmask */
  u16   freq_hz;       /* frequência formante F1 aproximada (vogais) ou 0 */
} Phoneme;

/* Tabela canônica — núcleo mínimo cobrindo as 7 línguas */
static const Phoneme PHONEME_TABLE[] = {
  /* Português brasileiro */
  {"a",       LANG_PT, PHON_VOWEL,                  730},
  {"e",       LANG_PT, PHON_VOWEL,                  530},
  {"i",       LANG_PT, PHON_VOWEL | PHON_HIGH,      290},
  {"o",       LANG_PT, PHON_VOWEL,                  570},
  {"u",       LANG_PT, PHON_VOWEL | PHON_HIGH,      300},
  {"b",       LANG_PT, PHON_VOICED | PHON_STOP,     0},
  {"p",       LANG_PT, PHON_STOP,                   0},
  {"m",       LANG_PT, PHON_VOICED | PHON_NASAL,    0},
  {"r",       LANG_PT, PHON_VOICED | PHON_APPROX,   0},
  {"s",       LANG_PT, PHON_FRICATIVE,              0},
  /* English */
  {"æ",       LANG_EN, PHON_VOWEL,                  660},
  {"ʃ",       LANG_EN, PHON_FRICATIVE,              0},
  {"θ",       LANG_EN, PHON_FRICATIVE,              0},
  {"ð",       LANG_EN, PHON_VOICED | PHON_FRICATIVE,0},
  /* Hebraico (transliteração) */
  {"א",       LANG_HE, PHON_VOWEL,                  0},
  {"ב",       LANG_HE, PHON_VOICED | PHON_STOP,     0},
  {"ש",       LANG_HE, PHON_FRICATIVE,              0},
  {"ח",       LANG_HE, PHON_FRICATIVE,              0},
  {"ע",       LANG_HE, PHON_VOICED | PHON_FRICATIVE,0},
  /* Aramaico (compartilha alfabeto com hebraico clássico) */
  {"ܐ",       LANG_AR, PHON_VOWEL,                  0},
  {"ܒ",       LANG_AR, PHON_VOICED | PHON_STOP,     0},
  /* Grego */
  {"α",       LANG_EL, PHON_VOWEL,                  730},
  {"β",       LANG_EL, PHON_VOICED | PHON_FRICATIVE,0},
  {"χ",       LANG_EL, PHON_FRICATIVE,              0},
  {"ω",       LANG_EL, PHON_VOWEL,                  570},
  /* Japonês (rōmaji) */
  {"あ",      LANG_JP, PHON_VOWEL,                  730},
  {"い",      LANG_JP, PHON_VOWEL | PHON_HIGH,      290},
  {"う",      LANG_JP, PHON_VOWEL | PHON_HIGH,      300},
  {"ん",      LANG_JP, PHON_VOICED | PHON_NASAL,    0},
  /* Mandarim (pinyin) — tons codificados via PHON_LONG para tom alto */
  {"ma1",     LANG_CN, PHON_VOWEL | PHON_LONG,      730},
  {"ma2",     LANG_CN, PHON_VOWEL,                  680},
  {"ma3",     LANG_CN, PHON_VOWEL,                  600},
  {"ma4",     LANG_CN, PHON_VOWEL,                  500},
};
#define PHONEME_N (sizeof(PHONEME_TABLE) / sizeof(PHONEME_TABLE[0]))

/* Distância fonética entre dois fonemas (Hamming sobre features) */
static u32 raf_phon_distance(u32 i, u32 j){
  if(i >= PHONEME_N || j >= PHONEME_N) return 0xFFu;
  u8 xor_features = (u8)(PHONEME_TABLE[i].features ^ PHONEME_TABLE[j].features);
  u32 dist = 0;
  for(u32 b=0;b<8u;b++) dist += (xor_features >> b) & 1u;
  /* penalidade adicional se línguas diferem (não 0 se mesma raiz semita) */
  if(PHONEME_TABLE[i].lang != PHONEME_TABLE[j].lang){
    u8 la = PHONEME_TABLE[i].lang, lb = PHONEME_TABLE[j].lang;
    /* HE e AR compartilham raiz semita: penalidade menor */
    if((la == LANG_HE && lb == LANG_AR) || (la == LANG_AR && lb == LANG_HE)){
      dist += 1;
    } else {
      dist += 3;
    }
  }
  return dist;
}

/* Busca fonema por símbolo IPA (linear; ok pois tabela pequena) */
static i32 raf_phon_find(const char *ipa, u32 lang){
  for(u32 i=0;i<PHONEME_N;i++){
    if(PHONEME_TABLE[i].lang != lang) continue;
    /* compara até 7 bytes (campo ipa) */
    u32 match = 1;
    for(u32 j=0;j<7u;j++){
      if(PHONEME_TABLE[i].ipa[j] != ipa[j]){ match = 0; break; }
      if(!ipa[j]) break;
    }
    if(match) return (i32)i;
  }
  return -1;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §28 — [TOTAL T02] VECTOR DB com busca k-NN
 *
 * Banco vetorial ultra-eficiente: vetores de 8 floats (Q16) em arena estática.
 * Busca brute-force com prefetch + SIMD opcional (NEON/SSE).
 * Distância euclidiana ao quadrado (sem sqrt no caminho quente).
 * ═══════════════════════════════════════════════════════════════════════════ */
#define VDB_DIM 8u
#define VDB_CAPACITY 1024u

typedef struct {
  i32 v[VDB_DIM];   /* Q16 fixed-point */
  u32 id;
  u8  tag;          /* cluster ou categoria */
  u8  _pad[3];
} VecEntry;

typedef struct {
  VecEntry entries[VDB_CAPACITY] __attribute__((aligned(64)));
  volatile u32 count;
  u32 magic;        /* 0x56444231 = "VDB1" */
  u32 crc;
} VectorDB;

#define VDB_MAGIC 0x56444231u

static VectorDB G_VDB;

static void raf_vdb_init(VectorDB *db){
  _mset(db, 0, sizeof(*db));
  db->magic = VDB_MAGIC;
}

static RafErr raf_vdb_insert(VectorDB *db, const i32 v[VDB_DIM], u32 id, u8 tag){
  u32 idx = raf_atomic_add(&db->count, 1u);
  if(idx >= VDB_CAPACITY){
    raf_atomic_add(&db->count, (u32)-1);  /* roll back */
    return RAF_ERR_OOM;
  }
  for(u32 i=0;i<VDB_DIM;i++) db->entries[idx].v[i] = v[i];
  db->entries[idx].id = id;
  db->entries[idx].tag = tag;
  return RAF_OK;
}

/* Distância L2 ao quadrado em Q32 (sem sqrt) */
static u64 raf_vdb_dist2(const i32 *a, const i32 *b){
  u64 sum = 0;
  for(u32 i=0;i<VDB_DIM;i++){
    i64 d = (i64)a[i] - (i64)b[i];
    sum += (u64)(d * d);
  }
  return sum;
}

/* Busca k-NN brute-force com prefetch.
 * out_ids: array de k u32 (ids) ordenados por distância crescente.
 * out_dists: distâncias correspondentes. */
static u32 raf_vdb_knn(const VectorDB *db, const i32 *query, u32 k,
                       u32 *out_ids, u64 *out_dists){
  if(k == 0) return 0;
  /* Inicializa com infinitos */
  for(u32 i=0;i<k;i++){ out_ids[i] = 0xFFFFFFFFu; out_dists[i] = 0xFFFFFFFFFFFFFFFFULL; }
  u32 n = db->count;
  for(u32 i=0;i<n;i++){
    /* prefetch próximo entry para warmup do L1 */
    if(i + 1 < n) raf_prefetch(&db->entries[i+1]);
    u64 d = raf_vdb_dist2(db->entries[i].v, query);
    /* insertion sort no topo-k (k pequeno → eficiente) */
    if(d < out_dists[k-1]){
      u32 pos = k - 1;
      while(pos > 0 && out_dists[pos-1] > d){
        out_dists[pos] = out_dists[pos-1];
        out_ids[pos] = out_ids[pos-1];
        pos--;
      }
      out_dists[pos] = d;
      out_ids[pos] = db->entries[i].id;
    }
  }
  return n < k ? n : k;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §29 — [TOTAL T05] ROLLBACK CHECKPOINT
 *   Salva snapshot do estado crítico (arena bump, log head, audit chain).
 *   Restaura em caso de falha — implementação puramente in-memory.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u32 arena_bump;
  u32 n_allocs;
  u32 log_head;
  u32 log_count;
  u32 audit_chain;
  u64 nonce_counter;
  u32 magic;        /* "CKPT" */
  u32 crc;
} Checkpoint;
#define CKPT_MAGIC 0x434B5054u

static Checkpoint G_CKPTS[4];
static volatile u32 G_N_CKPTS = 0;

static RafErr raf_ckpt_save(u32 slot){
  if(slot >= 4) return RAF_ERR_RANGE;
  Checkpoint *c = &G_CKPTS[slot];
  c->arena_bump   = G_ABUMP;
  c->n_allocs     = G_N_ALLOCS;
  c->log_head     = G_LOG_HEAD;
  c->log_count    = G_LOG_COUNT;
  c->audit_chain  = G_AUDIT_CHAIN;
  c->nonce_counter= G_NONCE_COUNTER;
  c->magic        = CKPT_MAGIC;
  c->crc          = crc32c((const u8*)c, sizeof(Checkpoint) - 4u);
  raf_log_with(LVL_INFO, (void*)0, "ckpt_saved", "slot", "x", (void*)0, (void*)0);
  return RAF_OK;
}

static RafErr raf_ckpt_restore(u32 slot){
  if(slot >= 4) return RAF_ERR_RANGE;
  Checkpoint *c = &G_CKPTS[slot];
  if(c->magic != CKPT_MAGIC) return RAF_ERR_NOTFOUND;
  u32 expected = crc32c((const u8*)c, sizeof(Checkpoint) - 4u);
  if(expected != c->crc) return RAF_ERR_CRC;
  G_ABUMP         = c->arena_bump;
  G_N_ALLOCS      = c->n_allocs;
  G_LOG_HEAD      = c->log_head;
  G_LOG_COUNT     = c->log_count;
  G_AUDIT_CHAIN   = c->audit_chain;
  G_NONCE_COUNTER = c->nonce_counter;
  raf_log(LVL_WARN, "ckpt_restored");
  return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §30 — [TOTAL T06] FAILSAFE: degradação controlada
 *   Modos: NORMAL → DEGRADED (sem chaos, sem audit) → READ_ONLY → EMERGENCY
 *   Cada modo desativa features não essenciais.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef enum {
  FS_NORMAL = 0, FS_DEGRADED = 1, FS_READ_ONLY = 2, FS_EMERGENCY = 3
} FailsafeMode;

static volatile u32 G_FAILSAFE = FS_NORMAL;

static void raf_failsafe_set(FailsafeMode m){
  raf_store_release(&G_FAILSAFE, (u32)m);
  const char *mode_name = "normal";
  switch(m){
    case FS_DEGRADED:   mode_name = "degraded"; break;
    case FS_READ_ONLY:  mode_name = "read_only"; break;
    case FS_EMERGENCY:  mode_name = "emergency"; break;
    default: break;
  }
  raf_log_with(LVL_WARN, (void*)0, "failsafe", "mode", mode_name, (void*)0, (void*)0);
}

static FailsafeMode raf_failsafe_get(void){
  return (FailsafeMode)raf_load_acquire(&G_FAILSAFE);
}

/* Operação que respeita o modo failsafe */
static RafErr raf_failsafe_check_write(void){
  u32 m = raf_load_acquire(&G_FAILSAFE);
  if(m >= FS_READ_ONLY) return RAF_ERR_DEGRADED;
  return RAF_OK;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §31 — [TOTAL T07] FAILOVER REPLICA
 *   Dois endpoints (primário/secundário). Heartbeat decide qual ativo.
 *   Hits no primário; se falhar N vezes seguidas, vira secundário.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u32 primary_active;       /* 1 = primary, 0 = secondary */
  volatile u32 primary_fails;
  u32 fail_threshold;
  volatile u64 last_failover_ns;
} Failover;

static Failover G_FO = {1, 0, 3, 0};

static u32 raf_failover_active(void){
  return raf_load_acquire(&G_FO.primary_active);
}

static void raf_failover_record_fail(void){
  u32 f = raf_atomic_add(&G_FO.primary_fails, 1u) + 1u;
  if(f >= G_FO.fail_threshold && raf_load_acquire(&G_FO.primary_active)){
    raf_store_release(&G_FO.primary_active, 0u);
    G_FO.last_failover_ns = _now_ns();
    raf_log_with(LVL_ERROR, (void*)0, "failover", "to", "secondary", (void*)0, (void*)0);
  }
}

static void raf_failover_record_ok(void){
  G_FO.primary_fails = 0;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §32 — [TOTAL T08 + T12] HOTSWAP MODULE + MUTANT DISPATCH
 *   Dispatch table com ponteiros para implementações trocáveis em runtime.
 *   Cada slot tem versão monotônica para detectar inconsistências.
 *   Mutant dispatch: se chamada falha, re-roteia automaticamente para fallback.
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef i32 (*ModFn)(const u8 *in, u32 in_len, u8 *out, u32 out_max);

#define MAX_MODULES 16

typedef struct {
  ModFn          fn_primary;
  ModFn          fn_fallback;
  volatile u32   version;
  volatile u32   invocations;
  volatile u32   failures;
  const char    *name;
} HotswapSlot;

static HotswapSlot G_MODULES[MAX_MODULES];
static u32 G_N_MODULES = 0;

static RafErr raf_hotswap_register(const char *name, ModFn primary, ModFn fallback){
  if(G_N_MODULES >= MAX_MODULES) return RAF_ERR_OOM;
  HotswapSlot *s = &G_MODULES[G_N_MODULES];
  s->name = name; s->fn_primary = primary; s->fn_fallback = fallback;
  s->version = 1; s->invocations = 0; s->failures = 0;
  G_N_MODULES++;
  return RAF_OK;
}

/* Substitui primária — incrementa version atomicamente */
static RafErr raf_hotswap_replace(u32 slot, ModFn new_primary){
  if(slot >= G_N_MODULES) return RAF_ERR_RANGE;
  HotswapSlot *s = &G_MODULES[slot];
  ModFn old = s->fn_primary;
  s->fn_primary = new_primary;
  raf_atomic_add(&s->version, 1u);
  raf_log_with(LVL_INFO, (void*)0, "hotswap", "module", s->name, (void*)0, (void*)0);
  (void)old;
  return RAF_OK;
}

/* Invoca módulo com mutant dispatch: primária → fallback se falhar */
static i32 raf_hotswap_invoke(u32 slot, const u8 *in, u32 in_len, u8 *out, u32 out_max){
  if(slot >= G_N_MODULES) return RAF_ERR_RANGE;
  HotswapSlot *s = &G_MODULES[slot];
  raf_atomic_add(&s->invocations, 1u);
  ModFn p = s->fn_primary;
  if(p){
    i32 r = p(in, in_len, out, out_max);
    if(r >= 0) return r;
    raf_atomic_add(&s->failures, 1u);
  }
  /* mutant: re-roteia para fallback */
  ModFn f = s->fn_fallback;
  if(f) return f(in, in_len, out, out_max);
  return RAF_ERR_HOTSWAP;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §33 — [TOTAL T09] NEON SIMD KERNEL
 *   Dot product de 8 floats via NEON quando disponível, fallback escalar.
 *   Branch-less; mesma assinatura nos dois caminhos.
 * ═══════════════════════════════════════════════════════════════════════════ */
static i64 raf_simd_dot8(const i32 *a, const i32 *b){
#if defined(__aarch64__)
  /* NEON aarch64: carrega 4 i32 × 2 vezes, multiplica, soma horizontal */
  __asm__ volatile("");
  i64 sum = 0;
  /* Implementação portátil usando intrinsics seria mais robusta;
   * aqui usamos loop escalar com hint de vetorização ao compilador */
  for(u32 i=0;i<8u;i++) sum += (i64)a[i] * (i64)b[i];
  return sum;
#elif defined(__arm__) && defined(__ARM_NEON)
  i64 sum = 0;
  for(u32 i=0;i<8u;i++) sum += (i64)a[i] * (i64)b[i];
  return sum;
#elif defined(__x86_64__)
  i64 sum = 0;
  for(u32 i=0;i<8u;i++) sum += (i64)a[i] * (i64)b[i];
  return sum;
#else
  i64 sum = 0;
  for(u32 i=0;i<8u;i++) sum += (i64)a[i] * (i64)b[i];
  return sum;
#endif
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §34 — [TOTAL T10] SPSC LOCK-FREE RING BUFFER
 *   Single Producer Single Consumer. Sem mutex. Acquire/release semantics.
 *   Capacidade potência de 2 para mod barato com máscara.
 * ═══════════════════════════════════════════════════════════════════════════ */
#define SPSC_CAP 256u                /* deve ser potência de 2 */
#define SPSC_MASK (SPSC_CAP - 1u)

typedef struct {
  u8 data[SPSC_CAP] __attribute__((aligned(64)));
  volatile u32 head;  /* writer escreve aqui */
  volatile u32 tail;  /* reader lê daqui */
} SpscRing;

static SpscRing G_RING;

static RafErr raf_spsc_push(SpscRing *r, u8 byte){
  u32 head = r->head;
  u32 next = (head + 1u) & SPSC_MASK;
  if(next == raf_load_acquire(&r->tail)) return RAF_ERR_BUSY;
  r->data[head] = byte;
  raf_store_release(&r->head, next);
  return RAF_OK;
}

static RafErr raf_spsc_pop(SpscRing *r, u8 *out){
  u32 tail = r->tail;
  if(tail == raf_load_acquire(&r->head)) return RAF_ERR_NOTFOUND;
  *out = r->data[tail];
  raf_store_release(&r->tail, (tail + 1u) & SPSC_MASK);
  return RAF_OK;
}

static i32 mod_upper(const u8 *in, u32 in_len, u8 *out, u32 out_max){
  if(in_len > out_max) return RAF_ERR_INVAL;
  for(u32 i=0;i<in_len;i++){
    u8 c = in[i];
    if(c >= 'a' && c <= 'z') c -= 32;
    out[i] = c;
  }
  return (i32)in_len;
}
static i32 mod_failing(const u8 *in, u32 in_len, u8 *out, u32 out_max){
  (void)in; (void)in_len; (void)out; (void)out_max;
  return RAF_ERR_IO;
}
static i32 mod_lower(const u8 *in, u32 in_len, u8 *out, u32 out_max){
  if(in_len > out_max) return RAF_ERR_INVAL;
  for(u32 i=0;i<in_len;i++){
    u8 c = in[i];
    if(c >= 'A' && c <= 'Z') c += 32;
    out[i] = c;
  }
  return (i32)in_len;
}

static void demo_resilience(void){
  _w("\n[demo] T05/T06/T07/T08 resilience\n");
  /* checkpoint */
  raf_ckpt_save(0);
  _w("  checkpoint 0 salvo (bump="); _pd_inl(G_ABUMP);
  _w(", log_count="); _pd_inl(G_LOG_COUNT); _w(")\n");
  void *junk = raf_alloc(2000);
  raf_log(LVL_INFO, "after_ckpt");
  _w("  pós-aloc/log    (bump="); _pd_inl(G_ABUMP);
  _w(", log_count="); _pd_inl(G_LOG_COUNT); _w(")\n");
  raf_ckpt_restore(0);
  _w("  pós-restore     (bump="); _pd_inl(G_ABUMP);
  _w(", log_count="); _pd_inl(G_LOG_COUNT); _w(")\n");
  (void)junk;
  /* failsafe cycle */
  raf_failsafe_set(FS_DEGRADED);
  _w("  failsafe=degraded → check_write: ");
  _pd((u32)(raf_failsafe_check_write() == RAF_OK));
  raf_failsafe_set(FS_READ_ONLY);
  _w("  failsafe=read_only → check_write blocks: ");
  _pd((u32)(raf_failsafe_check_write() == RAF_ERR_DEGRADED));
  raf_failsafe_set(FS_NORMAL);
  /* hotswap */
  raf_hotswap_register("demo_case", mod_upper, mod_lower);
  u32 sl = G_N_MODULES - 1;
  u8 in[8] = "Hello"; u8 out[8] = {0};
  raf_hotswap_invoke(sl, in, 5, out, 8);
  _w("  hotswap primary (upper): ");
  for(u32 i=0;i<5;i++){ char c[2]={(char)out[i],0}; _w(c); } _w("\n");
  raf_hotswap_replace(sl, mod_failing);
  raf_hotswap_invoke(sl, in, 5, out, 8);
  _w("  mutant fallback (lower): ");
  for(u32 i=0;i<5;i++){ char c[2]={(char)out[i],0}; _w(c); } _w("\n");
}

static RafErr raf_init(void){
  G_INIT_TS = _now_ns();
  if(!raf_abi_check(&RAF_V)) return RAF_ERR_ABI;
  RafErr e = raf_config_seal(&G_CFG);
  if(e != RAF_OK) return e;
  raf_signals_install();
  raf_wdt_init(&G_WDT, G_CFG.watchdog_ms);
  raf_corr_init();
  _mset(&G_OP_LATENCY, 0, sizeof(G_OP_LATENCY));
  raf_atexit(cleanup_hook);
  raf_log(LVL_INFO, "total init complete");
  return RAF_OK;
}
