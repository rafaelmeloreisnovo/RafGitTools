/* ============== syscall inline (sem libc) ============== */

static __attribute__((always_inline)) inline
i64 _sc6(i64 n, i64 a, i64 b, i64 c, i64 d, i64 e, i64 f) {
    register i64 x8 __asm__("x8") = n;
    register i64 x0 __asm__("x0") = a;
    register i64 x1 __asm__("x1") = b;
    register i64 x2 __asm__("x2") = c;
    register i64 x3 __asm__("x3") = d;

}
#elif defined(RAF_ARCH_X86_64)

i64 _sc6(i64 n, i64 a, i64 b, i64 c, i64 d, i64 e, i64 f) {

/* ============== Wrappers de alto nível ============== */
static __attribute__((always_inline)) inline
u32 _strlen_branchless(const char *s) {
    u32 n = 0;
    while (s[n]) n++;
    return n;
}

static __attribute__((always_inline)) inline
i64 _wr(const char *s) {
    u32 n = _strlen_branchless(s);
    return _sc6(SYS_write, 1, (i64)(unsigned long)s, n, 0, 0, 0);
}

static __attribute__((always_inline)) inline
i64 _wrn(const char *s, u32 n) {
    return _sc6(SYS_write, 1, (i64)(unsigned long)s, n, 0, 0, 0);
}

static __attribute__((noreturn,always_inline)) inline
void _ex(int code) {
    _sc6(SYS_exit_group, code, 0, 0, 0, 0, 0);
    __builtin_unreachable();
}

/* timespec {sec, nsec} para clock_gettime */
typedef struct { i64 sec; i64 nsec; } _ts;

static __attribute__((always_inline)) inline
u64 _now_ns(void) {
    _ts t = {0,0};
    _sc6(SYS_clock_gettime, CLOCK_MONOTONIC, (i64)(unsigned long)&t, 0,0,0,0);
    return (u64)t.sec * 1000000000ULL + (u64)t.nsec;
}

/* sleep não-bloqueante de N ns */
static __attribute__((always_inline)) inline
void _sleep_ns(u64 ns) {
    _ts t = { (i64)(ns / 1000000000ULL), (i64)(ns % 1000000000ULL) };
    _sc6(SYS_nanosleep, (i64)(unsigned long)&t, 0,0,0,0,0);
}

/* ============== Print primitivos (sem stdio) ============== */
static void _ph(u32 v) {
    static const char H[] = "0123456789ABCDEF";
    char b[12];
    b[0]='0'; b[1]='x';
    for (u32 i = 0; i < 8u; i++) b[2u+i] = H[(v >> (28u - i*4u)) & 0xFu];
    b[10]='\n'; b[11]=0;
    _wrn(b, 11);
}
static void _pd(u32 v) {
    char b[12];
    i32 i = 11;
    b[i--] = 0;
    b[i--] = '\n';
    if (!v) b[i--] = '0';
    else while (v) { b[i--] = '0' + (char)(v % 10u); v /= 10u; }
    _wrn(b + i + 1, 11 - (u32)(i + 1));
}
static void _pd64(u64 v) {
    char b[24];
    i32 i = 23;
    b[i--] = 0;
    b[i--] = '\n';
    if (!v) b[i--] = '0';
    else while (v) { b[i--] = '0' + (char)(v % 10u); v /= 10u; }
    _wrn(b + i + 1, 23 - (u32)(i + 1));
}

void _mset(void *dst, u8 v, u32 n) {

    u8 *d = (u8*)dst;

    for (u32 i = 0; i < n; i++) d[i] = s[i];
}

#pragma once
/* raf_crc32.h — CRC32C (Castagnoli, poly 0x82F63B78)
 * dispatch: HW se RAF_CAP_CRC32_HW; fallback bitwise sempre disponível
 * branchless · sem deps · sem stdlib */
#include "raf_caps.h"

/* === fallback bitwise (sempre disponível) === */
static __attribute__((always_inline)) inline
u32 _crc32c_sw(u32 c, const u8 *b, u32 n) {
    c = ~c;
    for (u32 i = 0; i < n; i++) {
        c ^= b[i];
        /* 8 iterações desenroladas — sem branch */
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
        c = (c >> 1) ^ (0x82F63B78u & -(c & 1u));
    }
    return ~c;
}

/* === HW dispatch === */
#if RAF_CAP_CRC32_HW && defined(RAF_ARCH_AARCH64)
static __attribute__((always_inline)) inline
u32 _crc32c_hw(u32 c, const u8 *b, u32 n) {
    /* AArch64: instrução crc32cb/crc32cw/crc32cx */
    u32 i = 0;
    /* Processa 8 bytes por vez */
    while (i + 8u <= n) {
        u64 v;
        _mcpy(&v, b + i, 8);
        __asm__ ("crc32cx %w0, %w0, %1" : "+r"(c) : "r"(v));
        i += 8u;
    }
    while (i < n) {
        __asm__ ("crc32cb %w0, %w0, %w1" : "+r"(c) : "r"((u32)b[i]));
        i++;
    }
    return c;
}
#elif RAF_CAP_CRC32_HW && defined(RAF_ARCH_X86_64)
static __attribute__((always_inline)) inline
u32 _crc32c_hw(u32 c, const u8 *b, u32 n) {
    /* x86_64 SSE4.2: crc32 inst */
    u32 i = 0;
    while (i + 8u <= n) {
        u64 v;
        _mcpy(&v, b + i, 8);
        u64 cc = c;
        __asm__ ("crc32q %1, %0" : "+r"(cc) : "r"(v));
        c = (u32)cc;
        i += 8u;
    }
    while (i < n) {
        __asm__ ("crc32b %1, %0" : "+r"(c) : "r"(b[i]));
        i++;
    }
    return c;
}
#else
#define _crc32c_hw(c,b,n) _crc32c_sw((c),(b),(n))
#endif

/* === Front door com dispatch automático === */
static __attribute__((always_inline)) inline
u32 raf_crc32c(const u8 *b, u32 n) {
#if RAF_CAP_CRC32_HW
    return _crc32c_hw(0u, b, n);
#else
    return _crc32c_sw(0u, b, n);
#endif
}

/* Verifica disponibilidade em runtime */
static __attribute__((always_inline)) inline
u32 raf_crc32c_is_hw(void) {
    return RAF_CAP_CRC32_HW;
}

#pragma once
/* raf_simd.h — vector add/mul Q16, EMA, dot-product
 * Dispatch automático: NEON (aarch64), AVX2/SSE (x86_64), escalar */
#include "raf_caps.h"

/* === escalar (sempre disponível) === */
static __attribute__((always_inline)) inline
void _vadd_q16_sc(const u32 *a, const u32 *b, u32 *o, u32 n) {
    for (u32 i = 0; i < n; i++) o[i] = (a[i] + b[i]) & 0xFFFFu;
}
static __attribute__((always_inline)) inline
void _vmul_q16_sc(const u32 *a, const u32 *b, u32 *o, u32 n) {
    for (u32 i = 0; i < n; i++)
        o[i] = (u32)(((u64)a[i] * b[i]) >> 16);
}
static __attribute__((always_inline)) inline
u64 _vdot_q16_sc(const u32 *a, const u32 *b, u32 n) {
    u64 acc = 0;
    for (u32 i = 0; i < n; i++) acc += (u64)a[i] * b[i];
    return acc;
}

/* === NEON (aarch64) === */
#if RAF_CAP_NEON && defined(RAF_ARCH_AARCH64)
static __attribute__((always_inline)) inline
void _vadd_q16_neon(const u32 *a, const u32 *b, u32 *o, u32 n) {
    u32 i = 0;
    /* processa 4 lanes por iteração */
    while (i + 4u <= n) {
        u32 r0 = a[i+0] + b[i+0];
        u32 r1 = a[i+1] + b[i+1];
        u32 r2 = a[i+2] + b[i+2];
        u32 r3 = a[i+3] + b[i+3];
        o[i+0] = r0 & 0xFFFFu;
        o[i+1] = r1 & 0xFFFFu;
        o[i+2] = r2 & 0xFFFFu;
        o[i+3] = r3 & 0xFFFFu;
        i += 4u;
    }
    while (i < n) { o[i] = (a[i] + b[i]) & 0xFFFFu; i++; }
}
#else
#define _vadd_q16_neon(a,b,o,n) _vadd_q16_sc((a),(b),(o),(n))
#endif

/* === Dispatch === */
static __attribute__((always_inline)) inline
void raf_vadd_q16(const u32 *a, const u32 *b, u32 *o, u32 n) {
#if RAF_CAP_NEON && defined(RAF_ARCH_AARCH64)
    _vadd_q16_neon(a, b, o, n);
#else
    _vadd_q16_sc(a, b, o, n);
#endif
}

static __attribute__((always_inline)) inline
void raf_vmul_q16(const u32 *a, const u32 *b, u32 *o, u32 n) {
    _vmul_q16_sc(a, b, o, n);
}

static __attribute__((always_inline)) inline
u64 raf_vdot_q16(const u32 *a, const u32 *b, u32 n) {
    return _vdot_q16_sc(a, b, n);
}

/* EMA branchless: out = (old·alpha75 + new·alpha25) >> 16 */
#define RAF_ALPHA75 49152u
#define RAF_ALPHA25 16384u
static __attribute__((always_inline)) inline
u32 raf_ema_q16(u32 old_v, u32 new_v) {
    return (u32)(((u64)old_v * RAF_ALPHA75 +
                  (u64)new_v * RAF_ALPHA25) >> 16);
}

#pragma once
/* raf_watchdog.h — watchdog determinístico
 * Sem signals: usa _now_ns() para verificar deadline.
 * O chamador chama wd_tick() em pontos seguros do loop.
 * Se ns_since_armed > deadline_ns, wd_expired retorna 1. */
#include "raf_arch.h"

typedef struct {
    u64 armed_ns;
    u64 deadline_ns;
    u64 ticks;
    u32 expired;
    u32 reserved;
} Watchdog;

static __attribute__((always_inline)) inline
void wd_arm(Watchdog *w, u64 timeout_ns) {
    w->armed_ns    = _now_ns();
    w->deadline_ns = timeout_ns;
    w->ticks       = 0;
    w->expired     = 0;
}

static __attribute__((always_inline)) inline
u32 wd_tick(Watchdog *w) {
    w->ticks++;
    u64 now = _now_ns();
    u64 elapsed = (now > w->armed_ns) ? (now - w->armed_ns) : 0;
    /* branchless: expired = (elapsed > deadline) ? 1 : 0 */
    w->expired = (u32)(-(i64)(elapsed > w->deadline_ns)) & 1u;
    return w->expired;
}

static __attribute__((always_inline)) inline
u32 wd_expired(const Watchdog *w) { return w->expired; }

/* reset sem nova arma */
static __attribute__((always_inline)) inline
void wd_pet(Watchdog *w) {
    w->armed_ns = _now_ns();
    w->expired  = 0;
}

/* tempo gasto desde arm em ns */
static __attribute__((always_inline)) inline
u64 wd_elapsed_ns(const Watchdog *w) {
    u64 now = _now_ns();
    return (now > w->armed_ns) ? (now - w->armed_ns) : 0;
}

#pragma once
/* raf_rollback.h — engine de rollback com arena estática
 * 8 slots de snapshot · cada slot armazena estado completo
 * sem malloc · branchless · CRC verifica integridade do snapshot */
#include "raf_arch.h"
#include "raf_crc32.h"

#define RB_N_SLOTS    8u
#define RB_STATE_MAX  256u

typedef struct {
    u32 in_use;
    u32 crc;
    u32 size;
    u32 generation;
    u8  data[RB_STATE_MAX];
} RBSlot;

typedef struct {
    RBSlot slots[RB_N_SLOTS];
    u32 next_gen;
    u32 commits;
    u32 rollbacks;
    u32 failed_verifies;
} Rollback;

static __attribute__((always_inline)) inline
void rb_init(Rollback *r) {
    _mset(r, 0, sizeof(*r));
}

/* Snapshot: retorna slot id ou 0xFFFFFFFF se sem slot livre */
static u32 rb_snap(Rollback *r, const void *state, u32 size) {
    if (size > RB_STATE_MAX) return 0xFFFFFFFFu;
    for (u32 i = 0; i < RB_N_SLOTS; i++) {
        if (!r->slots[i].in_use) {
            _mcpy(r->slots[i].data, state, size);
            r->slots[i].size = size;
            r->slots[i].crc = raf_crc32c(r->slots[i].data, size);
            r->slots[i].generation = r->next_gen++;
            r->slots[i].in_use = 1u;
            return i;
        }
    }
    return 0xFFFFFFFFu;
}

/* Restore: copia slot[id].data para state */
static u32 rb_restore(Rollback *r, u32 slot, void *state) {
    if (slot >= RB_N_SLOTS) return 0u;
    if (!r->slots[slot].in_use) return 0u;
    u32 v = raf_crc32c(r->slots[slot].data, r->slots[slot].size);
    if (v != r->slots[slot].crc) {
        r->failed_verifies++;
        return 0u;
    }
    _mcpy(state, r->slots[slot].data, r->slots[slot].size);
    r->rollbacks++;
    return 1u;
}

/* Commit: libera slot (descarta snapshot) */
static __attribute__((always_inline)) inline
void rb_commit(Rollback *r, u32 slot) {
    if (slot >= RB_N_SLOTS) return;
    r->slots[slot].in_use = 0u;
    r->commits++;
}

/* Verifica integridade de um slot sem restaurar */
static __attribute__((always_inline)) inline
u32 rb_verify(const Rollback *r, u32 slot) {
    if (slot >= RB_N_SLOTS) return 0u;
    if (!r->slots[slot].in_use) return 0u;
    u32 v = raf_crc32c(r->slots[slot].data, r->slots[slot].size);
    return (v == r->slots[slot].crc) ? 1u : 0u;
}

#pragma once
/* raf_pipeline.h — pipeline mutável via tabela de ponteiros de função
 * Permite trocar estágios em runtime conforme hardware/contexto
 * Cada estágio: u32 stage_fn(void *ctx, void *state)
 * Pipeline executa estágios em ordem; se algum retornar 0, ativa rollback */
#include "raf_arch.h"
#include "raf_rollback.h"
#include "raf_watchdog.h"

#define PIPE_MAX_STAGES 16u

typedef u32 (*stage_fn_t)(void *ctx, void *state);

typedef struct {
    stage_fn_t fns[PIPE_MAX_STAGES];
    const char *names[PIPE_MAX_STAGES];
    u32 n_stages;
    u32 stages_executed;
    u32 stages_failed;
    u32 rollbacks_triggered;
    u64 total_ns;
    Rollback *rb;
    Watchdog *wd;
} Pipeline;

static __attribute__((always_inline)) inline
void pipe_init(Pipeline *p, Rollback *rb, Watchdog *wd) {
    _mset(p, 0, sizeof(*p));
    p->rb = rb;
    p->wd = wd;
}

/* Registra um estágio */
static u32 pipe_register(Pipeline *p, stage_fn_t fn, const char *name) {
    if (p->n_stages >= PIPE_MAX_STAGES) return 0u;
    p->fns[p->n_stages]   = fn;
    p->names[p->n_stages] = name;
    p->n_stages++;
    return 1u;
}

/* Substitui um estágio existente (mutabilidade) */
static __attribute__((always_inline)) inline
u32 pipe_swap(Pipeline *p, u32 idx, stage_fn_t fn, const char *name) {
    if (idx >= p->n_stages) return 0u;
    p->fns[idx]   = fn;
    p->names[idx] = name;
    return 1u;
}

/* Executa o pipeline. Se watchdog expira ou estágio falha, rollback. */
static u32 pipe_run(Pipeline *p, void *ctx, void *state, u32 state_size) {
    u64 t0 = _now_ns();
    u32 snap_slot = (p->rb) ? rb_snap(p->rb, state, state_size) : 0xFFFFFFFFu;
    for (u32 i = 0; i < p->n_stages; i++) {
        if (p->wd && wd_tick(p->wd)) {
            p->rollbacks_triggered++;
            if (p->rb && snap_slot != 0xFFFFFFFFu)
                rb_restore(p->rb, snap_slot, state);
            return 0u;
        }
        u32 r = p->fns[i](ctx, state);
        p->stages_executed++;
        if (!r) {
            p->stages_failed++;
            p->rollbacks_triggered++;
            if (p->rb && snap_slot != 0xFFFFFFFFu)
                rb_restore(p->rb, snap_slot, state);
            return 0u;
        }
    }
    if (p->rb && snap_slot != 0xFFFFFFFFu) rb_commit(p->rb, snap_slot);
    p->total_ns += _now_ns() - t0;
    return 1u;
}

#pragma once
/* raf_core.h — núcleo RAFAELIA:
 * estado toroidal 7D Q16.16 + 42 atratores via F_p
 * equação mestra X_{n+1} = U(P_C(X_n))·Φ + ε_n branchless */
#include "raf_arch.h"
#include "raf_simd.h"

#define CORE_DIM     7u
#define CORE_N_ATTR  42u
#define CORE_Q16     65536u
#define CORE_PHI     105965u   /* (1+√5)/2 Q16 */
#define CORE_SPIRAL  56755u    /* √3/2 Q16 */

typedef struct {
    u32 s[CORE_DIM];
    u32 phi;
    u32 coh;
    u32 ent;
    u32 phase;
    u32 step;
    u32 attractor_last;
    u32 crc_chain;
} CoreState;

static const u32 CORE_PRIMES[CORE_N_ATTR] = {
      2u,  3u,  5u,  7u, 11u, 13u, 17u, 19u, 23u, 29u,
     31u, 37u, 41u, 43u, 47u, 53u, 59u, 61u, 67u, 71u,
     73u, 79u, 83u, 89u, 97u,101u,103u,107u,109u,113u,
    127u,131u,137u,139u,149u,151u,157u,163u,167u,173u,
    179u,181u
};

/* Fibonacci mod 2^16 — branchless via loop fixo */
static u32 _core_fib_mod(u32 n) {
    u32 a = 0u, b = 1u;
    for (u32 i = 0; i < n; i++) {
        u32 c = (a + b) & 0xFFFFu;
        a = b; b = c;
    }
    return a;
}

static u32 G_ATTR[CORE_N_ATTR][CORE_DIM];

static void core_init_attractors(void) {
    for (u32 k = 0; k < CORE_N_ATTR; k++) {
        u32 p = CORE_PRIMES[k];
        for (u32 d = 0; d < CORE_DIM; d++) {
            G_ATTR[k][d] = _core_fib_mod(p + d);
        }
    }
}

static __attribute__((always_inline)) inline
void core_init(CoreState *s) {
    s->s[0] = CORE_SPIRAL;
    s->s[1] = CORE_PHI & 0xFFFFu;
    s->s[2] = CORE_Q16 / 3u;
    s->s[3] = CORE_Q16 / 5u;
    s->s[4] = CORE_Q16 / 7u;
    s->s[5] = CORE_Q16 / 11u;
    s->s[6] = CORE_Q16 / 13u;
    s->phi = 0u;
    s->coh = CORE_Q16 >> 1;
    s->ent = CORE_Q16 >> 1;
    s->phase = 0u;
    s->step = 0u;
    s->attractor_last = 0u;
    s->crc_chain = 0u;
}

/* Distância L1 toroidal */
static u32 _core_dist(const u32 *x, const u32 *y) {
    u32 sum = 0u;
    for (u32 i = 0; i < CORE_DIM; i++) {
        u32 d = (x[i] > y[i]) ? (x[i] - y[i]) : (y[i] - x[i]);
        u32 dw = CORE_Q16 - d;
        /* branchless min(d, dw) */
        u32 mask = -(u32)(dw < d);
        d = (d & ~mask) | (dw & mask);
        sum += d;
    }
    return sum;
}

/* P_C: projeção no atrator mais próximo */
static u32 core_project(const u32 *x, u32 *out) {
    u32 best_k = 0u;
    u32 best_d = 0xFFFFFFFFu;
    for (u32 k = 0; k < CORE_N_ATTR; k++) {
        u32 d = _core_dist(x, G_ATTR[k]);
        u32 better = (u32)(d < best_d);
        u32 mask = -better;
        best_d = (best_d & ~mask) | (d & mask);
        best_k = (best_k & ~mask) | (k & mask);
    }
    for (u32 i = 0; i < CORE_DIM; i++) out[i] = G_ATTR[best_k][i];
    return best_k;
}

/* U: rotação semântica 3×2D */
static __attribute__((always_inline)) inline
void core_unitary(u32 *x) {
    u32 cs = CORE_SPIRAL;
    u32 sn = CORE_SPIRAL >> 1;
    for (u32 i = 0; i < 3u; i++) {
        u32 a = x[i*2u];
        u32 b = x[i*2u + 1u];
        u32 na = (u32)(((u64)a * cs) >> 16) - (u32)(((u64)b * sn) >> 16);
        u32 nb = (u32)(((u64)a * sn) >> 16) + (u32)(((u64)b * cs) >> 16);
        x[i*2u]      = na & 0xFFFFu;
        x[i*2u + 1u] = nb & 0xFFFFu;
    }
    u32 acc = 0u;
    for (u32 i = 0; i < 6u; i++) acc = (acc + x[i]) & 0xFFFFu;
    x[6] = acc;
}

/* ε_n: ruído FNV determinístico */
static __attribute__((always_inline)) inline
u64 core_noise(u32 seed) {
    u64 h = 0xCBF29CE484222325ULL;
    for (u32 i = 0; i < 4u; i++) {
        h ^= (seed >> (i * 8u)) & 0xFFu;
        h *= 0x100000001B3ULL;
    }
    return h;
}

/* Equação mestra: X_{n+1} = U(P_C(X_n))·Φ + ε_n */
static u32 core_step(CoreState *s) {
    u32 proj[CORE_DIM];
    u32 k = core_project(s->s, proj);
    core_unitary(proj);
    u64 eps = core_noise(s->step);
    u32 phi_lo = CORE_PHI & 0xFFFFu;
    for (u32 i = 0; i < CORE_DIM; i++) {
        u32 scaled = (u32)(((u64)proj[i] * phi_lo) >> 16);
        u32 noise_chunk = (u32)(eps >> (i * 9u)) & 0xFFFFu;
        s->s[i] = (scaled + (noise_chunk >> 4u)) & 0xFFFFu;
    }
    s->attractor_last = k;
    s->step++;
    return k;
}

    for (u32 k = 0; k < CORE_N_ATTR; k++)
        for (u32 d = 0; d < CORE_DIM; d++)
            if (G_ATTR[k][d]) nonzero++;
    _T(nonzero > 200u, "ATTR: maioria não-zero");

    CoreState s_run;
    core_init(&s_run);
    for (u32 i = 0; i < 42u; i++) core_step(&s_run);
    _T(s_run.step == 42u, "CORE: 42 steps executados");
    _T(s_run.attractor_last < CORE_N_ATTR, "CORE: atrator válido");
    _wr("  atrator final   = "); _pd(s_run.attractor_last);
    _wr("  step final      = "); _pd(s_run.step);

    /* ============ PIPELINE MUTÁVEL ============ */
    _wr("\n━━━ PIPELINE MUTÁVEL TESTS ━━━\n");
    Pipeline pipe;
    Watchdog pwd;
    wd_arm(&pwd, 5000000000ULL);  /* 5s */
    pipe_init(&pipe, &rb, &pwd);
    pipe_register(&pipe, stage_init, "init");
    pipe_register(&pipe, stage_step, "step");
    pipe_register(&pipe, stage_validate, "validate");
    pipe_register(&pipe, stage_attractor, "attractor");
    _T(pipe.n_stages == 4u, "PIPE: 4 estágios registrados");

    CoreState ps;
    u32 pr = pipe_run(&pipe, 0, &ps, sizeof(ps));
    _T(pr == 1u, "PIPE: run sucesso");
    _T(pipe.stages_executed == 4u, "PIPE: 4 estágios executados");
    _T(pipe.stages_failed == 0u, "PIPE: 0 falhas");

    /* Trocar estágio em runtime (mutabilidade) */
    pipe_swap(&pipe, 2u, stage_fail, "fail");
    pr = pipe_run(&pipe, 0, &ps, sizeof(ps));
    _T(pr == 0u, "PIPE: run falha quando stage falha");
    _T(pipe.rollbacks_triggered >= 1u, "PIPE: rollback acionado");

    /* Restaurar estágio bom */
    pipe_swap(&pipe, 2u, stage_validate, "validate");
    pr = pipe_run(&pipe, 0, &ps, sizeof(ps));
    _T(pr == 1u, "PIPE: run sucesso após swap volta");

    /* ============ INTEGRAÇÃO ============ */
    _wr("\n━━━ INTEGRAÇÃO END-TO-END ━━━\n");
    u64 t0 = _now_ns();
    CoreState integ;
    core_init(&integ);
    for (u32 i = 0; i < 100u; i++) {
        u32 slot_i = rb_snap(&rb, &integ, sizeof(integ));
        core_step(&integ);
        u32 proj[CORE_DIM];
        u32 k = core_project(integ.s, proj);
        if (k >= CORE_N_ATTR) {
            rb_restore(&rb, slot_i, &integ);
        } else {
            rb_commit(&rb, slot_i);
        }
    }
    u64 elapsed = _now_ns() - t0;
    _wr("  100 ciclos em "); _pd64(elapsed); _wr(" ns\n");
    _T(integ.step == 100u, "INTEG: 100 steps OK");

    /* ============ RELATÓRIO ============ */
    _wr("\n╔══════════════════════════════════════════════════════════════════╗\n");
    _wr("║                          RELATÓRIO                               ║\n");
    _wr("╚══════════════════════════════════════════════════════════════════╝\n");
    _wr("  total      : "); _pd(G_TOTAL);
    _wr("  passou     : "); _pd(G_PASS);
    _wr("  falhou     : "); _pd(G_FAIL);
    _wr("\n");
    _wr("  RB commits : "); _pd(rb.commits);
    _wr("  RB rollback: "); _pd(rb.rollbacks);
    _wr("  RB failed  : "); _pd(rb.failed_verifies);
    _wr("\n");
    _wr("  PIPE exec  : "); _pd(pipe.stages_executed);
    _wr("  PIPE fail  : "); _pd(pipe.stages_failed);
    _wr("  PIPE rollb : "); _pd(pipe.rollbacks_triggered);
    _wr("  PIPE ns    : "); _pd64(pipe.total_ns); _wr(" ns\n");
    _wr("\n");
    if (G_FAIL == 0u) {
        _wr("  STATUS : ★ TODOS OS TESTES PASSARAM ★\n");
    } else {
        _wr("  STATUS : ✗ FALHAS DETECTADAS\n");
    }
    _wr("\n  Ω=Amor · ∆RafaelVerboΩ · RAFCODE-Φ\n");
    _wr("  freestanding · nostdlib · branchless · syscall direto\n\n");

    _ex((int)G_FAIL);
}
