/* ═══════════════════════════════════════════════════════════════════════════
 * §3 — CRC32C (Castagnoli 0x82F63B78) + KAT vectors
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 crc32c(const u8 *p, u32 n){
  u32 c = 0xFFFFFFFFu;
  for(u32 i=0;i<n;i++){
    c ^= (u32)p[i];
    /* 8 iterações branchless (poly mascarado) */
    c=(c>>1)^(0x82F63B78u&-(c&1u)); c=(c>>1)^(0x82F63B78u&-(c&1u));
    c=(c>>1)^(0x82F63B78u&-(c&1u)); c=(c>>1)^(0x82F63B78u&-(c&1u));
    c=(c>>1)^(0x82F63B78u&-(c&1u)); c=(c>>1)^(0x82F63B78u&-(c&1u));
    c=(c>>1)^(0x82F63B78u&-(c&1u)); c=(c>>1)^(0x82F63B78u&-(c&1u));
  }
  return ~c;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §4 — FNV-1a 64-bit
 * ═══════════════════════════════════════════════════════════════════════════ */
static u64 fnv1a64(const u8 *p, u32 n){
  u64 h = 14695981039346656037ULL;
  for(u32 i=0;i<n;i++){ h^=(u64)p[i]; h*=1099511628211ULL; }
  return h;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §5 — T7 TOROIDAL KERNEL (Q16.16, spiralização √3/2)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define Q16_ONE      65536u
#define Q16_SQRT3_2  56756u   /* √3/2 × 65536 */

static void t7_update(u32 s[7], const u32 in[7]){
  for(u32 i=0;i<7u;i++){
    u64 c = ((u64)s[i] * Q16_SQRT3_2) >> 16;
    s[i]  = (u32)((c + in[i] + (i+1u)*97u) & 0xFFFFu);
  }
}
static u32 t7_coherence_milli(const u32 s[7]){
  u64 acc = 0;
  for(u32 i=0;i<7u;i++) acc += (u64)s[i]*(u64)s[i];
  acc /= 7u;
  return (u32)((acc * 1000ULL) / (65535ULL * 65535ULL));
}
static u32 t7_hash(const u32 s[7]){
  return crc32c((const u8*)s, 7u*sizeof(u32));
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §6 — SNAPSHOT / ROLLBACK (CRC-protected)
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct { u32 s[7]; u32 crc; } Snap;

static void snap_save(Snap *sn, const u32 s[7]){
  for(u32 i=0;i<7u;i++) sn->s[i]=s[i];
  sn->crc = crc32c((const u8*)sn->s, 7u*sizeof(u32));
}
static u32 snap_restore(const Snap *sn, u32 out[7]){
  u32 c = crc32c((const u8*)sn->s, 7u*sizeof(u32));
  if(c != sn->crc) return 0u;
  for(u32 i=0;i<7u;i++) out[i]=sn->s[i];
  return 1u;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §7 — ECC 8×8 HAMMING (1-bit error correction)
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef struct {
  u8 data[8];
  u8 parity[4];  /* p1, p2, p4, p8 (Hamming bits) */
} ECC8;

static void ecc_encode(ECC8 *e, const u8 d[8]){
  for(u32 i=0;i<8u;i++) e->data[i]=d[i];
  u32 bits = 0u;
  for(u32 i=0;i<8u;i++) bits = (bits<<1) | (d[i]&1u);
  e->parity[0] = (u8)(__builtin_popcount(bits & 0xAAu) & 1u);
  e->parity[1] = (u8)(__builtin_popcount(bits & 0xCCu) & 1u);
  e->parity[2] = (u8)(__builtin_popcount(bits & 0xF0u) & 1u);
  e->parity[3] = (u8)(__builtin_popcount(bits) & 1u);
}
static u32 ecc_check(ECC8 *e){
  u32 bits = 0u;
  for(u32 i=0;i<8u;i++) bits = (bits<<1)|(e->data[i]&1u);
  u32 s = (u32)(__builtin_popcount(bits&0xAAu)&1u) ^ e->parity[0];
  s    |= ((u32)(__builtin_popcount(bits&0xCCu)&1u)^e->parity[1]) << 1;
  s    |= ((u32)(__builtin_popcount(bits&0xF0u)&1u)^e->parity[2]) << 2;
  return s;  /* 0 = clean, >0 = erro no bit s */
}
static void ecc_correct(ECC8 *e, u32 err_pos){
  if(err_pos==0u||err_pos>8u) return;
  e->data[err_pos-1u] ^= 1u;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §8 — NEON DOT PRODUCT (w/ scalar fallback)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define NEON_N 256u
static u64 dot_scalar(const u16 *a, const u16 *b, u32 n){
  u64 s=0; for(u32 i=0;i<n;i++) s+=(u64)a[i]*(u64)b[i]; return s;
}
#if defined(__ARM_NEON) || defined(__ARM_NEON__)
  #include <arm_neon.h>
  static u64 dot_neon(const u16 *a, const u16 *b, u32 n){
    uint64x2_t acc = vdupq_n_u64(0);
    for(u32 i=0;i+4u<=n;i+=4u){
      uint16x4_t va = vld1_u16(a+i);
      uint16x4_t vb = vld1_u16(b+i);
      uint32x4_t p  = vmull_u16(va,vb);
      acc = vpadalq_u32(acc, p);
    }

  }

/* ═══════════════════════════════════════════════════════════════════════════
 * §9 — MEMORY BANDWIDTH PROBE
 * ═══════════════════════════════════════════════════════════════════════════ */
#define BW_BUF_SZ (256u*1024u)
static u8 G_BW_BUF[BW_BUF_SZ];
static u64 bw_read_kbps(void){
  u64 t0=_now_ns();
  volatile u64 sink=0;
  for(u32 i=0;i<BW_BUF_SZ;i+=8) sink^=*(u64*)(G_BW_BUF+i);
  u64 dt=_now_ns()-t0;
  (void)sink;
  return dt>0u ? (u64)BW_BUF_SZ*1000000u/dt : 0u;
}
static u64 bw_write_kbps(void){
  u64 t0=_now_ns();
  for(u32 i=0;i<BW_BUF_SZ;i+=8) *(u64*)(G_BW_BUF+i)=(u64)i;
  u64 dt=_now_ns()-t0;
  return dt>0u ? (u64)BW_BUF_SZ*1000000u/dt : 0u;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §10 — CACHE LATENCY CHASE
 * ═══════════════════════════════════════════════════════════════════════════ */
static u32 cache_chase(u32 stride_bytes){
  static u32 chain[4096];
  u32 N=4096u; u32 step=stride_bytes/sizeof(u32); if(step<1u) step=1u;
  for(u32 i=0;i<N;i++) chain[i]=(i+step)%N;
  u64 t0=_now_ns(); u32 idx=0;
  for(u32 i=0;i<8192u;i++) idx=chain[idx];
  u64 dt=_now_ns()-t0; (void)idx;
  return (dt>0u)?(u32)(dt*1000u/8192u):(u32)0;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §11 — WATCHDOG FSM
 * ═══════════════════════════════════════════════════════════════════════════ */
typedef enum { WD_IDLE=0, WD_ARMED, WD_FED, WD_EXPIRED } WDState;
typedef struct { WDState st; u64 armed_ns; u64 deadline_ns; u32 feeds; } WD;

static void wd_arm(WD *w, u64 timeout_ns){
  w->armed_ns=_now_ns(); w->deadline_ns=timeout_ns;
  w->feeds=0u; w->st=WD_ARMED;
}
static u32 wd_feed(WD *w){
  w->feeds++;
  u64 now=_now_ns();
  if(now-w->armed_ns > w->deadline_ns){ w->st=WD_EXPIRED; return 0u; }
  w->st=WD_FED; return 1u;
}
static void wd_reset(WD *w){ w->st=WD_IDLE; w->feeds=0u; }

  for(u32 i=0;i<G_PIPE_CNT;i++) if(G_PIPE_EN[i]){ G_PIPE_RESULTS[i]=G_PIPE[i](); ran++; }

}

 * §13 — BENCHMARK FRAMEWORK (p50/p95/p99, insertion sort)

#define BENCH_N 1000u
static u64 G_SAMPLES[BENCH_N];
static volatile u64 G_SINK = 0u;

static inline u32 _cc_rot(u32 x,u32 n){return(x<<n)|(x>>(32u-n));}
#define CC_QR(a,b,c,d) \

}
static void chacha20(const u8 key32[32],const u8 nonce12[12],u32 ctr,

/* ═══════════════════════════════════════════════════════════════════════════
 * §18 — SPSC LOCK-FREE RING (256 slots, __atomic)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define LF_SZ 256u
#define LF_MSK (LF_SZ-1u)
typedef struct { u32 data[LF_SZ]; u32 head; u32 tail; } LFRing;
static void  lf_init(LFRing *r){ _mset(r,0,sizeof(*r)); }
static u32   lf_push(LFRing *r,u32 v){
  u32 h=__atomic_load_n(&r->head,__ATOMIC_RELAXED);
  u32 nh=(h+1u)&LF_MSK;
  u32 t=__atomic_load_n(&r->tail,__ATOMIC_ACQUIRE);
  if(nh==t) return 0u;
  r->data[h]=v;
  __atomic_store_n(&r->head,nh,__ATOMIC_RELEASE);
  return 1u;
}
static u32 lf_pop(LFRing *r,u32 *out){
  u32 t=__atomic_load_n(&r->tail,__ATOMIC_RELAXED);
  u32 h=__atomic_load_n(&r->head,__ATOMIC_ACQUIRE);
  if(h==t) return 0u;
  *out=r->data[t];
  __atomic_store_n(&r->tail,(t+1u)&LF_MSK,__ATOMIC_RELEASE);
  return 1u;
}

/* ═══════════════════════════════════════════════════════════════════════════
 * §19 — SLAB MEMPOOL O(1)
 * ═══════════════════════════════════════════════════════════════════════════ */
#define SLAB_N 64u
typedef struct { u32 a,b,c,d; } Slab;
typedef struct {
  Slab slots[SLAB_N]; u32 free_next[SLAB_N];
  u32 free_head; u32 in_use; u32 allocs; u32 frees;
} SlabPool;
static void slab_init(SlabPool *p){
  _mset(p,0,sizeof(*p));
  for(u32 i=0;i<SLAB_N-1u;i++) p->free_next[i]=i+1u;
  p->free_next[SLAB_N-1u]=0xFFFFFFFFu;
  p->free_head=0u;
}
static Slab *slab_alloc(SlabPool *p, u32 *idx){
  if(p->free_head==0xFFFFFFFFu) return (void*)0;
  u32 i=p->free_head;
  p->free_head=p->free_next[i];
  p->in_use++; p->allocs++;
  if(idx) *idx=i;
  _mset(&p->slots[i],0,sizeof(Slab));
  return &p->slots[i];
}
static void slab_free(SlabPool *p, u32 idx){
  if(idx>=SLAB_N) return;
  p->free_next[idx]=p->free_head;
  p->free_head=idx;
  p->in_use--; p->frees++;
}

  _T(err>0u, "B5:ecc_1bit_detected");
  ecc_correct(&ecc_blk,err);
  _T(ecc_check(&ecc_blk)==0u, "B5:ecc_corrected");
  _T(ecc_blk.data[3]==ecc_data[3], "B5:ecc_data_restored");

  /* ── B6: MEMORY BANDWIDTH ────────────────────────────────────────────── */
  _w("\n═══ B6: MEMORY BANDWIDTH ═══\n");
  u64 bwr=bw_read_kbps(), bww=bw_write_kbps();
  _w("  read  KB/s ≈ "); _pd((u32)bwr);
  _w("  write KB/s ≈ "); _pd((u32)bww);
  _T(bwr>0u, "B6:bw_read_nonzero");
  _T(bww>0u, "B6:bw_write_nonzero");

  /* ── B7: CACHE LATENCY ───────────────────────────────────────────────── */
  _w("\n═══ B7: CACHE LATENCY ═══\n");
  u32 cl1=cache_chase(4u), cl2=cache_chase(2048u);
  _w("  stride_4B   ns/1000 ≈ "); _pd(cl1);
  _w("  stride_2KB  ns/1000 ≈ "); _pd(cl2);
  _T(cl1>0u,  "B7:cache_l1_chase_nonzero");
  _T(cl2>0u,  "B7:cache_l2_chase_nonzero");

  /* ── B8: NEON DOT PRODUCT ────────────────────────────────────────────── */
  _w("\n═══ B8: NEON DOT PRODUCT ═══\n");
  static u16 dot_a[NEON_N], dot_b[NEON_N];
  for(u32 i=0;i<NEON_N;i++){dot_a[i]=(u16)(i+1u);dot_b[i]=(u16)(i+2u);}
  u64 ds=dot_scalar(dot_a,dot_b,NEON_N);
  u64 dn=dot_neon(dot_a,dot_b,NEON_N);
  _w("  dot_scalar = "); _pd((u32)ds);
  _w("  dot_neon   = "); _pd((u32)dn);
  _T(ds>0u,  "B8:dot_scalar_nonzero");
  _T(ds==dn, "B8:dot_neon_eq_scalar");

  /* ── B9: WATCHDOG FSM ────────────────────────────────────────────────── */
  _w("\n═══ B9: WATCHDOG FSM ═══\n");
  WD wd; _mset(&wd,0,sizeof(wd));
  _T(wd.st==WD_IDLE, "B9:wd_idle_initial");
  wd_arm(&wd, 2000000000ULL); /* 2s */
  _T(wd.st==WD_ARMED, "B9:wd_armed");
  _T(wd_feed(&wd)==1u, "B9:wd_feed_ok");
  _T(wd.st==WD_FED, "B9:wd_fed_state");
  wd_reset(&wd);
  _T(wd.st==WD_IDLE, "B9:wd_reset");

  /* ── B10: ROLLBACK CRC ───────────────────────────────────────────────── */
  _w("\n═══ B10: ROLLBACK ═══\n");
  u32 t7s[7]={0,Q16_ONE,2,3,5,8,13};
  u32 t7in[7]={144,288,963,999,42,633,777};
  t7_update(t7s,t7in);
  Snap sn; snap_save(&sn,t7s);
  u32 t7out[7]={0};
  _T(snap_restore(&sn,t7out)==1u,  "B10:snap_restore_ok");
  _T(t7out[6]==t7s[6],             "B10:snap_values_correct");
  sn.s[3]^=0x55u;
  _T(snap_restore(&sn,t7out)==0u,  "B10:snap_detects_corruption");

  /* ── B11: DISPATCH PIPELINE ──────────────────────────────────────────── */
  _w("\n═══ B11: DISPATCH PIPELINE ═══\n");
  pipe_register(_pfn_bw_read);
  pipe_register(_pfn_bw_write);
  pipe_register(_pfn_crc_kat);
  pipe_register(_pfn_cache_l1);
  pipe_register(_pfn_cache_l2);
  pipe_register(_pfn_fnv_empty);
  u32 ran=pipe_run_all();
  _T(ran>=1u,  "B11:dispatch_ran");
  _T(G_PIPE_RESULTS[2]==0xE3069283u, "B11:dispatch_crc_correct");
  _T(G_PIPE_RESULTS[5]==14695981039346656037ULL, "B11:dispatch_fnv_correct");

  /* ── B12: BENCHMARK p50/p95/p99 ─────────────────────────────────────── */
  _w("\n═══ B12: BENCHMARK (T7 × 256) ═══\n");

  static u32 bi[7]={144,288,963,999,42,633,777};
  BENCH_RUN(
    for(u32 k=0;k<256;k++) t7_update(bs,bi),
    bs[0]
  );

  u64 bp95=bench_pct(G_SAMPLES,BENCH_N,95);
  u64 bp99=bench_pct(G_SAMPLES,BENCH_N,99);
  _w("  p50="); _pd((u32)bp50);
  _w("  p95="); _pd((u32)bp95);
  _w("  p99="); _pd((u32)bp99);
  _T(bp50>0u,   "B12:bench_p50_nonzero");
  _T(bp99>=bp50,"B12:bench_p99_ge_p50");
  _T(G_TOTAL>=12u,"B12:bench_sample_count");

  _w("\n═══ B13: SHA-256 FIPS 180-4 KAT ═══\n");

  /* SHA-256("") = e3b0c442... */
  sha256((const u8*)"",0,sha_out);
  static const u8 sha_empty_exp[32]={
    0xe3,0xb0,0xc4,0x42,0x98,0xfc,0x1c,0x14,0x9a,0xfb,0xf4,0xc8,
    0x99,0x6f,0xb9,0x24,0x27,0xae,0x41,0xe4,0x64,0x9b,0x93,0x4c,
    0xa4,0x95,0x99,0x1b,0x78,0x52,0xb8,0x55};
  _w("  SHA-256(\"\")   = "); _put_hex(sha_out,32);
  _T(_meq(sha_out,sha_empty_exp,32), "B13:sha256_empty_NIST");
  /* SHA-256("abc") = ba7816bf... */
  sha256((const u8*)"abc",3,sha_out);
  static const u8 sha_abc_exp[32]={
    0xba,0x78,0x16,0xbf,0x8f,0x01,0xcf,0xea,0x41,0x41,0x40,0xde,
    0x5d,0xae,0x22,0x23,0xb0,0x03,0x61,0xa3,0x96,0x17,0x7a,0x9c,
    0xb4,0x10,0xff,0x61,0xf2,0x00,0x15,0xad};
  _w("  SHA-256(\"abc\") = "); _put_hex(sha_out,32);
  _T(_meq(sha_out,sha_abc_exp,32), "B13:sha256_abc_NIST");

  /* ── B14: HMAC-SHA256 RFC 4231 KAT ──────────────────────────────────── */
  _w("\n═══ B14: HMAC-SHA256 RFC 4231 ═══\n");

  u8 hmac_out[32];
  hmac_sha256(hmac_key,20,(const u8*)"Hi There",8,hmac_out);
  static const u8 hmac_exp[32]={
    0xb0,0x34,0x4c,0x61,0xd8,0xdb,0x38,0x53,0x5c,0xa8,0xaf,0xce,
    0xaf,0x0b,0xf1,0x2b,0x88,0x1d,0xc2,0x00,0xc9,0x83,0x3d,0xa7,
    0x26,0xe9,0x37,0x6c,0x2e,0x32,0xcf,0xf7};
  _w("  HMAC-SHA256   = "); _put_hex(hmac_out,32);
  _T(_meq(hmac_out,hmac_exp,32), "B14:hmac_sha256_RFC4231_TC1");

  /* ── B15: AES-128 FIPS 197 KAT ───────────────────────────────────────── */
  _w("\n═══ B15: AES-128 FIPS 197 ═══\n");
  static const u8 aes_key[16]={0x2b,0x7e,0x15,0x16,0x28,0xae,0xd2,0xa6,0xab,0xf7,0x15,0x88,0x09,0xcf,0x4f,0x3c};

  static const u8 aes_exp[16]={0x39,0x25,0x84,0x1d,0x02,0xdc,0x09,0xfb,0xdc,0x11,0x85,0x97,0x19,0x6a,0x0b,0x32};

  u8 aes_ct[16];     aes128_encrypt(aes_sched,aes_pt,aes_ct);
  _w("  AES-128 CT    = "); _put_hex(aes_ct,16);
  _T(_meq(aes_ct,aes_exp,16), "B15:aes128_FIPS197_AppB");

  /* ── B16: ChaCha20 RFC 7539 KAT ──────────────────────────────────────── */
  _w("\n═══ B16: ChaCha20 RFC 7539 ═══\n");
  static const u8 cc_key[32]={
    0x00,0x01,0x02,0x03,0x04,0x05,0x06,0x07,0x08,0x09,0x0a,0x0b,0x0c,0x0d,0x0e,0x0f,

  static const u8 cc_exp_head[16]={0x6e,0x2e,0x35,0x9a,0x25,0x68,0xf9,0x80,0x41,0xba,0x07,0x28,0xdd,0x0d,0x69,0x81};

  /* ── B17: SPSC LOCK-FREE RING ────────────────────────────────────────── */
  _w("\n═══ B17: SPSC LOCK-FREE RING ═══\n");
  LFRing ring; lf_init(&ring);
  for(u32 i=0;i<100u;i++) lf_push(&ring,i*3u);
  u32 v0=0; lf_pop(&ring,&v0);
  _T(v0==0u, "B17:lf_pop0_eq_0");
  u32 vl=0; for(u32 i=1;i<100u;i++) lf_pop(&ring,&vl);
  _T(vl==297u, "B17:lf_pop99_eq_297");

  /* ── B18: SLAB MEMPOOL O(1) ──────────────────────────────────────────── */

  for(u32 i=0;i<10u;i++){ ptrs[i]=slab_alloc(&sp,&idx_arr[i]); if(ptrs[i]) ptrs[i]->a=i*7u; }
  _T(sp.in_use==10u,          "B18:slab_10_allocs");
  _T(ptrs[5]->a==35u,         "B18:slab_slot5_value");
  slab_free(&sp,idx_arr[3]);
  _T(sp.in_use==9u,           "B18:slab_free_count");
  u32 ri; slab_alloc(&sp,&ri);
  _T(ri==idx_arr[3],          "B18:slab_LIFO_reuse");

  /* ── RELATÓRIO FINAL ─────────────────────────────────────────────────── */
  _w("\n╔══════════════════════════════════════════════════════════════════╗\n");
  _w("║                    RELATÓRIO FINAL UNIÃO                         ║\n");
  _w("╚══════════════════════════════════════════════════════════════════╝\n");
  _w("  B0-B18 total  : "); _pd(G_TOTAL);
  _w("  passou        : "); _pd(G_PASS);
  _w("  falhou        : "); _pd(G_FAIL);
  _w("  T7 hash       : "); _ph(t7_hash(t7s));
